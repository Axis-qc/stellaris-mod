#选择保存领袖
country_event = {
    id = lxleader.101
    hide_window = yes  # 隐藏窗口
    is_triggered_only = yes  # 仅在触发条件满足时触发
    trigger = {
        is_ai = no
        has_country_flag = yes_two_traits
    }
    immediate = {
        #查找符合条件的领袖
        random_owned_leader = {
            limit = {
                # 基础技能>4
                has_base_skill > 4
                # 没有特殊标记
                NOT = { has_leader_flag = has_two_traits }
                # 拥有特定特质组合
                # OR = {
                #     AND = {
                #         has_trait = leader_trait_brilliant_shipwright
                #         has_trait = leader_trait_engineer
                #     }
                #     AND = {
                #         has_trait = leader_trait_exceptional
                #         has_trait = leader_trait_archaeologist
                #     }
                # }
            }
            #保存找到的领袖
            save_event_target_as = has_two_traits_leader
            if = {
                limit = {
                    event_target:has_two_traits_leader = {
                        exists = planet
                    }
                }
                planet = {
                    save_event_target_as = has_two_traits_leade_planet
                }
            }
        }
        #指挥官
        if = {
            limit = {
                event_target:has_two_traits_leader = {
                    leader_class = commander
                }
            }
            country_event = {
                days = 30
                id = lxleader.103
            }
        }
        #行政官
        if = {
            limit = {
                event_target:has_two_traits_leader = {
                    leader_class = official
                }
            }
            country_event = {
                days = 30
                id = lxleader.103
            }
        }
        #科学官
        if = {
            limit = {
                event_target:has_two_traits_leader = {
                    leader_class = scientist
                }
            }
            country_event = {
                days = 30
                id = lxleader.103
            }
        }
        # assign_leader = event_target:a#分配领袖
    }
}
