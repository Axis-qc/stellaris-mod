axis_fleet_effect_2 = {
    #显示条件
    potential = {
        show_axis_fleet = yes
    }
    allow = {
        if = {
            limit = {
                has_country_flag = axis_fleet_1
            }
            NOT = { has_global_flag = has_axis_fleet_1 }
        }
        this = {
            is_scope_type = fleet
            NOR = {
                has_fleet_flag = axis_fleet_1
                has_fleet_flag = axis_fleet_2
                has_fleet_flag = axis_fleet_3
                has_fleet_flag = axis_fleet_4
                has_fleet_flag = axis_fleet_5
                has_fleet_flag = axis_fleet_6
            }
        }
        from = {
            OR = {
                has_country_flag = axis_fleet_1
                has_country_flag = axis_fleet_2
                has_country_flag = axis_fleet_3
                has_country_flag = axis_fleet_4
                has_country_flag = axis_fleet_5
                has_country_flag = axis_fleet_6
            }
        }
        #忘了这个是干什么的了
        # NOT = {
        #     has_global_flag = go_ship_1
        # }
    }
    effect = {
        this = {
            if = {
                limit = {
                    from = {
                        has_country_flag = axis_fleet_1
                    }
                }
                save_global_event_target_as = axis_fleet_1
                set_fleet_flag = axis_fleet_1
                #标记位置已经被占用
                from = {
                    set_global_flag = has_axis_fleet_1
                }
            }
            if = {
                limit = {
                    from = {
                        has_country_flag = axis_fleet_2
                    }
                }
                save_global_event_target_as = axis_fleet_2
                set_fleet_flag = axis_fleet_2
                #标记位置已经被占用
                from = {
                    set_global_flag = has_axis_fleet_2
                }
            }
        }
        from = {

        }
        root = {
        }
    }
}

axis_fleet_effect_3 = {
    #显示条件
    potential = {
        show_axis_fleet = yes
    }
    allow = {
        always = yes
    }
    effect = {
        from = {
            if = {
                limit = { has_country_flag = axis_fleet_1 }
                event_target:axis_fleet_1 = {
                    remove_fleet_flag = axis_fleet_1
                }
                clear_global_event_target = axis_fleet_1
                remove_global_flag = has_axis_fleet_1
            }
            if = {
                limit = { has_country_flag = axis_fleet_2 }
                event_target:axis_fleet_2 = {
                    remove_fleet_flag = axis_fleet_2
                }
                clear_global_event_target = axis_fleet_2
                remove_global_flag = has_axis_fleet_2
            }
            country_event = { id = ammunition.103 }
        }
    }
}
