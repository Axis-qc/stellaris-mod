                #     AND = {
                #         has_trait = leader_trait_exceptional
                #         has_trait = leader_trait_archaeologist
                #     }
                # }
            }
            #保存找到的领袖
            save_event_target_as = has_two_traits_leader
        }
        #指挥官
        if = {
            limit = {
                event_target:has_two_traits_leader = {
                    leader_class = commander
                }
            }
            country_event = {
                days = 30
                id = lxleader.103
            }
        }
        #行政官
        if = {

                event_target:has_two_traits_leader = {
