country_event = {
    id = planet_ck.102
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = no
        has_global_flag = play_open_planet_pop
    }

    immediate = {
        #根据住房提供人口增速加成或减成
        every_owned_planet = {
            limit = {
                not = {
                    is_planet_class = pc_cosmogenesis_world
                }
            }
            #获取可用住房
            set_variable = {
                which = planet_house
                value = trigger:free_housing
            }
            #除100，一百住房增加1%增长
            divide_variable = {
                which = planet_house
                value = 100
            }
            add_modifier = {
                modifier = planet_pop_speed_add
                mult = planet_house
                # multiplier =
                # time_multiplier =
                # clear_on_owner_change = yes
                days = -1
            }
            #住房小于0时减少人口
            if = {
                limit = { check_variable = { which = planet_house value < 0 } }
                # multiply_variable = { which = planet_house value = 3 }
                add_modifier = {
                    modifier = planet_pop_speed_die
                    mult = planet_house
                    days = -1
                }
            }
            if = {
                limit = { check_variable = { which = planet_house value > 0 } }
                remove_modifier = planet_pop_speed_die
            }
            #负50k以上直接清人口
            if = {
                limit = { check_variable = { which = planet_house value < -500 } }
                # multiply_variable = { which = planet_house value = 3 }
                random_owned_pop_group = {
                    limit = {
                        OR = {
                            is_pop_category = civilian
                            is_pop_category = maintenance_drone
                        }
                        # or = {
                        #     has_trait = trait_machine_unit
                        #     has_trait = trait_mechanical
                        # }
                    }
                    kill_pop_group = {
                        pop_group = this
                        # amount =1
                        percentage = 0.1
                        # random =
                        # growth_category =
                    }
                }
            }
        }
    }
}
