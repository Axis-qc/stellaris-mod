namespace = play_move_pop
country_event = {
	id = play_move_pop.1
	is_triggered_only = yes
	hide_window = yes
	immediate = {
		if = {
			limit = {
				# has_edict = play_move_pop_1
				any_owned_planet = {
					is_planet_class = pc_cosmogenesis_world
				}
			}
			random_owned_planet = {
				limit = {
					is_planet_class = pc_cosmogenesis_world
				}
				save_event_target_as = ATR_cosmogenesis
			}
			every_owned_planet = {
				limit = {
					NOT = {
						has_planet_flag = ATR_no_this_planet
					}
				}
				every_owned_pop_group = {
					limit = {
						OR = {
							is_pop_category = civilian
							is_pop_category = maintenance_drone
						}
						not = {
							has_trait = trait_robot_digital_2
						}
					}
					resettle_pop_group = {
						POP_GROUP = this
						PLANET = event_target:ATR_cosmogenesis
						#AMOUNT = 100%
						PERCENTAGE = 100
					}
				}
			}
		}
	}
	after = {
		country_event = {
			id = play_move_pop.1
			days = 42
		}
	}
}
