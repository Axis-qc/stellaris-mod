namespace = shipadd

country_event = {
    id = shipadd.10  # 战斗减速开关
    title = shipadd.title
    desc = shipadd.desc
    picture = GFX_evt_space_station
    show_sound = event_space_cloud
    is_triggered_only = yes
    fire_only_once = yes
    trigger = {
        is_ai = no
    }
    # 不启用
    option = {
        name = shipadd.2
        set_global_flag = shipadd_combat_debuff_disabled
    }
    # 启用战斗减速
    option = {
        name = shipadd.1
        set_global_flag = shipadd_combat_debuff_enabled
    }
}

# 接战：为本舰队挂战斗减速buff并启动循环
fleet_event = {
    id = shipadd.11
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = shipadd_combat_debuff_enabled
        # 短路：已挂buff且循环在跑 → 不处理
        NOT = {
            AND = {
                has_modifier = shipadd_combat_speed_buff
                owner = { has_country_flag = shipadd_combat_buff_active }
            }
        }
    }
    immediate = {
        # 1. 给自己上buff（若没有）
        if = {
            limit = {
                NOT = {
                    has_modifier = shipadd_combat_speed_buff
                }
            }
            add_modifier = {
                modifier = shipadd_combat_speed_buff
                days = -1
            }
        }
        # 2. 启动shipadd.12循环（若未启动）
        owner = {
            if = {
                limit = {
                    NOT = {
                        has_country_flag = shipadd_combat_buff_active
                    }
                }
                set_country_flag = shipadd_combat_buff_active
                country_event = {
                    id = shipadd.12
                    days = 15
                }
            }
        }
    }
}

# shipadd.12：每15天轮询，接战期间持续维护减速buff
# - 先移除所有脱战舰队的减速buff，防止buff持续残留
# - 有接战 → 为战斗中舰队补挂buff（兜底，正常由 shipadd.11 即时挂）→ 继续循环
# - 无接战 → 清除所有舰队的减速buff与标记
country_event = {
    id = shipadd.12
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_country_flag = shipadd_combat_buff_active
    }
    immediate = {
        # 1. 移除所有不在战斗中的舰队的减速buff，防止脱战后buff持续残留
        every_owned_fleet = {
            limit = {
                NOT = {
                    is_in_combat = yes
                }
            }
            remove_modifier = shipadd_combat_speed_buff
        }
        # 2. 检查是否仍有舰队在战斗
        if = {
            limit = {
                any_owned_fleet = {
                    is_in_combat = yes
                }
            }
            # 为仍在战斗的舰队补挂buff（兜底，新船通常已由shipadd.11即时上buff）
            every_owned_fleet = {
                limit = {
                    is_in_combat = yes
                }
                add_modifier = {
                    modifier = shipadd_combat_speed_buff
                    days = -1
                }
            }
            country_event = {
                id = shipadd.12
                days = 15
            }
        }
        else = {
            # 无舰队在战斗：清除所有减速buff与标记
            every_owned_fleet = {
                remove_modifier = shipadd_combat_speed_buff
            }
            remove_country_flag = shipadd_combat_buff_active
        }
    }
}

# shipadd.13：年度兜底清理（on_yearly_pulse_country 触发，每年一次）
# - 移除所有脱战舰队的减速buff，防止 shipadd.12 循环意外中断后buff持续残留
# - 仍有舰队在战斗 → 确保其带有buff；若15天循环已停则重启 shipadd.12
# - 无舰队在战斗 → 清除所有减速buff与标记
country_event = {
    id = shipadd.13
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = shipadd_combat_debuff_enabled
        OR = {
            has_country_flag = shipadd_combat_buff_active
            any_owned_fleet = {
                is_in_combat = yes
            }
            any_owned_fleet = {
                has_modifier = shipadd_combat_speed_buff
            }
        }
    }
    immediate = {
        # 1. 移除所有不在战斗中的舰队的减速buff
        every_owned_fleet = {
            limit = {
                NOT = {
                    is_in_combat = yes
                }
            }
            remove_modifier = shipadd_combat_speed_buff
        }
        # 2. 检查是否仍有舰队在战斗
        if = {
            limit = {
                any_owned_fleet = {
                    is_in_combat = yes
                }
            }
            # 仍有舰队在战斗：确保其带有减速buff
            every_owned_fleet = {
                limit = {
                    is_in_combat = yes
                }
                add_modifier = {
                    modifier = shipadd_combat_speed_buff
                    days = -1
                }
            }
            # 若15天循环已停（标记缺失），则重启循环
            if = {
                limit = {
                    NOT = {
                        has_country_flag = shipadd_combat_buff_active
                    }
                }
                set_country_flag = shipadd_combat_buff_active
                country_event = {
                    id = shipadd.12
                    days = 15
                }
            }
        }
        else = {
            # 无舰队在战斗：清除所有减速buff与标记
            every_owned_fleet = {
                remove_modifier = shipadd_combat_speed_buff
            }
            remove_country_flag = shipadd_combat_buff_active
        }
    }
}
