namespace = shipadd

country_event = {
    id = shipadd.10  # 战斗减速开关
    title = shipadd.title
    desc = shipadd.desc
    picture = GFX_evt_space_station
    show_sound = event_space_cloud
    is_triggered_only = yes
    fire_only_once = yes
    trigger = {
        is_ai = no
    }
    # 不启用
    option = {
        name = shipadd.2
        set_global_flag = shipadd_combat_debuff_disabled
    }
    # 启用战斗减速
    option = {
        name = shipadd.1
        set_global_flag = shipadd_combat_debuff_enabled
    }
}

# 接战：为本星系所有舰队挂战斗减速buff
fleet_event = {
    id = shipadd.11
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = shipadd_combat_debuff_enabled
    }
    immediate = {
        solar_system = {
            every_fleet_in_system = {
                add_modifier = {
                    modifier = shipadd_combat_speed_buff
                    days = -1
                }
            }
        }
        owner = {
            if = {
                limit = {
                    NOT = {
                        has_country_flag = shipadd_combat_buff_active
                    }
                }
                set_country_flag = shipadd_combat_buff_active
                country_event = {
                    id = shipadd.12
                    days = 15
                }
            }
        }
    }
}

# shipadd.12：每15天检测是否仍有舰队接战
# - 有接战 → 重新挂buff（覆盖新船/增援）→ 继续循环
# - 无接战 → 清除所有舰队的减速buff
country_event = {
    id = shipadd.12
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_country_flag = shipadd_combat_buff_active
    }
    immediate = {
        if = {
            limit = {
                any_owned_fleet = {
                    is_in_combat = yes
                }
            }
            every_owned_fleet = {
                limit = {
                    is_in_combat = yes
                }
                add_modifier = {
                    modifier = shipadd_combat_speed_buff
                    days = -1
                }
            }
            country_event = {
                id = shipadd.12
                days = 15
            }
        }
        else = {
            every_owned_fleet = {
                remove_modifier = shipadd_combat_speed_buff
            }
            remove_country_flag = shipadd_combat_buff_active
        }
    }
}
