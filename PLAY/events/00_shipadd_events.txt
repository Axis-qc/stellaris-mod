namespace = shipadd

country_event = {
    id = shipadd.10  # 战斗减速开关
    title = shipadd.title
    desc = shipadd.desc
    picture = GFX_evt_space_station
    show_sound = event_space_cloud
    is_triggered_only = yes
    fire_only_once = yes
    trigger = {
        is_ai = no
        # 彼岸星舟起源自带减速机制，本开关不弹
        NOT = {
            has_origin = origin_br_battle_royale
        }
    }
    # 不启用
    option = {
        name = shipadd.2
        set_global_flag = shipadd_combat_debuff_disabled
    }
    # 启用战斗减速，并撤掉大逃杀战斗减速的启用标记
    option = {
        name = shipadd.1
        set_global_flag = shipadd_combat_debuff_enabled
        remove_global_flag = br_combat_slow_enabled
    }
}

# 接战：为本舰队挂战斗减速并启动循环
fleet_event = {
    id = shipadd.11
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = shipadd_combat_debuff_enabled
		NOR = {
			has_global_flag = br_mode_pve
			has_global_flag = br_mode_pvp
		}
    }
    immediate = {
        # 1. 按保留比例套用减益
        shipadd_clamp_combat_speed_effect = yes
        # 2. 启动shipadd.12循环（若未启动）
        owner = {
            if = {
                limit = {
                    NOT = {
                        has_country_flag = shipadd_combat_buff_active
                    }
                }
                set_country_flag = shipadd_combat_buff_active
                country_event = {
                    id = shipadd.12
                    days = 15
                }
            }
        }
    }
}

# shipadd.12：每15天轮询，接战期间持续维护减速
# - 先移除所有脱战舰队的减速补正，防止残留
# - 有接战 → 为战斗中舰队补挂（兜底，正常由 shipadd.11 即时挂）→ 继续循环
# - 无接战 → 清除所有舰队的减速补正与标记
country_event = {
    id = shipadd.12
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_country_flag = shipadd_combat_buff_active
    }
    immediate = {
        # 1. 移除所有不在战斗中的舰队的减速补正
        every_owned_fleet = {
            limit = {
                NOT = {
                    is_in_combat = yes
                }
            }
            shipadd_clamp_combat_speed_clear_effect = yes
        }
        # 2. 检查是否仍有舰队在战斗
        if = {
            limit = {
                any_owned_fleet = {
                    is_in_combat = yes
                }
            }
            # 为仍在战斗的舰队补挂（兜底，新船通常已由 shipadd.11 即时上）
            every_owned_fleet = {
                limit = {
                    is_in_combat = yes
                }
                shipadd_clamp_combat_speed_effect = yes
            }
            country_event = {
                id = shipadd.12
                days = 15
            }
        }
        else = {
            # 无舰队在战斗：清除所有减速补正与标记
            every_owned_fleet = {
                shipadd_clamp_combat_speed_clear_effect = yes
            }
            remove_country_flag = shipadd_combat_buff_active
        }
    }
}

# shipadd.13：年度兜底清理（on_yearly_pulse_country 触发，每年一次）
# - 移除所有脱战舰队的减速补正，防止 shipadd.12 循环意外中断后残留
# - 仍有舰队在战斗 → 确保其带有减速；若15天循环已停则重启 shipadd.12
# - 无舰队在战斗 → 清除所有减速补正与标记
country_event = {
    id = shipadd.13
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = shipadd_combat_debuff_enabled
        OR = {
            has_country_flag = shipadd_combat_buff_active
            any_owned_fleet = {
                is_in_combat = yes
            }
            any_owned_fleet = {
                any_owned_ship = {
                    has_ship_flag = shipadd_clamp_active
                }
            }
        }
    }
    immediate = {
        # 1. 移除所有不在战斗中的舰队的减速补正
        every_owned_fleet = {
            limit = {
                NOT = {
                    is_in_combat = yes
                }
            }
            shipadd_clamp_combat_speed_clear_effect = yes
        }
        # 2. 检查是否仍有舰队在战斗
        if = {
            limit = {
                any_owned_fleet = {
                    is_in_combat = yes
                }
            }
            # 仍有舰队在战斗：确保其带有减速
            every_owned_fleet = {
                limit = {
                    is_in_combat = yes
                }
                shipadd_clamp_combat_speed_effect = yes
            }
            # 若15天循环已停（标记缺失），则重启循环
            if = {
                limit = {
                    NOT = {
                        has_country_flag = shipadd_combat_buff_active
                    }
                }
                set_country_flag = shipadd_combat_buff_active
                country_event = {
                    id = shipadd.12
                    days = 15
                }
            }
        }
        else = {
            # 无舰队在战斗：清除所有减速补正与标记
            every_owned_fleet = {
                shipadd_clamp_combat_speed_clear_effect = yes
            }
            remove_country_flag = shipadd_combat_buff_active
        }
    }
}
