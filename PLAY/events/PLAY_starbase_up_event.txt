#星环基地+1建造
country_event = {
    id = starbaseup.100
    hide_window = yes

    is_triggered_only = yes
    trigger = {
        is_ai = no
        any_owned_planet = {
            NOT = {
                has_modifier = play_orbital_ring
            }
        }
    }

    immediate = {
        every_owned_planet = {
            limit = {
                exists = orbital_defence
            }
            add_modifier = {
                modifier = play_orbital_ring
                days = -1
            }
        }
    }
}
