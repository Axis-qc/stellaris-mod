    icon = "GFX_edict_type_policy"
    #启动条件
    allow = {
        always = yes
    }
    #显示条件
    potential = {
        end_game_years_passed = 1
        any_country = {
            is_ai = yes

        }

        category = edicts

    effect = {

    }

    ai_weight = {
        weight = 0
    }
}
