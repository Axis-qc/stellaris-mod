#迁移人口脚本
#PLANET_TARGET: 目标行星的event_target名称（如 move_pop_planet）
#支持两种模式：AMOUNT（精确数量）和 PERCENTAGE（百分比）
play_move_pop_effect = {
	[[PERCENTAGE]
	# 先统计源星球符合条件的总人口
	set_variable = {
		which = play_total_pop
		value = 0
	}
	every_owned_planet = {
		limit = {
			not = {
				is_planet_class = pc_cosmogenesis_world
			}
			has_modifier = play_move_pop_to_planet
		}
		every_owned_pop_group = {
			limit = {
				OR = {
					is_pop_category = civilian
					is_pop_category = maintenance_drone
				}
				not = {
					has_trait = trait_robot_digital_2
				}
			}
			owner = {
				change_variable = {
					which = play_total_pop
					value = prev.trigger:pop_group_size
				}
			}
		}
	}
	# 总人口 × 百分比 = 目标迁移量
	multiply_variable = {
		which = play_total_pop
		value = $PERCENTAGE$
	}
	# 复用 AMOUNT 的 while 循环逻辑
	set_variable = {
		which = play_remaining
		value = play_total_pop
	}
	set_variable = {
		which = play_local_move
		value = 0
	}
	while = {
		limit = {
			check_variable = {
				which = play_remaining
				value > 0
			}
			any_owned_planet = {
				not = {
					is_planet_class = pc_cosmogenesis_world
				}
				has_modifier = play_move_pop_to_planet
				any_owned_pop_group = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
					not = {
						has_trait = trait_robot_digital_2
					}
				}
			}
		}
		random_owned_planet = {
			limit = {
				not = {
					is_planet_class = pc_cosmogenesis_world
				}
				has_modifier = play_move_pop_to_planet
				any_owned_pop_group = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
					not = {
						has_trait = trait_robot_digital_2
					}
				}
			}
			random_owned_pop_group = {
				limit = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
					not = {
						has_trait = trait_robot_digital_2
					}
				}
				owner = {
					set_variable = {
						which = play_local_move
						value = prev.trigger:pop_group_size
					}
					if = {
						limit = {
							check_variable = {
								which = play_local_move
								value > play_remaining
							}
						}
						set_variable = {
							which = play_local_move
							value = play_remaining
						}
					}
				}
				save_event_target_as = play_source_pop
				resettle_pop_group = {
					POP_GROUP = this
					PLANET = event_target:$PLANET_TARGET$
					AMOUNT = owner.play_local_move
				}
				owner = {
					subtract_variable = {
						which = play_remaining
						value = play_local_move
					}
				}
			}
		}
	}
	clear_variable = play_total_pop
	clear_variable = play_remaining
	clear_variable = play_local_move
	]
	[[AMOUNT]
	set_variable = {
		which = play_remaining
		value = $AMOUNT$
	}
	set_variable = {
		which = play_local_move
		value = 0
	}
	while = {
		limit = {
			check_variable = {
				which = play_remaining
				value > 0
			}
			any_owned_planet = {
				not = {
					is_planet_class = pc_cosmogenesis_world
				}
				has_modifier = play_move_pop_to_planet
				any_owned_pop_group = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
					not = {
						has_trait = trait_robot_digital_2
					}
				}
			}
		}
		random_owned_planet = {
			limit = {
				not = {
					is_planet_class = pc_cosmogenesis_world
				}
				has_modifier = play_move_pop_to_planet
				any_owned_pop_group = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
					not = {
						has_trait = trait_robot_digital_2
					}
				}
			}
			random_owned_pop_group = {
				limit = {
					OR = {
						is_pop_category = civilian
						is_pop_category = maintenance_drone
					}
					not = {
						has_trait = trait_robot_digital_2
					}
				}
				owner = {
					set_variable = {
						which = play_local_move
						value = prev.trigger:pop_group_size
					}
					if = {
						limit = {
							check_variable = {
								which = play_local_move
								value > play_remaining
							}
						}
						set_variable = {
							which = play_local_move
							value = play_remaining
						}
					}
				}
				save_event_target_as = play_source_pop
				resettle_pop_group = {
					POP_GROUP = this
					PLANET = event_target:$PLANET_TARGET$
					AMOUNT = owner.play_local_move
				}
				owner = {
					subtract_variable = {
						which = play_remaining
						value = play_local_move
					}
				}
			}
		}
	}
	clear_variable = play_remaining
	clear_variable = play_local_move
	]
}
