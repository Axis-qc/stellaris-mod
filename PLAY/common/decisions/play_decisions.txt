#迁入人口
play_move_pop_to_this = {
    owned_planets_only = yes
    icon = decision_art_monument
    resources = {
        category = decisions
        cost = {
            energy = 1
        }
    }
    potential = {
        # from = {
        #     any_owned_planet = {
        #         has_modifier = play_move_pop_to_planet
        #     }
        # }
        not = {
            has_modifier = play_move_pop_to_planet
        }
    }
    effect = {
        if = {
            limit = {
                from = {
                    any_owned_planet = {
                        has_modifier = play_move_pop_to_planet
                    }
                }
            }
            from = {
                country_event = { id = play_move_pop.100 }
            }
        }
        if = {
            limit = {
                from = {
                    any_owned_planet = {
                        NOT = {
                            has_modifier = play_move_pop_to_planet
                        }
                    }
                }
            }
            from = {
                country_event = { id = play_move_pop.101 }
            }
        }
        this = {
            #保存迁入星球
            save_global_event_target_as = move_pop_planet
            save_global_event_target_as = play_move_pop_to_this_planet
        }
        event_target:move_pop_planet = {
            set_variable = { which = move_pop_planet_variable value = trigger:count_owned_pop_amount }
        }
        add_modifier = {
            modifier = play_move_pop_to_this_planet
            days = 10
        }
    }
    ai_weight = {
        weight = 0
    }
}
