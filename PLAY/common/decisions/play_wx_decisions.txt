    enactment_time = 720
    important = yes

        cost = {

    potential = {

            is_planet_class = pc_habitat

            has_ascension_perk = ap_mastery_of_nature
        }
    }

    effect = {
        change_variable = { which = planet_qh_add_value value = 2 }
        add_modifier = {
            modifier = planet_qh_add
            mult = planet_qh_add_value

    ai_weight = {
        weight = 0

            free_district_slots > 0
            pop_amount < 5000 #carrying capacity bonus always good at that stage
        }
