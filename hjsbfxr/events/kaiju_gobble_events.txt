# 噬天巨兽（原版 Behemoth）吞噬星球：充能期星球侧粒子驱动事件
# 配套资源：gfx/particles/kaiju_gobble_*、gfx/models/combat_items/kaiju_gobble_entities.asset、
#           common/ambient_objects/kaiju_gobble_ambient.txt、common/on_actions/kaiju_gobble_on_actions.txt
namespace = bfxr

# bfxr.6001：下命令（windup 起点）→ 打标记，10 天后叫 bfxr.6003 出粒子
# bfxr.6002：取消或被吞 → 清标记 + 收掉粒子（同一事件被两个 on_action 引用）
# bfxr.6003：延迟后实际创建粒子（this = 星球，from = 舰队）
carrier_event = {
	id = bfxr.6001
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_planet_flag = bfxr_kaiju_gobble_pending
		# from（舰队）必须显式传下去，否则延续事件里 from 会变成星球，朝向就丢了
		planet_event = {
			id = bfxr.6003
			days = 20
			scopes = { from = from }
		}
	}
}

planet_event = {
	id = bfxr.6003
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		# 20 天内取消过订单就不出粒子（标记在 bfxr.6002 里清）
		if = {
			limit = {
				has_planet_flag = bfxr_kaiju_gobble_pending
			}
			create_ambient_object = {
				type = bfxr_kaiju_gobble_windup_vfx
				location = this
				use_3d_location = yes
				entity_scale_to_size = yes
				# 0.02 = defines 的 PLANET_DESTRUCTION_HIT_ENTITY_SCALE，与引擎给 planet_hit_entity 的缩放一致
				scale = 0.02
				base_angle_towards = from
				entity_face_object = from
				play_animation_once = no
				duration = 400
			}
			last_created_ambient_object = {
				set_ambient_object_flag = bfxr_kaiju_gobble_windup
			}
		}
	}
}

carrier_event = {
	id = bfxr.6002
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		remove_planet_flag = bfxr_kaiju_gobble_pending
		solar_system = {
			every_system_ambient_object = {
				limit = {
					has_ambient_object_flag = bfxr_kaiju_gobble_windup
				}
				destroy_ambient_object = this
			}
		}
	}
}
