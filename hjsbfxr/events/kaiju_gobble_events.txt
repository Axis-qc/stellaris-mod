# 噬天巨兽吞噬星球：充能期星球侧粒子 + 碎片驱动事件
#
# 事件流：
# - bfxr.6001：下命令 → 打标记，20 天后调 bfxr.6003（from 传舰队，朝向要用）
# - bfxr.6003：藏原星球 + 建 2 个环境物体（碎片 + 粒子）
# - bfxr.6002：成功被吞或取消 → 清标记 + 收环境物体（不恢复星球）
# - bfxr.6004：仅取消时触发 → 恢复星球渲染实体
#
# 注意：
# - 粒子环境物体要 base_angle_towards / entity_face_object = from（吸力在粒子局部 +Z）；
#   碎片环境物体不要设朝向，否则碎片横着飞。
# - 藏星球用 kaiju_gobble_invisible_planet_entity（有几何体的全透明网格），
#   不能用 locator_mesh：entity_scale_to_size 算出 0 会把粒子碎片一起缩没。
# - _once_init_mod_files.py 会覆盖本文件，别跑。
namespace = bfxr

carrier_event = {
	id = bfxr.6001
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_planet_flag = bfxr_kaiju_gobble_pending
		# from（舰队）必须显式传下去，否则延续事件里 from 会变成星球，朝向就丢
		planet_event = {
			id = bfxr.6003
			days = 20
			scopes = { from = from }
		}
	}
}

planet_event = {
	id = bfxr.6003
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		# 20 天内取消过订单就不出效果（标记在 bfxr.6002 里清）
		if = {
			limit = {
				has_planet_flag = bfxr_kaiju_gobble_pending
			}
			# 藏原星球防重影（只换渲染实体，游戏数据不受影响）
			set_planet_entity = { entity = kaiju_gobble_invisible_planet_entity }

			# 碎片：不设朝向 => 竖直向上
			create_ambient_object = {
				type = bfxr_kaiju_gobble_shards_vfx
				location = this
				use_3d_location = yes
				entity_scale_to_size = yes
				scale =0.2
				play_animation_once = no
				duration = 400
			}
			last_created_ambient_object = {
				set_ambient_object_flag = bfxr_kaiju_gobble_windup
			}

			# 粒子：要对准巨兽
			create_ambient_object = {
				type = bfxr_kaiju_gobble_windup_vfx
				location = this
				use_3d_location = yes
				entity_scale_to_size = yes
				scale =0.02
				base_angle_towards = from
				entity_face_object = from
				play_animation_once = no
				duration = 400
			}
			last_created_ambient_object = {
				set_ambient_object_flag = bfxr_kaiju_gobble_windup
			}
		}
	}
}

carrier_event = {
	id = bfxr.6002
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		remove_planet_flag = bfxr_kaiju_gobble_pending
		solar_system = {
			every_system_ambient_object = {
				limit = {
					has_ambient_object_flag = bfxr_kaiju_gobble_windup
				}
				destroy_ambient_object = this
			}
		}
	}
}
