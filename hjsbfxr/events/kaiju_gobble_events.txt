# 噬天巨兽吞噬星球：星球侧粒子 + 碎片。
# 6001 下命令打标记 → 6005 每 10 天等巨兽到位 → 6003 放特效；6002 清理、6004 取消时恢复星球。
#
# 硬警告：
# - 到位判定必须下沉到舰船（any_owned_ship）：fleet 没有 3D 坐标，对 fleet 用 distance / orbit
#   都不行（orbit 直接报 Invalid context switch）。阈值 30 = 巨兽贴上目标轨道的距离。
# - 粒子与碎片环境物体都不设朝向：粒子靠基准坐标 +Y 竖直上升；碎片设了朝向会横着飞。
# - 非球体目标（居住站 / 环形区段 / 方舟）不放碎片，只留粒子。
# - 藏星球用 kaiju_gobble_invisible_planet_entity，不能用 locator_mesh
#   （entity_scale_to_size 会算成 0 把粒子碎片一起缩没）。
# - 碎片贴图族见 kaiju_gobble_entities.asset（生成脚本 kb_gen_shards_materials.py）。
namespace = bfxr

carrier_event = {
	id = bfxr.6001
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_planet_flag = bfxr_kaiju_gobble_pending
		# from（舰队）要一路传下去：6005 靠它查到位距离
		planet_event = {
			id = bfxr.6005
			days = 15
			scopes = { from = from }
		}
	}
}

# bfxr.6005：等待巨兽到位（每 10 天自查一次）。判定见文件头。
planet_event = {
	id = bfxr.6005
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				has_planet_flag = bfxr_kaiju_gobble_pending
			}
			if = {
				limit = {
					exists = from
					from = {
						any_owned_ship = {
							distance = {
								source = root
								max_distance <= 30
								same_solar_system = yes
							}
						}
					}
				}
				planet_event = {
					id = bfxr.6003
					scopes = { from = from }
				}
			}
			else = {
				planet_event = {
					id = bfxr.6005
					days = 15
					scopes = { from = from }
				}
			}
		}
	}
}

planet_event = {
	id = bfxr.6003
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				has_planet_flag = bfxr_kaiju_gobble_pending
			}
			# 藏原星球防重影（只换渲染实体，游戏数据不受影响）
			set_planet_entity = { entity = kaiju_gobble_invisible_planet_entity }

			# 碎片：按星球类型选贴图族（非球体目标不放碎片）
			if = {
				limit = {
					NOR = {
						is_planet_class = pc_habitat
						is_planet_class = pc_ringworld_habitable
						is_planet_class = pc_shattered_ring_habitable
						is_planet_class = pc_crystal_habitat
						is_planet_class = pc_warden_guardian
						is_planet_class = pc_ark
					}
				}
				switch = {
					trigger = is_planet_class
					pc_desert = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_desert } }
					pc_arid = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_arid } }
					pc_savannah = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_savannah } }
					pc_tropical = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_tropical } }
					pc_continental = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_continental } }
					pc_ocean = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_ocean } }
					pc_tundra = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_tundra } }
					# 死寂星球的地表贴图原版就是 tundra_01（核爆感来自云层），共用同一族
					pc_nuked = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_tundra } }
					pc_arctic = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_arctic } }
					pc_alpine = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_alpine } }
					pc_volcanic = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_volcanic } }
					pc_gaia = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_gaia } }
					pc_hive = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_infested } }
					pc_infested = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_infested } }
					pc_machine = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_ai } }
					pc_city = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_city } }
					pc_relic = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_relic } }
					# 其余球体类（含别的 mod 新增类）兜底用 tundra 族，别留空
					default = { kaiju_gobble_spawn_shards = { TYPE = bfxr_kaiju_gobble_shards_vfx_tundra } }
				}
			}

			# 粒子：不设朝向，竖直上升（力场在 kaiju_gobble_planet.asset）
			create_ambient_object = {
				type = bfxr_kaiju_gobble_windup_vfx
				location = this
				use_3d_location = yes
				entity_scale_to_size = yes
				scale =0.02
				play_animation_once = no
				duration = 400
			}
			last_created_ambient_object = {
				set_ambient_object_flag = bfxr_kaiju_gobble_windup
			}
		}
	}
}

carrier_event = {
	id = bfxr.6002
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		remove_planet_flag = bfxr_kaiju_gobble_pending
		solar_system = {
			every_system_ambient_object = {
				limit = {
					has_ambient_object_flag = bfxr_kaiju_gobble_windup
				}
				destroy_ambient_object = this
			}
		}
	}
}
