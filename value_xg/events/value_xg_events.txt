    id = value_xg.100
    hide_window = yes  # 隐藏窗口
    is_triggered_only = yes  # 仅在触发条件满足时触发
    trigger = {
        not = {
            has_country_flag = yes_xg
        }
    }
    immediate = {
        random_country = {
            limit = {
                not = {
                    has_country_flag = 30_day_cd
                }
            }
            random_owned_fleet = {
                limit = {
                    is_ship_class = shipclass_military
                    not = {
                        has_fleet_flag = ship_value_del
                    }
                }
                set_timed_fleet_flag = {
                    days = 120
                    flag = ship_value_del
                }
                save_event_target_as = root_power
            }
            set_timed_country_flag = { flag = 30_day_cd days = 30 }
            country_event = { id = value_xg.3000 days = 1 }
            country_event = { id = value_xg.101 days = 1 }
        }
        
    }

}
