#星球刷虫子
#原版两年一次 每次-2星球规模，扣完摧毁星球 没有就啃普通球
country_event = {
    # 国家事件
    id = yesTZcz.1011
    hide_window = yes

    is_triggered_only = yes

    trigger = {
        is_country_type = swarm  #是虫子
        num_ships < 5000        # 船只数量
        any_system_within_border = { exists = starbase }  # 边境内有星基存在
    }
    immediate = {
        save_event_target_as = chongzi  # 将当前国家保存为普雷索林事件目标
        owner_species = { save_event_target_as = prethoryn_species }  # 将所有者物种保存为普雷索林物种事件目标
        every_planet_within_border = {
            limit = { is_planet_class = pc_infested }
            set_owner = event_target:chongzi
        }
        if = {
            limit = {
                num_owned_planets > 0    # 拥有至少一个行星
            }
            #从孵化星球补
            random_owned_planet = {
                limit = {
                    is_planet_class = pc_infested
                    solar_system = {
                        exists = starbase
                    }
                }
                save_event_target_as = spawn_location
                planet_event = {
                    id = yesTZcz.1020
                }
                chongzi_fleet = yes  # 生成虫群繁殖舰队
            }
        }
        #没有孵化就啃普通球
        if = {
            limit = {
                # galaxy_percentage > 0.40  # 控制星系占全星系比例超过40%
                num_owned_planets = 0
            }
            random_owned_planet = {
                limit = {
                    or = {
                        is_planet_class = pc_gas_giant
                        is_planet_class = pc_molten
                        is_planet_class = pc_barren
                        is_planet_class = pc_barren_cold
                        is_planet_class = pc_toxic
                        is_planet_class = pc_frozen
                    }
                }
                save_event_target_as = spawn_location
                chongzi_fleet = yes  # 生成虫群繁殖舰队
                change_pc = pc_broken
            }
        }
    }
}
#拆星球
planet_event = {
    id = yesTZcz.1020
    hide_window = yes

    is_triggered_only = yes  # 该事件只能通过触发器触发

    immediate = {
        if = {
            limit = {
                not = {
                    has_planet_flag = has_chongz_planet
                }
            }
            set_variable = { which = yesTZcz value = trigger:planet_size }

            set_planet_flag = has_chongz_planet
        }
        if = {
            limit = {
                has_planet_flag = has_chongz_planet
            }
            change_variable = {
                which = yesTZcz
                value = -2
            }
        }
        if = {
            limit = {
                check_variable = {
                    which = yesTZcz
                    value <= 0
                }
            }
            change_pc = pc_broken
        }
    }
}
