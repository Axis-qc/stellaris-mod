# br.6010：湿润或盖亚星球被母舰采掘后发放生态类资源奖励喵
country_event = {
	id = br.6010
	title = "BR_DESTRUCTION_REWARD_BIOSPHERE_TITLE"
	desc = "BR_DESTRUCTION_REWARD_BIOSPHERE_DESC"
	picture = GFX_evt_asteroid_field
	is_triggered_only = yes
	option = {
		name = "BR_DESTRUCTION_REWARD_BIOSPHERE_FOOD"
		add_resource = {
			food = @reward_alloy_base
			mult = br_power_level
		}
	}
	option = {
		name = "BR_DESTRUCTION_REWARD_BIOSPHERE_SOCIETY"
		trigger = {
			OR = {
				NOT = {
					has_global_flag = br_mode_pve
				}
				check_variable = {
					which = br_power_level
					value >= 2
				}
			}
		}
		add_resource = {
			society_research = @reward_alloy_base
			mult = br_power_level
		}
	}
	option = {
		name = "BR_DESTRUCTION_REWARD_BIOSPHERE_CONSUMER"
		trigger = {
			OR = {
				NOT = {
					has_global_flag = br_mode_pve
				}
				check_variable = {
					which = br_power_level
					value >= 3
				}
			}
		}
		add_resource = {
			consumer_goods = 30
			mult = br_power_level
		}
	}
}
