# 防御协议激活（由 on_br_defense_skill_activated 触发，需 br_skill_phase_shift_unlocked）
fleet_event = {
    id = heroship.1200
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        br_has_defense_vanguard = yes
        owner = {
            has_country_flag = br_skill_phase_shift_unlocked
            br_vanguard_defense_skill_cd_ready = yes
        }
    }

    immediate = {
        every_owned_ship = {
            limit = {
                is_ship_size = br_vanguard
                has_component = BR_VANGUARD_CORE_DEFENSE
            }
            add_modifier = {
                modifier = br_vanguard_phase_shift
                mult = value:br_skill_phase_shift_dmgreduct
                days = 1
                time_multiplier = value:br_skill_phase_shift_day
            }
        }
        solar_system = {
            every_fleet_in_system = {
                limit = {
                    exists = owner
                    owner = { NOT = { is_hostile = root.owner } }
                }
                add_modifier = {
                    modifier = br_vanguard_phase_shift_regen
                    mult = value:br_skill_phase_shift_regen_amount
                    days = 1
                    time_multiplier = value:br_skill_phase_shift_regen_day
                }
            }
        }
        owner = {
            set_variable = { which = br_vanguard_defense_skill_cd value = value:br_skill_phase_shift_cd }
            if = {
                limit = { NOT = { has_country_flag = br_vanguard_skill_cd_tick_running } }
                set_country_flag = br_vanguard_skill_cd_tick_running
                country_event = { id = heroship.1014 days = 1 }
            }
        }
    }
}

                is_ship_size = br_vanguard

                modifier = br_vanguard_def_structure

        }
    }
}

# 超维结构撕裂（由 on_br_defense_ultimate_activated 触发，压制装甲护盾+船体DOT）
fleet_event = {
    id = heroship.1205
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        br_has_defense_vanguard = yes
        owner = {
            has_country_flag = br_skill_collapse_unlocked
            br_vanguard_defense_ultimate_cd_ready = yes
        }
    }

    immediate = {
        every_owned_ship = {
            limit = {
                is_ship_size = br_vanguard
                has_component = BR_VANGUARD_CORE_DEFENSE
            }
            # 英雄舰自伤
            add_modifier = {
                modifier = br_ultimate_hull_damage
                days = 1
                time_multiplier = value:br_vanguard_collapse_damage_day
                mult = value:br_vanguard_collapse_damage
            }
            # 星系敌舰伤害（modifier DOT 方式，不触发事件杀）
            solar_system = {
                every_ship_in_system = {
                    limit = {
                        exists = owner
                        owner = { is_hostile = root.owner }
                    }
                    # damage_ship = owner.br_display_collapse 不能damage，会断事件
                    add_modifier = {
                        modifier = br_ultimate_hull_damage
                        days = 1
                        #作用域在敌方，得转回自己读变量
                        time_multiplier = root.owner.value:br_vanguard_collapse_damage_day
                        mult = root.owner.value:br_vanguard_collapse_damage
                    }
                }
            }
        }
        owner = {
            set_variable = { which = br_vanguard_defense_ultimate_cd value = value:br_vanguard_collapse_cd }
            if = {
                limit = { NOT = { has_country_flag = br_vanguard_skill_cd_tick_running } }
                set_country_flag = br_vanguard_skill_cd_tick_running
                country_event = { id = heroship.1014 days = 1 }
            }
        }
    }
}

# 更新大招显示数值（每月脉冲触发，有解锁技能时才计算）
country_event = {
    id = heroship.1207
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        OR = {
            has_country_flag = br_skill_collapse_unlocked
            has_country_flag = br_skill_projection_unlocked
        }
        any_owned_ship = {
            is_ship_size = br_vanguard
            has_component = BR_VANGUARD_CORE_DEFENSE
        }
    }

    immediate = {
        every_owned_ship = {
            limit = {
                is_ship_size = br_vanguard
                has_component = BR_VANGUARD_CORE_DEFENSE
            }
            export_trigger_value_to_variable = { trigger = has_max_hp variable = br_ship_hp }
            export_trigger_value_to_variable = { trigger = has_max_armor_hp variable = br_ship_armor }
            export_trigger_value_to_variable = { trigger = has_shield_hp variable = br_ship_shield }
            save_event_target_as = br_hero_ship
        }
        set_variable = { which = br_display_total value = event_target:br_hero_ship.br_ship_hp }
        change_variable = { which = br_display_total value = event_target:br_hero_ship.br_ship_armor }
        change_variable = { which = br_display_total value = event_target:br_hero_ship.br_ship_shield }
        # collapse/projection 数值由 script_values 自动计算 (br_vanguard_collapse_damage / br_vanguard_projection_amount)
    }
}
