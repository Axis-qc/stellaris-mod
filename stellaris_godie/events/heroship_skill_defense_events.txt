namespace = heroship

# ========================
# 防御核心技能 — heroship.1200-1299
# ========================

# 自修复激活（由 on_br_defense_skill_activated 触发）
fleet_event = {
    id = heroship.1200
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        br_has_defense_vanguard = yes
        owner = {
            has_country_flag = br_skill_self_repair_unlocked
            br_vanguard_defense_skill_cd_ready = yes
        }
    }

    immediate = {
        every_owned_ship = {
            limit = {
                is_ship_size = br_vanguard
                has_component = BR_VANGUARD_CORE_DEFENSE
            }
            add_modifier = {
                modifier = br_vanguard_self_repair
                days = 30
            }
        }
        owner = {
            set_variable = { which = br_vanguard_defense_skill_cd value = 90 }
            if = {
                limit = { NOT = { has_country_flag = br_vanguard_skill_cd_tick_running } }
                set_country_flag = br_vanguard_skill_cd_tick_running
                country_event = { id = heroship.1014 days = 1 }
            }
        }
    }
}
