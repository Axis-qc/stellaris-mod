# br.1001：对所有危险区里的舰船造成持续伤害，并按缩圈阶段提高伤害喵
country_event = {
    id = br.1001
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        NOT = { has_global_flag = br_cycle_resetting }
        NOT = { has_global_flag = br_shrink_paused }
    }

    immediate = {
        every_system = {
            limit = { has_star_flag = br_danger }

            every_ship_in_system = {
                limit = {
                    exists = controller
                }

                # 基础毒伤：前期给玩家撤离时间，后期不能靠回复硬抗喵
                damage_ship = 50

                # 大图按缩圈次数推进，小图额外按剩余安全圈数兜底，避免伤害永远到不了终盘喵
                if = {
                    limit = {
                        OR = {
                            root = {
                                NOT = {
                                    check_variable = {
                                        which = br_shrink_count
                                        value < 10
                                    }
                                }
                            }
                            br_safe_ring_at_most_15 = yes
                        }
                    }
                    damage_ship = 100
                }
                if = {
                    limit = {
                        OR = {
                            root = {
                                NOT = {
                                    check_variable = {
                                        which = br_shrink_count
                                        value < 15
                                    }
                                }
                            }
                            br_safe_ring_at_most_10 = yes
                        }
                    }
                    damage_ship = 200
                }
                if = {
                    limit = {
                        OR = {
                            root = {
                                NOT = {
                                    check_variable = {
                                        which = br_shrink_count
                                        value < 20
                                    }
                                }
                            }
                            br_safe_ring_at_most_8 = yes
                        }
                    }
                    damage_ship = 400
                }
                if = {
                    limit = {
                        OR = {
                            root = {
                                NOT = {
                                    check_variable = {
                                        which = br_shrink_count
                                        value < 25
                                    }
                                }
                            }
                            br_safe_ring_at_most_6 = yes
                        }
                    }
                    damage_ship = 800
                }
                if = {
                    limit = {
                        OR = {
                            root = {
                                NOT = {
                                    check_variable = {
                                        which = br_shrink_count
                                        value < 30
                                    }
                                }
                            }
                            br_safe_ring_at_most_5 = yes
                        }
                    }
                    damage_ship = 1200
                }
                if = {
                    limit = {
                        OR = {
                            root = {
                                NOT = {
                                    check_variable = {
                                        which = br_shrink_count
                                        value < 35
                                    }
                                }
                            }
                            br_safe_ring_at_most_4 = yes
                        }
                    }
                    damage_ship = 1600
                }
                if = {
                    limit = {
                        OR = {
                            root = {
                                NOT = {
                                    check_variable = {
                                        which = br_shrink_count
                                        value < 40
                                    }
                                }
                            }
                            br_safe_ring_at_most_3 = yes
                        }
                    }
                    damage_ship = 2200
                }
                if = {
                    limit = {
                        OR = {
                            root = {
                                NOT = {
                                    check_variable = {
                                        which = br_shrink_count
                                        value < 45
                                    }
                                }
                            }
                            br_safe_ring_at_most_2 = yes
                        }
                    }
                    damage_ship = 3000
                }
            }
        }

        if = {
            limit = { has_global_flag = br_speed_fast }
            country_event = { id = br.1001 days = 5 }
        }
        else_if = {
            limit = { has_global_flag = br_speed_normal }
            country_event = { id = br.1001 days = 10 }
        }
        else_if = {
            limit = { has_global_flag = br_speed_slow }
            country_event = { id = br.1001 days = 15 }
        }
        else = {
            country_event = { id = br.1001 days = 10 }
        }
    }
}

# br.1010：给所有br_danger星系添加星系内特效（无人区无法用尘云，改用ambient_object）喵
country_event = {
    id = br.1010
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        every_system = {
            limit = {
                has_star_flag = br_danger
                NOT = { any_system_ambient_object = { has_ambient_object_flag = br_danger_cloud_object } }
            }
            star = {
                create_ambient_object = {
                    type = br_danger_cloud
                    location = this
                }
                last_created_ambient_object = {
                    set_ambient_object_flag = br_danger_cloud_object
                    set_location = {
                        target = prev
                        distance = 0
                        angle = random
                    }
                }
            }
        }
    }
}
