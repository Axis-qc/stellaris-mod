# PVE普通敌人刷敌事件：br.800-br.806
# br.800：刷敌调度入口
# br.801-br.806：分批刷敌，N=1→10从外圈到内圈递增
# 每星系16支舰队：1BB + 2CA + 3DD + 5护卫 + 5雷击
# 内圈(ring_1~3)精英，外圈(ring_4~25)普通
# 根据br_pve_template_artillery/brawler/mixed选择设计
# ============================================================
country_event = {
    id = br.800
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        NOT = { has_global_flag = br_pve_initialized }
        exists = event_target:br_pve_enemy_country
    }

    immediate = {
        if = {
            limit = { NOT = { has_global_flag = br_pve_spawn_started } }
            set_global_flag = br_pve_spawn_started
        }
        country_event = { id = br.801 days = 3 }
    }
}
