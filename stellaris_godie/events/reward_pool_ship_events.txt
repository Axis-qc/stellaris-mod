namespace = reward

# ========================
# 舰船池 — reward.4001-4999
# 只能由 reward.1 总触发器调用。
# 五级制分流，具体奖励只做显示检测：无可用落点不显示。
# ========================

# 舰船池入口
country_event = {
    id = reward.4001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        country_event = { id = reward.4101 }
    }
}

# 舰船池等级触发器
country_event = {
    id = reward.4101
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { check_variable = { which = br_power_level value < 3 } }
            country_event = { id = reward.4102 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 6 } }
            country_event = { id = reward.4103 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 9 } }
            country_event = { id = reward.4104 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 12 } }
            country_event = { id = reward.4105 }
        }
        else = {
            country_event = { id = reward.4106 }
        }
    }
}

# T1 舰船 (Lv1-2) — 小型支援舰队
country_event = {
    id = reward.4102
    is_triggered_only = yes
    title = REWARD_POOL_SHIP_T1_TITLE
    desc = REWARD_POOL_SHIP_T1_DESC
    picture = GFX_evt_ship_in_orbit

    option = {
        name = REWARD_SHIP_ESCORT_T1
        trigger = { any_owned_fleet = { } }
        custom_tooltip = REWARD_SHIP_ESCORT_T1_TT
        hidden_effect = {
            random_owned_fleet = { save_event_target_as = reward_spawn_fleet }
            create_fleet = {
                name = random
                effect = {
                    set_owner = root
                    random_list = { 50 = { create_ship = { name = random design = "BR_PVE_ARTILLERY_DESTROYER" } } 50 = { create_ship = { name = random design = "BR_PVE_BRAWLER_DESTROYER" } } }
                    random_list = { 50 = { create_ship = { name = random design = "BR_PVE_ARTILLERY_CORVETTE" } } 50 = { create_ship = { name = random design = "BR_PVE_BRAWLER_CORVETTE" } } }
                    random_list = { 50 = { create_ship = { name = random design = "BR_PVE_ARTILLERY_CORVETTE" } } 50 = { create_ship = { name = random design = "BR_PVE_BRAWLER_CORVETTE" } } }
                    set_location = { target = event_target:reward_spawn_fleet distance = 35 angle = random }
                }
            }
        }
    }
}
