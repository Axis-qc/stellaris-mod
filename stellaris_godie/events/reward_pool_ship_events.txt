namespace = reward

# ========================
# 舰船池 — reward.4001-4099
# 只能由 reward.1 总触发器调用。
# 统一基础数×战力等级，按舰船占用分配（小船多，大船少）。
# 舰队固定在母舰位置生成。
# 需对应舰船科技才能选择。
# ========================

# 舰船池入口
country_event = {
    id = reward.4001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        # 计算各类型舰船数量：level / 权重
        # 权重：护卫1 驱逐2 巡洋4 战列8 → base = 1/权重
        # 大船解锁等级：驱逐2 巡洋4 战列8
        # 护卫舰：level × 1，无门槛
        set_variable = { which = br_ship_corvette value = 1 }
        multiply_variable = { which = br_ship_corvette value = br_power_level }

        # 驱逐舰：level × 0.5，需 level ≥ 2
        set_variable = { which = br_ship_destroyer value = 0 }
        if = {
            limit = { check_variable = { which = br_power_level value >= 2 } }
            set_variable = { which = br_ship_destroyer value = 0.5 }
            multiply_variable = { which = br_ship_destroyer value = br_power_level }
        }

        # 巡洋舰：level × 0.25，需 level ≥ 4
        set_variable = { which = br_ship_cruiser value = 0 }
        if = {
            limit = { check_variable = { which = br_power_level value >= 4 } }
            set_variable = { which = br_ship_cruiser value = 0.25 }
            multiply_variable = { which = br_ship_cruiser value = br_power_level }
        }

        # 战列舰：level × 0.125，需 level ≥ 8
        set_variable = { which = br_ship_battleship value = 0 }
        if = {
            limit = { check_variable = { which = br_power_level value >= 8 } }
            set_variable = { which = br_ship_battleship value = 0.125 }
            multiply_variable = { which = br_ship_battleship value = br_power_level }
        }

        floor_variable = br_ship_corvette
        floor_variable = br_ship_destroyer
        floor_variable = br_ship_cruiser
        floor_variable = br_ship_battleship

        country_event = { id = reward.4101 }
    }
}

                        angle = random
