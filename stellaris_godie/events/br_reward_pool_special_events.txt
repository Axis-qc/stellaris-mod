namespace = br

# 特殊奖励池。ID 6970-6989。
# 提供舰队指挥官、科学家和维修支援舰船奖励，按等级划分。

# ========================
# 特殊奖励池分流器
# ========================
country_event = {
    id = br.6970
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { check_variable = { which = br_power_level value < 3 } }
            country_event = { id = br.6971 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 5 } }
            country_event = { id = br.6972 }
        }
        else = {
            country_event = { id = br.6973 }
        }
    }
}

# ========================
# 1-2级特殊奖励（初级指挥官/科学家/小型支援舰队）
# ========================
country_event = {
    id = br.6971
    is_triggered_only = yes
    title = BR_REWARD_SPECIAL_LEVEL1_TITLE
    desc = BR_REWARD_SPECIAL_LEVEL1_DESC
    picture = GFX_evt_ship_in_orbit

    option = {
        name = BR_REWARD_SPECIAL_ADMIRAL1
        custom_tooltip = BR_REWARD_SPECIAL_ADMIRAL1_TT
        hidden_effect = {
            create_leader = {
                class = commander
                species = owner_main_species
                name = random
                skill = 2
                traits = {
                    trait = leader_trait_aggressive
                }
            }
        }
    }
    option = {
        name = BR_REWARD_SPECIAL_SCIENTIST1
        custom_tooltip = BR_REWARD_SPECIAL_SCIENTIST1_TT
        hidden_effect = {
            create_leader = {
                class = scientist
                species = owner_main_species
                name = random
                skill = 2
                traits = {
                    trait = leader_trait_roamer
                }
            }
        }
    }
    option = {
        name = BR_REWARD_SPECIAL_FLEET1
        trigger = { any_owned_fleet = { } }
        custom_tooltip = BR_REWARD_SPECIAL_FLEET1_TT
        hidden_effect = {
            random_owned_fleet = { save_event_target_as = br_reward_spawn_fleet }
            create_fleet = {
                name = random
                effect = {
                    set_owner = root

                        random_list = { 50 = { create_ship = { name = random design = "BR_PVE_ARTILLERY_DESTROYER" } } 50 = { create_ship = { name = random design = "BR_PVE_BRAWLER_DESTROYER" } } }
                        random_list = { 50 = { create_ship = { name = random design = "BR_PVE_ARTILLERY_CORVETTE" } } 50 = { create_ship = { name = random design = "BR_PVE_BRAWLER_CORVETTE" } } }

                    set_location = { target = event_target:br_reward_spawn_fleet distance = 35 angle = random }
                }
            }
        }
    }
    option = { name = BR_REWARD_OPTION_ALLOY add_resource = { alloys = @reward_alloy_base mult = br_power_level } }
    option = { name = BR_REWARD_OPTION_ENERGY add_resource = { energy = @reward_energy_base mult = br_power_level } }
    option = { name = BR_REWARD_OPTION_MOTES add_resource = { volatile_motes = @reward_strategic_base mult = br_power_level } }
    option = { name = BR_REWARD_OPTION_CRYSTALS add_resource = { rare_crystals = @reward_strategic_base mult = br_power_level } }
    option = { name = BR_REWARD_OPTION_GASES add_resource = { exotic_gases = @reward_strategic_base mult = br_power_level } }
}
