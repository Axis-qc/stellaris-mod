namespace = lev

# 利维坦·天龙 技能事件（lev.1900~1999，不得越界）
# on_action 触发时 THIS = 施放舰队、from = 施放舰队
# 数值动态：add_modifier 的 mult = root.owner.value:lev_sk_*（等级驱动，
# 见 common/script_values/lev_skill_values.txt）；同技能内曲线不同的字段
# 按维度拆成多个 modifier 分别挂（一个 add_modifier 只有一个 mult）。
# 本期不做直接伤害，敌舰一律用减益（modifier）替代

# sk1 潮汐吐息：本星系敌舰伤害/射速减益
fleet_event = {
    id = lev.1900
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        from.owner = { save_event_target_as = lev_caster }

        solar_system = {
            every_ship_in_system = {
                limit = {
                    owner = {
                        is_hostile = event_target:lev_caster
                    }
                }
                add_modifier = { modifier = lev_aquatic_sk1_dmg mult = root.owner.value:lev_sk_aqua_1_dmg days = 20 }
                add_modifier = { modifier = lev_aquatic_sk1_fr mult = root.owner.value:lev_sk_aqua_1_fr days = 20 }
            }
        }
    }
}

# sk2 深渊鳞壁：本舰护甲与再生增益
fleet_event = {
    id = lev.1901
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        every_owned_ship = {
            limit = { is_ship_size = lev_aquatic }
            add_modifier = { modifier = lev_aquatic_sk2_am mult = root.owner.value:lev_sk_aqua_2_am days = 30 }
            add_modifier = { modifier = lev_aquatic_sk2_hr mult = root.owner.value:lev_sk_aqua_2_hr days = 30 }
        }
    }
}
