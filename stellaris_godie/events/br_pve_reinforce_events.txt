namespace = br

# ============================================================
# PVE敌方年度增援事件：br.830
# 由 on_yearly_pulse_country 年度触发
# ============================================================

country_event = {
    id = br.830
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_country_type = br_player_pve
        exists = event_target:br_pve_enemy_country
        any_owned_fleet = { is_ship_size = br_dominator }
    }

    immediate = {
        random_owned_fleet = {
            limit = { is_ship_size = br_dominator }
            solar_system = {
                random_neighbor_system = {
                    limit = {
                        NOT = { any_fleet_in_system = { owner = { is_country_type = br_player_pve } } }
                    }
                    save_event_target_as = br_reinf_spawn_@root
                }
            }
        }

        if = {
            limit = { exists = event_target:br_reinf_spawn_@root }
            event_target:br_pve_enemy_country = {
                create_fleet = {
                    name = "BR_PVE_REINFORCEMENT"
                    effect = {
                        set_owner = PREV
                        set_location = event_target:br_reinf_spawn_@root
                        br_reinforce_pve_fleet = yes
                    }
                }
            }
        }

        if = {
            limit = { exists = event_target:br_reinf_spawn_@root }
            create_message = {
                type = FLEET_DETECTED_MESSAGE_TYPE
                localization = br.830.toast
                days = 30
                custom_toast_content_icon = GFX_message_fleet_detected
            }
        }
    }
}

# ============================================================
# PVE敌方年度突击编队：br.840
# 由 on_yearly_pulse_country 年度触发
# 以玩家母舰为中心，周边10跳内随机非危险区星系刷敌
# 每个玩家同时只存在一支，死亡后才按等级刷下一支
# ============================================================

country_event = {
    id = br.840
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_country_type = br_player_pve
        exists = event_target:br_pve_enemy_country
        any_owned_fleet = { is_ship_size = br_dominator }
    }

    immediate = {
        random_owned_fleet = {
            limit = { is_ship_size = br_dominator }
            solar_system = {
                save_event_target_as = br_dominator_target_@root
            }
        }

        if = {
            limit = { exists = event_target:br_assault_fleet_@root }
            event_target:br_assault_fleet_@root = {
                clear_fleet_actions = this
                queue_actions = { move_to = event_target:br_dominator_target_@root }
            }
        }
        else = {
            event_target:br_dominator_target_@root = {
                random_system = {
                    limit = {
                        distance = { source = PREV max_jumps = 10 }
                        NOT = { has_star_flag = br_danger }
                        NOT = { any_fleet_in_system = { owner = { is_country_type = br_player_pve } } }
                    }
                    save_event_target_as = br_assault_spawn_@root
                }
            }

            if = {
                limit = { exists = event_target:br_assault_spawn_@root }
                event_target:br_pve_enemy_country = {
                    create_fleet = {
                        name = "BR_PVE_ASSAULT"
                        effect = {
                            set_owner = PREV
                            set_location = event_target:br_assault_spawn_@root
                            br_reinforce_pve_fleet = yes
                            set_fleet_stance = aggressive
                            queue_actions = {
                                move_to = event_target:br_dominator_target_@root
                            }
                        }
                    }
                    last_created_fleet = {
                        save_event_target_as = br_assault_fleet_@root
                    }
                }
                create_message = {
                    type = FLEET_DETECTED_MESSAGE_TYPE
                    localization = br.840.toast
                    days = 30
                    custom_toast_content_icon = GFX_message_fleet_detected
                }
            }
        }
    }
}
