namespace = br
# ============================================================
# PVE母舰追踪刷新：br.830
# 由 br.850 年度多次触发（11/100/200/300天）
# 遍历敌方所有带 br_assault_@root flag 的舰队，重新绑定跟随攻击母舰
# ============================================================
country_event = {
	id = br.830
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_country_flag = br_player_country
		exists = event_target:br_pve_enemy_country
		any_owned_fleet = {
			br_is_arkship_ship = yes
		}
	}
	immediate = {
		random_owned_fleet = {
			limit = {
				br_is_arkship_ship = yes
			}
			save_event_target_as = br_dominator_fleet_@root
		}
		event_target:br_pve_enemy_country = {
			every_owned_fleet = {
				limit = {
					has_fleet_flag = br_assault_@root
				}
				auto_follow_fleet = {
					target = event_target:br_dominator_fleet_@root
					attack_fleet = yes
				}
			}
		}
	}
}

# ============================================================
# PVE敌方年度突击编队：br.840
# 由 br.850 年度触发，每年无条件刷一支新舰队
# 以玩家母舰为中心，周边2-5跳内随机非危险区星系刷敌
# 舰队打上 br_assault_@root flag，可累积多支
# ============================================================
country_event = {
	id = br.840
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_country_flag = br_player_country
		exists = event_target:br_pve_enemy_country
		any_owned_fleet = {
			br_is_arkship_ship = yes
		}
	}
	immediate = {
		random_owned_fleet = {
			limit = {
				br_is_arkship_ship = yes
			}
			save_event_target_as = br_dominator_fleet_@root
			solar_system = {
				save_event_target_as = br_dominator_target_@root
			}
		}
		# 只负责创建，追踪交给 br.830
		event_target:br_dominator_target_@root = {
			random_system = {
				limit = {
					distance = {
						source = event_target:br_dominator_target_@root
						type = hyperlane
						min_jumps = 2
						max_jumps = 5
					}
					NOT = {
						has_star_flag = br_danger
					}
					NOT = {
						any_fleet_in_system = {
							owner = {
								has_country_flag = br_player_country
							}
						}
					}
				}
				save_event_target_as = br_assault_spawn_@root
			}
		}
		event_target:br_pve_enemy_country = {
			create_fleet = {
				name = "BR_PVE_ASSAULT"
				effect = {
					set_owner = PREV
					set_location = event_target:br_assault_spawn_@root
					br_reinforce_pve_fleet = yes
					set_fleet_flag = br_assault_@root
					set_fleet_stance = aggressive
					auto_follow_fleet = {
						target = event_target:br_dominator_fleet_@root
						attack_fleet = yes
					}
					save_global_event_target_as = br_assault_fleet_@root
				}
			}
		}
		create_message = {
			type = FLEET_DETECTED_MESSAGE_TYPE
			localization = br.840.toast
			days = 30
			custom_toast_content_icon = GFX_message_fleet_detected
		}
	}
}

# ============================================================
# 年度舰船设计复制 + 触发刷敌：br.850
# 由 on_yearly_pulse_country 年度触发
# 保存玩家舰船设计为全局 event_target → 无条件触发 br.840 刷敌
# 延迟多次触发 br.830 更新追击目标
# ============================================================
country_event = {
	id = br.850
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_country_flag = br_player_country
		any_owned_fleet = {
			br_is_arkship_ship = yes
		}
		NOT = {
			has_global_flag = br_designs_saved
		}
	}
	immediate = {
		# TODO: 未适配生物舰船 — 生物舰船没有 is_ship_size = corvette/destroyer/cruiser/battleship，
		# random_owned_design 无法匹配，导致设计保存为空，create_ship 刷不出船。
		# 后续需改用 any_owned_ship 遍历实际舰船来获取设计，或准备一套后备固定设计。
		random_owned_design = {
			limit = {
				is_ship_size = corvette
			}
			save_global_event_target_as = br_player_corvette
		}
		if = {
			limit = {
				check_variable = {
					which = br_power_level
					value >= 3
				}
			}
			random_owned_design = {
				limit = {
					is_ship_size = destroyer
				}
				save_global_event_target_as = br_player_destroyer
			}
		}
		if = {
			limit = {
				check_variable = {
					which = br_power_level
					value >= 5
				}
			}
			random_owned_design = {
				limit = {
					is_ship_size = cruiser
				}
				save_global_event_target_as = br_player_cruiser
			}
		}
		if = {
			limit = {
				check_variable = {
					which = br_power_level
					value >= 8
				}
			}
			random_owned_design = {
				limit = {
					is_ship_size = battleship
				}
				save_global_event_target_as = br_player_battleship
			}
		}
		# 年度无条件刷新突击舰队
		country_event = {
			id = br.840
			days = 10
		}
		# 更新追踪玩家星系（auto_follow_fleet 会持续追击，每年一次即可）
		country_event = {
			id = br.830
			days = 11
		}
		set_timed_global_flag = {
			flag = br_designs_saved
			days = 100
		}
	}
}
