# 舰队集合到母舰位置（由 slot=1 action 触发）
fleet_event = {
    id = heroship.1001
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_vanguard
        }
    }

    immediate = {
        owner = {
            random_owned_ship = {
                limit = {
                    is_ship_size = br_dominator
                }
                fleet = {
                    save_event_target_as = br_mothership_fleet
                }
            }
        }

        solar_system = {
            every_fleet_in_system = {
                limit = {
                    exists = controller
                    controller = {
                        is_same_value = root.owner
                    }
                    NOT = { is_same_value = root }
                }
                set_location = event_target:br_mothership_fleet
            }
        }
    }
}

# 主动技能变量 CD 每日递减。
country_event = {
    id = heroship.1014
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = {
                br_any_vanguard_active_skill_cd_running = yes
            }

            if = {
                limit = { br_vanguard_power_skill_cd_running = yes }
                change_variable = { which = br_vanguard_power_skill_cd value = -1 }
            }
            if = {
                limit = { br_vanguard_defense_skill_cd_running = yes }
                change_variable = { which = br_vanguard_defense_skill_cd value = -1 }
            }
            if = {
                limit = { br_vanguard_support_skill_cd_running = yes }
                change_variable = { which = br_vanguard_support_skill_cd value = -1 }
            }

            if = {
                limit = {
                    br_any_vanguard_active_skill_cd_running = yes
                }
                country_event = { id = heroship.1014 days = 1 }
            }
            else = {
                remove_country_flag = br_vanguard_skill_cd_tick_running
            }
        }
        else = {
            remove_country_flag = br_vanguard_skill_cd_tick_running
        }
    }
}
