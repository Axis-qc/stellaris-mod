# br.900：最终圈生成PVE Boss和护卫舰队喵
country_event = {
    id = br.900
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        has_global_flag = br_pve_initialized
        # has_global_flag = br_safe_1
        NOT = { has_global_flag = br_boss_spawned }
        exists = event_target:br_pve_enemy_country
    }

    immediate = {
        set_global_flag = br_boss_spawned

        # 当前先使用固定Boss模板，后续在这里接入虚空倾向权重选择喵
        event_target:battle_royale_center = {
            star = { save_event_target_as = br_boss_spawn_tmp }
        }

        event_target:br_pve_enemy_country = {
            create_fleet = {
                name = "BR_PVE_BOSS"
                effect = {
                    set_owner = event_target:br_pve_enemy_country
                    create_ship = { name = "BR_PVE_BOSS" design = "BR_PVE_BOSS" prefix = no }
                    set_location = { target = event_target:br_boss_spawn_tmp distance = 0 }

# br.901：Boss击杀胜利，由on_ship_destroyed_victim触发喵
country_event = {
    id = br.901
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        has_global_flag = br_boss_spawned
        NOT = { has_global_flag = br_pve_victory }
        is_country_type = br_pve_enemy
        fromfrom = { is_ship_size = br_pve_boss }
    }

    immediate = {
        set_global_flag = br_pve_victory

        every_country = {
            limit = { is_country_type = br_player_pve }
            set_country_flag = br_pve_victory
            add_resource = {
                alloys = 5000
                energy = 10000
                minerals = 5000
                unity = 5000
            }
        }

        # 房主弹出周目选择，非房主弹胜利通知喵
        event_target:br_pve_alliance_leader = {
            country_event = { id = br.950 }
        }
        every_country = {
            limit = {
                is_country_type = br_player_pve
                NOT = { is_same_value = event_target:br_pve_alliance_leader }
            }
            country_event = { id = br.902 }
        }
    }
}
