namespace = br
# ============================================================
# PVE 进星系生成敌军：br.804 / br.801
# 链路：br.804 进星系标记周边邻星 → br.801 遍历标记星系生成敌军
# 规则：仅空星系（星系内无任何舰队）才标记/生成，防止同星系重复生成多支敌军舰队
# ============================================================

# ============================================================
# br.804：进星系触发，标记周边1跳邻星 + 自身5年冷却 → br.801
# ============================================================
fleet_event = {
	id = br.804
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_global_flag = br_mode_pve
		owner = {
			NOR = {
				is_country_type = br_void
				is_country_type = br_pve_enemy
			}
		}
		solar_system = {
			NOT = {
				has_star_flag = br_system_spawn_center_used
			}
		}
	}
	immediate = {
		this = {
			solar_system = {
				# 中心冷却5年
				set_timed_star_flag = {
					flag = br_system_spawn_center_used
					days = 1825
				}
				# 周边1跳邻星：仅空星系（星系内无任何舰队）打1年标记
				every_neighbor_system = {
					limit = {
						# 空星系判定：星系内无任何舰队
						NOT = {
							any_fleet_in_system = { always = yes }
						}
					}
					set_timed_star_flag = {
						flag = br_pve_spawn_mark
						days = 365
					}
				}
			}
		}
		owner = {
			country_event = {
				id = br.801
			}
		}
	}
}

# ============================================================
# br.801：遍历标记星系生成敌军（仅空星系），生成后清除标记
# ============================================================
country_event = {
	id = br.801
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_global_flag = br_mode_pve
		exists = event_target:br_pve_enemy_country
		owner = {
			NOR = {
				is_country_type = br_void
				is_country_type = br_pve_enemy
			}
		}
	}
	immediate = {
		every_system = {
			limit = {
				has_star_flag = br_pve_spawn_mark
				NOT = {
					has_star_flag = br_danger
				}
				NOT = {
					has_star_flag = br_player_spawn_system
				}
				# 仅空星系（星系内无任何舰队）才生成，防止同星系重复生成
				NOT = {
					any_fleet_in_system = { always = yes }
				}
			}
			# 生成后清除标记
			remove_star_flag = br_pve_spawn_mark
			# 生成一支 grunt 舰队
			create_fleet = {
				name = "BR_PVE_GARRISON_FLEET"
				effect = {
					set_owner = event_target:br_pve_enemy_country
					set_location = {
						target = prev
						distance = 50
						angle = random
					}
					save_global_event_target_as = br_pve_enemy_fleet@root
					br_spawn_crisis_fleet = yes
				}
			}
		}
	}
}
