namespace = br

# ============================================================
# PVE内圈天灾刷敌：br.801
# 取消原来 br.801-br.813 的13个事件，统一由本事件处理
# 每天随机挑1个未刷的1-25圈星系刷天灾舰队
# 数量由 br_spawn_crisis_fleet 根据 br_spawn_tmp 的圈数flag自算
# ============================================================
country_event = {
    id = br.801
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        has_global_flag = br_pve_spawn_outer_done
        NOT = { has_global_flag = br_pve_initialized }
        exists = event_target:br_pve_enemy_country
    }

    immediate = {
        # 从1-25圈中随机挑1个未刷过的星系
        random_system = {
            limit = {
                OR = {
                    has_star_flag = br_ring_1  has_star_flag = br_ring_2
                    has_star_flag = br_ring_3  has_star_flag = br_ring_4
                    has_star_flag = br_ring_5  has_star_flag = br_ring_6
                    has_star_flag = br_ring_7  has_star_flag = br_ring_8
                    has_star_flag = br_ring_9  has_star_flag = br_ring_10
                    has_star_flag = br_ring_11 has_star_flag = br_ring_12
                    has_star_flag = br_ring_13 has_star_flag = br_ring_14
                    has_star_flag = br_ring_15 has_star_flag = br_ring_16
                    has_star_flag = br_ring_17 has_star_flag = br_ring_18
                    has_star_flag = br_ring_19 has_star_flag = br_ring_20
                    has_star_flag = br_ring_21 has_star_flag = br_ring_22
                    has_star_flag = br_ring_23 has_star_flag = br_ring_24
                    has_star_flag = br_ring_25
                }
                NOT = { has_star_flag = br_system_spawned }
            }
            save_global_event_target_as = br_spawn_tmp
            set_star_flag = br_system_spawned
            event_target:br_pve_enemy_country = { br_spawn_crisis_fleet = yes }
        }

        # 还有未刷星系 → 明天继续
        if = {
            limit = {
                any_system = {
                    OR = {
                        has_star_flag = br_ring_1  has_star_flag = br_ring_2
                        has_star_flag = br_ring_3  has_star_flag = br_ring_4
                        has_star_flag = br_ring_5  has_star_flag = br_ring_6
                        has_star_flag = br_ring_7  has_star_flag = br_ring_8
                        has_star_flag = br_ring_9  has_star_flag = br_ring_10
                        has_star_flag = br_ring_11 has_star_flag = br_ring_12
                        has_star_flag = br_ring_13 has_star_flag = br_ring_14
                        has_star_flag = br_ring_15 has_star_flag = br_ring_16
                        has_star_flag = br_ring_17 has_star_flag = br_ring_18
                        has_star_flag = br_ring_19 has_star_flag = br_ring_20
                        has_star_flag = br_ring_21 has_star_flag = br_ring_22
                        has_star_flag = br_ring_23 has_star_flag = br_ring_24
                        has_star_flag = br_ring_25
                    }
                    NOT = { has_star_flag = br_system_spawned }
                }
            }
            event_target:br_pve_enemy_country = { country_event = { id = br.801 days = 1 } }
        }
        # 全刷完了 → 初始化完成
        else = { set_global_flag = br_pve_initialized }
    }
}
