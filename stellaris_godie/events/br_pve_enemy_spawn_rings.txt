# ============================================================
# PVE内圈刷敌：br.801-br.813
# 每事件管2圈，每次处理1星系并自触发直到刷完，外圈弱→内圈强平滑递增，精英在1-3圈
# 默认编队 1battleship+2cruiser+4destroyer+8corvette+4frigate × N
# ============================================================

# br.801：ring 25-24，N=1
country_event = {
    id = br.801
    hide_window = yes
    is_triggered_only = yes
    trigger = { has_global_flag = br_mode_pve NOT = { has_global_flag = br_pve_initialized } has_global_flag = br_pve_spawn_outer_done NOT = { has_global_flag = br_pve_spawn_ring_24_25_done } exists = event_target:br_pve_enemy_country }
    immediate = {
        random_system = {
            limit = { OR = { has_star_flag = br_ring_24 has_star_flag = br_ring_25 } NOT = { has_star_flag = br_system_spawned } }
            star = { save_global_event_target_as = br_spawn_tmp }
            set_star_flag = br_system_spawned
            event_target:br_pve_enemy_country = { br_spawn_pve_fleet = { N = 1 } }
        }
        if = { limit = { any_system = { OR = { has_star_flag = br_ring_24 has_star_flag = br_ring_25 } NOT = { has_star_flag = br_system_spawned } } }
            country_event = { id = br.801 days = 1 } }
        else = { set_global_flag = br_pve_spawn_ring_24_25_done country_event = { id = br.802 days = 3 } }
    }
}
