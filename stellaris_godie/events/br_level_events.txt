# 事件ID区间：1100-1199。新增或修改事件ID不得超出此区间喵
# br.1100：初始化PVE玩家个人经验和等级变量喵
country_event = {
    id = br.1100
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        is_country_type = br_player_pve
        NOT = { has_country_flag = br_level_initialized }
    }

    immediate = {
        set_country_flag = br_level_initialized
        set_variable = {
            which = br_combat_score
            value = 0
        }
        set_variable = {
            which = br_power_level
            value = 1
        }
        set_variable = { which = br_kill_score value = 0 }
        give_technology = { tech = tech_corvettes message = no }
        set_variable = { which = br_level_month_counter value = 0 }

        # Boss倾向变量只挂在PVE房主国家上，后续最终Boss选择也必须从这个作用域读取喵
        if = {
            limit = { exists = event_target:br_pve_alliance_leader }
            event_target:br_pve_alliance_leader = {
                country_event = { id = br.1101 }
            }
        }
    }
}

# br.1102：摧毁PVE敌方舰船时给击杀者经验，并给房主累计征服倾向喵
country_event = {
    id = br.1102
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        is_country_type = br_player_pve
        from = { is_country_type = br_pve_enemy }
    }

    immediate = {
        country_event = { id = br.1100 }

        if = {
            limit = { fromfromfrom = { is_ship_size = br_pve_boss } }
            change_variable = {
                which = br_combat_score
                value = 1000
            }
            change_variable = {
                which = br_kill_score
                value = 1000
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.1101 }
                    change_variable = {
                        which = br_void_conquest
                        value = 20
                    }
                }
            }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = battleship } }
            change_variable = {
                which = br_combat_score
                value = 80
            }
            change_variable = {
                which = br_kill_score
                value = 80
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.1101 }
                    change_variable = {
                        which = br_void_conquest
                        value = 5
                    }
                }
            }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = cruiser } }
            change_variable = {
                which = br_combat_score
                value = 45
            }
            change_variable = {
                which = br_kill_score
                value = 45
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.1101 }
                    change_variable = {
                        which = br_void_conquest
                        value = 3
                    }
                }
            }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = destroyer } }
            change_variable = {
                which = br_combat_score
                value = 25
            }
            change_variable = {
                which = br_kill_score
                value = 25
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.1101 }
                    change_variable = {
                        which = br_void_conquest
                        value = 2
                    }
                }
            }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = corvette } }
            change_variable = {
                which = br_combat_score
                value = 12
            }
            change_variable = {
                which = br_kill_score
                value = 12
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.1101 }
                    change_variable = {
                        which = br_void_conquest
                        value = 1
                    }
                }
            }
        }
        else = {
            change_variable = {
                which = br_combat_score
                value = 5
            }
            change_variable = {
                which = br_kill_score
                value = 5
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.1101 }
                    change_variable = {
                        which = br_void_conquest
                        value = 1
                    }
                }
            }
        }

        # 每100击杀分触发一次奖励
        while = {
            limit = {
                check_variable = {
                    which = br_kill_score
                    value >= 100
                }
            }
            change_variable = { which = br_kill_score value = -100 }
            country_event = { id = br.6110 }
        }

        country_event = { id = br.1104 }
    }
}

# br.1104：根据累计经验重算玩家等级，并在首次升级时给房主累计进化倾向喵
country_event = {
    id = br.1104
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        is_country_type = br_player_pve
    }

    immediate = {
        if = {
            limit = { check_variable = { which = br_combat_score value >= 20000 } }
            if = {
                limit = { check_variable = { which = br_power_level value < 12 } exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = { country_event = { id = br.1101 } change_variable = { which = br_void_evolution value = 1 } }
                set_country_flag = br_reward_level_up_now
            }
            set_variable = { which = br_power_level value = 12 }
            every_country = { limit = { is_country_type = br_player_pve } set_variable = { which = br_power_level value = 12 } }

        else_if = {
            limit = { check_variable = { which = br_combat_score value >= 14000 } }

                limit = { check_variable = { which = br_power_level value < 11 } exists = event_target:br_pve_alliance_leader }

# br.1111: 年度等级窗口分流入口。由 br.1105 每年调用一次。
country_event = {
    id = br.1111
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        is_country_type = br_player_pve
    }

    immediate = {
        if = { limit = { check_variable = { which = br_power_level value >= 12 } } country_event = { id = br.1123 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 11 } } country_event = { id = br.1122 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 10 } } country_event = { id = br.1121 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 9 } } country_event = { id = br.1120 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 8 } } country_event = { id = br.1119 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 7 } } country_event = { id = br.1118 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 6 } } country_event = { id = br.1117 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 5 } } country_event = { id = br.1116 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 4 } } country_event = { id = br.1115 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 3 } } country_event = { id = br.1114 } }
        else_if = { limit = { check_variable = { which = br_power_level value >= 2 } } country_event = { id = br.1113 } }
        else = { country_event = { id = br.1112 } }
    }
}
