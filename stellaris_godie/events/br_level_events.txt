namespace = br

# br.770：初始化PVE玩家个人经验和等级变量喵
country_event = {
    id = br.770
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        is_country_type = br_player_pve
        NOT = { has_country_flag = br_level_initialized }
    }

    immediate = {
        set_country_flag = br_level_initialized
        set_variable = {
            which = br_combat_score
            value = 0
        }
        set_variable = {
            which = br_power_level
            value = 1
        }

        # Boss倾向变量只挂在PVE房主国家上，后续最终Boss选择也必须从这个作用域读取喵
        if = {
            limit = { exists = event_target:br_pve_alliance_leader }
            event_target:br_pve_alliance_leader = {
                country_event = { id = br.774 }
            }
        }
    }
}

# br.771：摧毁PVE敌方舰船时给击杀者经验，并给房主累计征服倾向喵
country_event = {
    id = br.771
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        is_country_type = br_player_pve
        from = { is_country_type = br_pve_enemy }
    }

    immediate = {
        country_event = { id = br.770 }

        if = {
            limit = { fromfromfrom = { is_ship_size = br_pve_boss } }
            change_variable = {
                which = br_combat_score
                value = 1000
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_conquest
                        value = 20
                    }
                }
            }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = battleship } }
            change_variable = {
                which = br_combat_score
                value = 80
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_conquest
                        value = 5
                    }
                }
            }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = cruiser } }
            change_variable = {
                which = br_combat_score
                value = 45
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_conquest
                        value = 3
                    }
                }
            }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = destroyer } }
            change_variable = {
                which = br_combat_score
                value = 25
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_conquest
                        value = 2
                    }
                }
            }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = corvette } }
            change_variable = {
                which = br_combat_score
                value = 12
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_conquest
                        value = 1
                    }
                }
            }
        }
        else = {
            change_variable = {
                which = br_combat_score
                value = 5
            }
            if = {
                limit = { exists = event_target:br_pve_alliance_leader }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_conquest
                        value = 1
                    }
                }
            }
        }

        country_event = { id = br.773 }
    }
}

# br.773：根据累计经验重算玩家等级，并在首次升级时给房主累计进化倾向喵
country_event = {
    id = br.773
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        is_country_type = br_player_pve
    }

    immediate = {
        if = {
            limit = {
                check_variable = {
                    which = br_combat_score
                    value >= 3000
                }
            }
            if = {
                limit = {
                    check_variable = {
                        which = br_power_level
                        value < 6
                    }
                    exists = event_target:br_pve_alliance_leader
                }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_evolution
                        value = 1
                    }
                }
            }
            set_variable = {
                which = br_power_level
                value = 6
            }
        }
        else_if = {
            limit = {
                check_variable = {
                    which = br_combat_score
                    value >= 1800
                }
            }
            if = {
                limit = {
                    check_variable = {
                        which = br_power_level
                        value < 5
                    }
                    exists = event_target:br_pve_alliance_leader
                }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_evolution
                        value = 1
                    }
                }
            }
            set_variable = {
                which = br_power_level
                value = 5
            }
        }
        else_if = {
            limit = {
                check_variable = {
                    which = br_combat_score
                    value >= 1000
                }
            }
            if = {
                limit = {
                    check_variable = {
                        which = br_power_level
                        value < 4
                    }
                    exists = event_target:br_pve_alliance_leader
                }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_evolution
                        value = 1
                    }
                }
            }
            set_variable = {
                which = br_power_level
                value = 4
            }
        }
        else_if = {
            limit = {
                check_variable = {
                    which = br_combat_score
                    value >= 450
                }
            }
            if = {
                limit = {
                    check_variable = {
                        which = br_power_level
                        value < 3
                    }
                    exists = event_target:br_pve_alliance_leader
                }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_evolution
                        value = 1
                    }
                }
            }
            set_variable = {
                which = br_power_level
                value = 3
            }
        }
        else_if = {
            limit = {
                check_variable = {
                    which = br_combat_score
                    value >= 150
                }
            }
            if = {
                limit = {
                    check_variable = {
                        which = br_power_level
                        value < 2
                    }
                    exists = event_target:br_pve_alliance_leader
                }
                event_target:br_pve_alliance_leader = {
                    country_event = { id = br.774 }
                    change_variable = {
                        which = br_void_evolution
                        value = 1
                    }
                }
            }
            set_variable = {
                which = br_power_level
                value = 2
            }
        }
        else = {
            set_variable = {
                which = br_power_level
                value = 1
            }
        }
    }
}
