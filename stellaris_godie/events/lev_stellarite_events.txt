# 发放（隐藏）：置 flag、刷设计、初始化变量
country_event = {
    id = lev.10
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_civic = lev_star_overlords
        host_has_dlc = "Leviathans Story Pack"
        NOT = {
            has_country_flag = lev_stellarite_unlocked
        }
    }
    immediate = {
        # 顺序不能改：先置 flag，否则自动设计里不会有 lev_stellarite
        set_country_flag = lev_stellarite_unlocked
        refresh_auto_generated_ship_designs = yes
        set_variable = {
            which = lev_stellarite_energy
            value = 0
        }
        # 成长值初始化：每只利维坦独立，TYPE 决定变量名后缀
        lev_growth_init = { TYPE = stellarite }
        # 延迟 1 天，给设计生成留结算窗口
        country_event = {
            id = lev.11
            days = 1
        }
    }
}

# 星系选择器 on_started 触发时舰队仍在原地（README 37：无需先抵达）
system_event = {
    id = lev.102
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        # 跨 20 天结算，event 级目标会随本事件注销，只能用 global；@root 按国家隔离，防多玩家串目标
        save_global_event_target_as = lev_beam_target@root
    }
    after = {
        # 清掉选择器排下的行军令并就地按住（同原版 02_contracts.txt:2434）
        if = {
            limit = { from = { is_scope_type = fleet } }
            from = {
                clear_fleet_actions = this
                clear_orders = yes
                queue_actions = {
                    wait = {
                        duration = 20
                    }
                }
            }
        }
        # 减速锁位，不用 set_event_locked（那会让舰船不可攻击）
        from = {
            set_timed_fleet_flag = { flag = lev_beam_charging days = 20 }
            every_owned_ship = {
                limit = { is_ship_size = lev_stellarite }
                add_modifier = { modifier = lev_stellarite_beam_charging days = 20 }
                lock_animation_state = breath_loop
            }
            fleet_event = { id = lev.106 days = 20 }
        }
    }
}

    }
}
