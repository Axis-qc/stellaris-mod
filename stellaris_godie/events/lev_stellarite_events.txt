namespace = lev

# 开局选择窗：带「星海共主」理念的玩家在此选定初始巨兽
# 本期只有噬星者一项，为后续多利维坦预留结构
country_event = {
    id = lev.12
    is_triggered_only = yes
    title = "lev.12.name"
    desc = "lev.12.desc"
    picture = GFX_evt_fleet_neutral
    trigger = {
        has_civic = lev_star_overlords
        host_has_dlc = "Leviathans Story Pack"
        NOT = {
            has_country_flag = lev_stellarite_unlocked
        }
    }
    option = {
        name = "lev.12.a"
        country_event = {
            id = lev.10
        }
    }
}

# 发放（隐藏）：置 flag、刷设计、初始化变量
# 不再由 on_game_start_country 直接调用，改由 lev.12 的选项触发
country_event = {
    id = lev.10
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_civic = lev_star_overlords
        # 基座实体 stellarite_entity 随利维坦 DLC 提供
        host_has_dlc = "Leviathans Story Pack"
        NOT = {
            has_country_flag = lev_stellarite_unlocked
        }
    }
    immediate = {
        # 顺序不能改：先置 flag，否则自动设计里不会有 lev_stellarite
        set_country_flag = lev_stellarite_unlocked
        refresh_auto_generated_ship_designs = yes
        set_variable = {
            which = lev_stellarite_energy
            value = 0
        }
        # 成长值系统初始化：成长值 / 成长等级 / 营养档位 / 阈值 / 各档消耗收益
        # 每只利维坦一套独立变量，TYPE 决定变量名后缀
        lev_growth_init = { TYPE = stellarite }
        # 延迟 1 天，给设计生成留结算窗口
        country_event = {
            id = lev.11
            days = 1
        }
    }
}

# 发放：折跃一艘到首都，同时保留星港建造权限
country_event = {
    id = lev.11
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        create_fleet = {
            name = "LEV_STELLARITE_FLEET"
            effect = {
                set_owner = root
                create_ship = {
                    name = random
                    random_existing_design = lev_stellarite
                    # 不能用 root，会把玩家文化套到守卫者模型上
                    graphical_culture = "guardian_01"
                }
                set_location = {
                    target = root.capital_star
                    distance = 35
                    angle = random
                }
            }
        }
    }
}
