namespace = lev

# 开局选择窗：选定初始巨兽（本期只有噬星者，为后续多利维坦预留结构）
country_event = {
    id = lev.12
    is_triggered_only = yes
    title = "lev.12.name"
    desc = "lev.12.desc"
    picture = GFX_evt_fleet_neutral
    trigger = {
        has_civic = lev_star_overlords
        host_has_dlc = "Leviathans Story Pack"
        NOT = {
            has_country_flag = lev_stellarite_unlocked
        }
    }
    option = {
        name = "lev.12.a"
        country_event = {
            id = lev.10
        }
    }
}

# 发放（隐藏）：置 flag、刷设计、初始化变量
country_event = {
    id = lev.10
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_civic = lev_star_overlords
        host_has_dlc = "Leviathans Story Pack"
        NOT = {
            has_country_flag = lev_stellarite_unlocked
        }
    }
    immediate = {
        # 顺序不能改：先置 flag，否则自动设计里不会有 lev_stellarite
        set_country_flag = lev_stellarite_unlocked
        refresh_auto_generated_ship_designs = yes
        set_variable = {
            which = lev_stellarite_energy
            value = 0
        }
        # 成长值初始化：每只利维坦独立，TYPE 决定变量名后缀
        lev_growth_init = { TYPE = stellarite }
        # 延迟 1 天，给设计生成留结算窗口
        country_event = {
            id = lev.11
            days = 1
        }
    }
}

# 发放：折跃一艘到首都，同时保留星港建造权限
country_event = {
    id = lev.11
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        create_fleet = {
            name = "LEV_STELLARITE_FLEET"
            effect = {
                set_owner = root
                create_ship = {
                    name = random
                    random_existing_design = lev_stellarite
                    # 不能用 root，会把玩家文化套到守卫者模型上
                    graphical_culture = "guardian_01"
                }
                set_location = {
                    target = root.capital_star
                    distance = 35
                    angle = random
                }
            }
        }
    }
}

# 技能完成：lev.101 THIS = 目标恒星、from = 舰队；lev.102 THIS = 舰队自身

planet_event = {
    id = lev.101
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        lev_stellarite_ignite_apply = yes
    }
}

fleet_event = {
    id = lev.102
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        lev_stellarite_beam_apply = yes
    }
}

# 灭星武器命中：THIS = 被开火的恒星，from = 施放舰队（照原版 planet_destruction.100 用 carrier_event）
