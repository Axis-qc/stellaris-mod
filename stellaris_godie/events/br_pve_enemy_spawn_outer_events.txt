# ============================================================
# 外圈杂兵分批刷敌：br.820-br.823
# 每事件管10圈，越靠近内圈舰队越强，有随机扰动
# 排除玩家初始出生星系
# ============================================================

# ============================================================
# br.820：ring 56-65
# ============================================================
country_event = {
    id = br.820
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_global_flag = br_mode_pve
        has_global_flag = br_pve_spawn_started
        NOT = { has_global_flag = br_pve_initialized }
        NOT = { has_global_flag = br_pve_spawn_outer_56_65_done }
        exists = event_target:br_pve_enemy_country
    }

    immediate = {
        every_system = {
            limit = { NOT = { has_star_flag = br_player_spawn_system } OR = { has_star_flag = br_ring_61 has_star_flag = br_ring_62 has_star_flag = br_ring_63 has_star_flag = br_ring_64 has_star_flag = br_ring_65 } }
            star = { save_global_event_target_as = br_spawn_tmp }
            event_target:br_pve_enemy_country = { br_spawn_pve_outer_fleet = { corvette = 2 destroyer = 0 } }
        }
        every_system = {
            limit = { NOT = { has_star_flag = br_player_spawn_system } OR = { has_star_flag = br_ring_56 has_star_flag = br_ring_57 has_star_flag = br_ring_58 has_star_flag = br_ring_59 has_star_flag = br_ring_60 } }
            star = { save_global_event_target_as = br_spawn_tmp }
            event_target:br_pve_enemy_country = { br_spawn_pve_outer_fleet = { corvette = 3 destroyer = 1 } }
        }
        set_global_flag = br_pve_spawn_outer_56_65_done
        country_event = { id = br.821 days = 3 }
    }
}
