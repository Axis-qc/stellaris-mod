# br.504：开局缩圈速度设置，完成设置后先扩圈，再进入PVP/PVE开局初始化喵
country_event = {
    id = br.504
    title = br_speed_settings_title
    desc = br_speed_settings_desc
    picture = GFX_evt_signing
    is_triggered_only = yes

    option = {
        name = br_speed_fast_name
        set_global_flag = br_speed_fast
        remove_global_flag = br_speed_normal
        remove_global_flag = br_speed_slow
        country_event = { id = br.100 }
        country_event = { id = br.300 days = 1080 }
        country_event = { id = br.600 days = 1080 }
        every_country = {
            limit = {
                is_ai = no
                NOT = { has_country_flag = br_player_replacement_scheduled }
                NOR = {
                    is_country_type = br_player
                    is_country_type = br_player_pvp
                    is_country_type = br_player_pve
                }
            }
            set_country_flag = br_player_replacement_scheduled
            country_event = { id = br.502 days = 1 }
        }
    }

    option = {
        name = br_speed_normal_name
        remove_global_flag = br_speed_fast
        set_global_flag = br_speed_normal
        remove_global_flag = br_speed_slow
        country_event = { id = br.100 }
        country_event = { id = br.300 days = 1080 }
        country_event = { id = br.600 days = 1080 }
        every_country = {
            limit = {
                is_ai = no
                NOT = { has_country_flag = br_player_replacement_scheduled }
                NOR = {
                    is_country_type = br_player
                    is_country_type = br_player_pvp
                    is_country_type = br_player_pve
                }
            }
            set_country_flag = br_player_replacement_scheduled
            country_event = { id = br.502 days = 1 }
        }
    }

    option = {
        name = br_speed_slow_name
        remove_global_flag = br_speed_fast
        remove_global_flag = br_speed_normal
        set_global_flag = br_speed_slow
        country_event = { id = br.100 }
        country_event = { id = br.300 days = 1080 }
        country_event = { id = br.600 days = 1080 }
        every_country = {
            limit = {
                is_ai = no
                NOT = { has_country_flag = br_player_replacement_scheduled }
                NOR = {
                    is_country_type = br_player
                    is_country_type = br_player_pvp
                    is_country_type = br_player_pve
                }
            }
            set_country_flag = br_player_replacement_scheduled
            country_event = { id = br.502 days = 1 }
        }
    }
}
