# 事件ID区间：6000-6099。新增或修改事件ID不得超出此区间喵
# br.6000：母舰采掘光束爆星后分流奖励事件，并把目标星球变成破碎星球喵
planet_event = {
    id = br.6000
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        save_event_target_as = br_harvested_planet
        set_planet_flag = br_harvested_planet

        from = {
            owner = {
                save_event_target_as = br_harvester_owner
                country_event = { id = br.1103 }
            }
        }

        # 普通奖励由 br.1103 直接触发 reward.1，等级重算由 br.1104 处理。
        # 爆星事件只负责基础资源、经验入口和星球处理。

        if = {
            limit = {
                is_colony = yes
            }
            destroy_colony = yes
        }

        clear_deposits = yes
        change_pc = pc_shattered
    }
}
