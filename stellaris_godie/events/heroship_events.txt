namespace = heroship
# ========================
# 英雄舰发放入口 — heroship.50
# 开局1年后由 on_yearly_pulse_country 触发
# BR/PVE 模式：直接发放英雄舰
# 普通模式：在首都隔壁星系创建古代造舰厂遗迹，探索3阶段后触发修复局势
# ========================
country_event = {
	id = heroship.50
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_ai = no
		NOT = {
			has_country_flag = br_hero_ship_received
		}
		NOT = {
			has_country_flag = br_hero_construction_started
		}
	}
	immediate = {
		if = {
			limit = {
				has_global_flag = br_mode_pve
			}
			country_event = {
				id = heroship.101
			}
		}
		else = {
			set_country_flag = br_hero_construction_started
			# 在首都隔壁星系的行星上创建古代造舰厂遗迹
			capital_scope = {
				solar_system = {
					random_neighbor_system = {
						random_system_planet = {
							limit = {
								is_colony = no
								has_anomaly = no
							}
							create_archaeological_site = site_heroship_flagship_wreckage
						}
					}
				}
			}
		}
	}
}

# ========================
# 英雄舰发放 — heroship.101
# 由 heroship.50 路由调用（BR/PVE 直接调用，普通模式由局势完成后调用）
# 核心已改为组件自由切换，本事件仅做通知 + 初始化等级系统，60天后先锋舰折跃抵达。
# ========================
country_event = {
	id = heroship.101
	title = heroship.101_title
	desc = heroship.101_desc
	is_triggered_only = yes
	trigger = {
		NOT = {
			has_country_flag = br_hero_ship_received
		}
	}
	immediate = {
		set_country_flag = br_hero_ship_received
		set_country_flag = br_hero_level_initialized
		set_variable = {
			which = br_hero_level
			value = 1
		}
		set_variable = {
			which = br_hero_xp
			value = 0
		}
		set_variable = {
			which = br_hero_xp_next
			value = 10
		}
		country_event = {
			id = heroship.102
			days = 60
		}
	}
	option = {
		name = heroship.101_opt
	}
}

# 先锋英雄舰折跃抵达（heroship.101 的延迟落地事件）
# 优先折跃至遗迹星系，有母舰时折跃至母舰附近
country_event = {
	id = heroship.102
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		random_owned_design = {
			limit = {
				is_ship_size = br_vanguard
			}
			save_event_target_as = br_reward_vanguard_design
		}
		# 有母舰时折跃至母舰附近（BR 模式优先）
		if = {
			limit = {
				any_owned_ship = {
					is_ship_size = br_dominator
				}
			}
			random_owned_ship = {
				limit = {
					is_ship_size = br_dominator
				}
				fleet = {
					save_event_target_as = br_vanguard_arrival_fleet
				}
			}
		}
		create_fleet = {
			name = "BR_VANGUARD_ARRIVAL"
			effect = {
				set_owner = root
				create_ship = {
					name = random
					design = event_target:br_reward_vanguard_design
					graphical_culture = root
				}
				if = {
					limit = {
						exists = event_target:br_vanguard_arrival_fleet
					}
					set_location = {
						target = event_target:br_vanguard_arrival_fleet
						distance = 35
						angle = random
					}
				}
				# 有遗迹时折跃至遗迹行星
				else_if = {
					limit = {
						exists = event_target:heroship_wreckage_planet@root
					}
					set_location = {
						target = event_target:heroship_wreckage_planet@root
						distance = 35
						angle = random
					}
				}
				# 保底：都没有就刷首都
				else = {
					set_location = {
						target = root.capital_scope
						distance = 35
						angle = random
					}
				}
			}
		}
	}
}
