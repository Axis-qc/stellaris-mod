namespace = reward

# ========================
# BUFF池 — reward.3001-3999
# 只能由 reward.1 总触发器调用。
# BUFF不分等级；效果 = 基础值 × br_power_level 叠加。
# 通用BUFF + 专精BUFF（后续实现）。
# ========================

# BUFF池入口
country_event = {
    id = reward.3001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        country_event = { id = reward.3101 }
    }
}

# BUFF池等级触发器
country_event = {
    id = reward.3101
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { check_variable = { which = br_power_level value < 6 } }
            country_event = { id = reward.3102 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 12 } }
            country_event = { id = reward.3103 }
        }
        else = {
            country_event = { id = reward.3104 }
        }
    }
}

# ========================
# 通用BUFF — 低等级 (Lv1-5)
# ========================
