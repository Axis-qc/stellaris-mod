namespace = reward

# ========================
# BUFF池 — reward.3001-3199
# 只能由 reward.1 总触发器调用。
# 每次从10个通用BUFF中随机选3个展示，玩家三选一。
# 扩展接口：其他系统可在3101之后、3102之前设置 br_buff_pick_4/5 变量以追加额外选项。
# ========================

# BUFF池入口
country_event = {
    id = reward.3001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        country_event = { id = reward.3101 }
    }
}

# 随机选3个BUFF选项（hide_window，不显示给玩家）
country_event = {
    id = reward.3101
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        # 清除上轮的pick变量
        set_variable = { which = br_buff_pick_1 value = 0 }
        set_variable = { which = br_buff_pick_2 value = 0 }
        set_variable = { which = br_buff_pick_3 value = 0 }
        set_variable = { which = br_buff_pick_4 value = 0 }

        # 第1轮：从10个BUFF中等权随机选1个
        random_list = {
            10 = { set_variable = { which = br_buff_pick_1 value = 1 } }
            10 = { set_variable = { which = br_buff_pick_1 value = 2 } }
            10 = { set_variable = { which = br_buff_pick_1 value = 3 } }
            10 = { set_variable = { which = br_buff_pick_1 value = 4 } }
            10 = { set_variable = { which = br_buff_pick_1 value = 5 } }
            10 = { set_variable = { which = br_buff_pick_1 value = 6 } }
            10 = { set_variable = { which = br_buff_pick_1 value = 7 } }
            10 = { set_variable = { which = br_buff_pick_1 value = 8 } }
            10 = { set_variable = { which = br_buff_pick_1 value = 9 } }
            10 = { set_variable = { which = br_buff_pick_1 value = 10 } }
        }

        # 第2轮：排除pick_1已选的值
        random_list = {
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 1 } } set_variable = { which = br_buff_pick_2 value = 1 } }
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 2 } } set_variable = { which = br_buff_pick_2 value = 2 } }
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 3 } } set_variable = { which = br_buff_pick_2 value = 3 } }
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 4 } } set_variable = { which = br_buff_pick_2 value = 4 } }
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 5 } } set_variable = { which = br_buff_pick_2 value = 5 } }
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 6 } } set_variable = { which = br_buff_pick_2 value = 6 } }
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 7 } } set_variable = { which = br_buff_pick_2 value = 7 } }
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 8 } } set_variable = { which = br_buff_pick_2 value = 8 } }
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 9 } } set_variable = { which = br_buff_pick_2 value = 9 } }
            10 = { modifier = { factor = 0 check_variable = { which = br_buff_pick_1 value = 10 } } set_variable = { which = br_buff_pick_2 value = 10 } }
        }

        # 第3轮：排除pick_1和pick_2已选的值
        random_list = {
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 1 } check_variable = { which = br_buff_pick_2 value = 1 } } } set_variable = { which = br_buff_pick_3 value = 1 } }
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 2 } check_variable = { which = br_buff_pick_2 value = 2 } } } set_variable = { which = br_buff_pick_3 value = 2 } }
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 3 } check_variable = { which = br_buff_pick_2 value = 3 } } } set_variable = { which = br_buff_pick_3 value = 3 } }
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 4 } check_variable = { which = br_buff_pick_2 value = 4 } } } set_variable = { which = br_buff_pick_3 value = 4 } }
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 5 } check_variable = { which = br_buff_pick_2 value = 5 } } } set_variable = { which = br_buff_pick_3 value = 5 } }
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 6 } check_variable = { which = br_buff_pick_2 value = 6 } } } set_variable = { which = br_buff_pick_3 value = 6 } }
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 7 } check_variable = { which = br_buff_pick_2 value = 7 } } } set_variable = { which = br_buff_pick_3 value = 7 } }
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 8 } check_variable = { which = br_buff_pick_2 value = 8 } } } set_variable = { which = br_buff_pick_3 value = 8 } }
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 9 } check_variable = { which = br_buff_pick_2 value = 9 } } } set_variable = { which = br_buff_pick_3 value = 9 } }
            10 = { modifier = { factor = 0 OR = { check_variable = { which = br_buff_pick_1 value = 10 } check_variable = { which = br_buff_pick_2 value = 10 } } } set_variable = { which = br_buff_pick_3 value = 10 } }
        }

        # 跳转统一展示事件
        country_event = { id = reward.3102 }
    }
}
