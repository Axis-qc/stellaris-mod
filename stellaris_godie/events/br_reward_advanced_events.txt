namespace = br

# 高级特殊奖励池，ID 6800-6899。
# 只由升级特殊池分流触发，用资源换舰船、强科技和强BUFF。
country_event = {
    id = br.6800
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        random_list = {
            34 = {
                if = {
                    limit = {
                        any_owned_fleet = { }
                        resource_stockpile_compare = { resource = alloys value >= 4500 }
                    }
                    country_event = { id = br.6810 }
                }
                else = {
                    country_event = { id = br.6140 }
                }
            }
            33 = {
                if = {
                    limit = {
                        OR = {
                            resource_stockpile_compare = { resource = physics_research value >= 8000 }
                            resource_stockpile_compare = { resource = engineering_research value >= 8000 }
                        }
                    }
                    country_event = { id = br.6820 }
                }
                else = {
                    country_event = { id = br.6160 }
                }
            }
            33 = {
                if = {
                    limit = {
                        resource_stockpile_compare = { resource = alloys value >= 3000 }
                        resource_stockpile_compare = { resource = volatile_motes value >= 100 }
                    }
                    country_event = { id = br.6830 }
                }
                else = {
                    country_event = { id = br.6150 }
                }
            }
        }
    }
}

country_event = {
    id = br.6810
    is_triggered_only = yes
    title = BR_REWARD_ADVANCED_SHIP_TITLE
    desc = BR_REWARD_ADVANCED_SHIP_DESC
    picture = GFX_evt_ship_in_orbit

    option = {
        name = BR_REWARD_ADVANCED_SHIP_ESCORT
        trigger = {
            any_owned_fleet = { }
            resource_stockpile_compare = { resource = alloys value >= 4500 }
            resource_stockpile_compare = { resource = energy value >= 2500 }
        }
        add_resource = { alloys = -4500 energy = -2500 }
        random_owned_fleet = { save_event_target_as = br_reward_spawn_fleet }
        create_fleet = {
            name = random
            effect = {
                set_owner = root

                    random_list = { 50 = { create_ship = { name = random design = "BR_PVE_ARTILLERY_DESTROYER" } } 50 = { create_ship = { name = random design = "BR_PVE_BRAWLER_DESTROYER" } } }
                    random_list = { 50 = { create_ship = { name = random design = "BR_PVE_ARTILLERY_DESTROYER" } } 50 = { create_ship = { name = random design = "BR_PVE_BRAWLER_DESTROYER" } } }
                    random_list = { 50 = { create_ship = { name = random design = "BR_PVE_ARTILLERY_CORVETTE" } } 50 = { create_ship = { name = random design = "BR_PVE_BRAWLER_CORVETTE" } } }

                set_location = { target = event_target:br_reward_spawn_fleet distance = 35 angle = random }
            }
        }
    }
    option = {
        name = BR_REWARD_ADVANCED_SHIP_CAPITAL
        trigger = {
            any_owned_fleet = { }
            check_variable = { which = br_power_level value >= 5 }
            resource_stockpile_compare = { resource = alloys value >= 9000 }
            resource_stockpile_compare = { resource = rare_crystals value >= 250 }
        }
        add_resource = { alloys = -9000 rare_crystals = -250 }
        random_owned_fleet = { save_event_target_as = br_reward_spawn_fleet }

            name = random
            effect = {
                set_owner = root

                    random_list = { 50 = { create_ship = { name = random design = "BR_PVE_ARTILLERY_ELITE_CRUISER" } } 50 = { create_ship = { name = random design = "BR_PVE_BRAWLER_ELITE_CRUISER" } } }

                set_location = { target = event_target:br_reward_spawn_fleet distance = 45 angle = random }
            }
        }
    }
    option = {
        name = BR_REWARD_ADVANCED_SHIP_DOMINATOR
        trigger = {
            any_owned_fleet = { }
            check_variable = { which = br_power_level value >= 7 }
            has_technology = tech_br_dominator
            resource_stockpile_compare = { resource = alloys value >= 18000 }
            resource_stockpile_compare = { resource = exotic_gases value >= 500 }
            resource_stockpile_compare = { resource = volatile_motes value >= 500 }
        }
        add_resource = { alloys = -18000 exotic_gases = -500 volatile_motes = -500 }
        random_owned_fleet = { save_event_target_as = br_reward_spawn_fleet }
        create_fleet = {
            name = "BR_MOTHERSHIP_FLEET"
            effect = {
                set_owner = root
                create_ship = { name = "NAME_BR_Dominator" design = "NAME_BR_Dominator" prefix = no }
                set_location = { target = event_target:br_reward_spawn_fleet distance = 55 angle = random }
            }
        }
    }
    option = { name = BR_REWARD_OPTION_ALLOY add_resource = { alloys = @reward_alloy_base mult = br_power_level } }
    option = { name = BR_REWARD_OPTION_ENERGY add_resource = { energy = @reward_energy_base mult = br_power_level } }
    option = { name = BR_REWARD_OPTION_MOTES add_resource = { volatile_motes = @reward_strategic_base mult = br_power_level } }
    option = { name = BR_REWARD_OPTION_CRYSTALS add_resource = { rare_crystals = @reward_strategic_base mult = br_power_level } }

}
