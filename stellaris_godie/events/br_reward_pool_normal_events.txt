# 普通奖励池。ID 6900-6949。
# 提供通用科技奖励，涵盖武器、护盾、装甲、A槽辅助、反应堆、推进器、作战电脑。
# 普通奖励最高比全局上限低一级（最高T4），流派专属T5武器不在此池。

# ========================
# 普通奖励池分流器
# ========================
country_event = {
    id = br.6900
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { check_variable = { which = br_power_level value < 2 } }
            country_event = { id = br.6901 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 3 } }
            country_event = { id = br.6902 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 4 } }
            country_event = { id = br.6903 }
        }
        else = {
            country_event = { id = br.6904 }
        }
    }
}
