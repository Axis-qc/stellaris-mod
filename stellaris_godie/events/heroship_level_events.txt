# 英雄舰杀敌经验（由 on_ship_destroyed_perp 直接调用，fromfromfrom = 被击杀舰船）
country_event = {
    id = heroship.2
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_country_type = br_player_pve
        from = {
            is_country_type = br_pve_enemy
        }
        exists = fromfromfrom
        any_owned_ship = {
            is_ship_size = br_vanguard
            solar_system = {
                is_same_value = fromfrom.solar_system
            }
        }
    }
    immediate = {
        #船体0.01转经验
        fromfromfrom = {
            export_trigger_value_to_variable = {
                # trigger = fleet_power #死亡读不到
                trigger = has_max_hp
                variable = br_tmp_kill_power
            }
        }
        set_variable = {
            which = br_hero_xp_add
            value = fromfromfrom.br_tmp_kill_power
        }
        divide_variable = {
            which = br_hero_xp_add
            value = 100
        }
        ceiling_variable = br_hero_xp_add
        change_variable = {
            which = br_hero_xp
            value = br_hero_xp_add
        }
        country_event = {
            id = heroship.3
        }
    }
}
