# 英雄等级经验阈值
@br_xp_level_2 = 10
@br_xp_level_3 = 40
@br_xp_level_4 = 110
@br_xp_level_5 = 240
@br_xp_level_6 = 460
@br_xp_level_7 = 810
@br_xp_level_8 = 1330
@br_xp_level_9 = 2080
@br_xp_level_10 = 3130
@br_xp_level_11 = 4630
@br_xp_level_12 = 6830
# 12 级后无限成长
@br_xp_next_12 = 5000

# 后备初始化（主初始化在 reward.6101 发放英雄舰时）
country_event = {
    id = heroship.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_country_type = br_player_pve
        NOT = { has_country_flag = br_hero_level_initialized }
    }

    immediate = {
        set_country_flag = br_hero_level_initialized
        set_variable = { which = br_hero_level value = 1 }
        set_variable = { which = br_hero_xp value = 0 }
        set_variable = { which = br_hero_xp_next value = 10 }
    }
}

# 英雄舰杀敌经验（由 on_ship_destroyed_perp 直接调用，fromfromfrom = 被击杀舰船）
country_event = {
    id = heroship.2
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_country_type = br_player_pve
        from = { is_country_type = br_pve_enemy }
        exists = fromfromfrom
        any_owned_ship = {
            is_ship_size = br_vanguard
            solar_system = {
                is_same_value = fromfrom.solar_system
            }
        }
    }

    immediate = {

        if = {
            limit = { fromfromfrom = { is_ship_size = br_pve_boss } }
            change_variable = { which = br_hero_xp value = 500 }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = battleship } }
            change_variable = { which = br_hero_xp value = 40 }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = cruiser } }
            change_variable = { which = br_hero_xp value = 20 }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = destroyer } }
            change_variable = { which = br_hero_xp value = 10 }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = corvette } }
            change_variable = { which = br_hero_xp value = 5 }
        }
        else = {
            change_variable = { which = br_hero_xp value = 3 }
        }

        country_event = { id = heroship.3 }
    }
}

# 等级检测 + 升级触发
country_event = {
    id = heroship.3
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        # 保存旧等级用于比较
        set_variable = { which = br_hero_level_prev value = br_hero_level }

        # 根据经验计算当前等级
        if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_12 } }
            set_variable = { which = br_hero_level value = 12 }
            set_variable = { which = br_hero_xp_next value = @br_xp_next_12 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_11 } }
            set_variable = { which = br_hero_level value = 11 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_12 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_10 } }
            set_variable = { which = br_hero_level value = 10 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_11 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_9 } }
            set_variable = { which = br_hero_level value = 9 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_10 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_8 } }
            set_variable = { which = br_hero_level value = 8 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_9 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_7 } }
            set_variable = { which = br_hero_level value = 7 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_8 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_6 } }
            set_variable = { which = br_hero_level value = 6 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_7 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_5 } }
            set_variable = { which = br_hero_level value = 5 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_6 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_4 } }
            set_variable = { which = br_hero_level value = 4 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_5 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_3 } }
            set_variable = { which = br_hero_level value = 3 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_4 }
        }
        else_if = {
            limit = { check_variable = { which = br_hero_xp value >= @br_xp_level_2 } }
            set_variable = { which = br_hero_level value = 2 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_3 }
        }
        else = {
            set_variable = { which = br_hero_level value = 1 }
            set_variable = { which = br_hero_xp_next value = @br_xp_level_2 }
        }

        # 12级后无限成长：每5000经验消耗并+1级
        if = {
            limit = { check_variable = { which = br_hero_level value >= 12 } }
            while = {
                limit = { check_variable = { which = br_hero_xp value >= @br_xp_next_12 } }
                subtract_variable = { which = br_hero_xp value = @br_xp_next_12 }
                change_variable = { which = br_hero_level value = 1 }
            }
        }

        # 升级触发：当前等级 > 旧等级时弹出选择窗口
        if = {
            limit = {
                check_variable = { which = br_hero_level value > br_hero_level_prev }
            }
            country_event = { id = heroship.4 }
        }
    }
}
