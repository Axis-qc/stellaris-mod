# ========================
# 资源池 — reward.1001-1199
# 只能由 reward.1 总触发器调用。
# 每次从14种资源中随机选4种展示，玩家四选一。
# 所有选项总价值相等：@reward_monthly_base × br_power_level 能量币/月。
# 单价（原版市场价）：能量/矿物/食物/贸易=1，消费品=2，合金=4，凝聚力=6，战略(8-10)=10，稀有(11-14)=20。
# 战略/稀有资源需对应科技才入池（br_res_lock_8~14），战力等级只影响数额。
# ========================

# 资源池入口
country_event = {
    id = reward.1001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        # 计算本次月产出增量（能量币当量总额）
        set_variable = { which = br_res_inc value = @reward_monthly_base }
        multiply_variable = { which = br_res_inc value = br_power_level }
        # ======================== 按价值折算各档金额（总额相等） ========================
        # 单价：能量/矿物/食物/贸易=1，消费品=2，合金=4，凝聚力=6，战略=10，稀有=20
        # 金额 = 总价值 ÷ 单价
        set_variable = { which = br_res_amt_cg value = br_res_inc }
        divide_variable = { which = br_res_amt_cg value = 2 }
        set_variable = { which = br_res_amt_alloy value = br_res_inc }
        divide_variable = { which = br_res_amt_alloy value = 4 }
        set_variable = { which = br_res_amt_unity value = br_res_inc }
        divide_variable = { which = br_res_amt_unity value = 6 }
        set_variable = { which = br_res_amt_sr value = br_res_inc }
        divide_variable = { which = br_res_amt_sr value = 10 }
        set_variable = { which = br_res_amt_rare value = br_res_inc }
        divide_variable = { which = br_res_amt_rare value = 20 }
        # ======================== 战略/稀有资源科技解锁（有对应科技才入池） ========================
        set_variable = { which = br_res_lock_8 value = 0 }
        if = {
            limit = { OR = { has_technology = tech_volatile_motes has_technology = tech_mine_volatile_motes } }
            set_variable = { which = br_res_lock_8 value = 1 }
        }
        set_variable = { which = br_res_lock_9 value = 0 }
        if = {
            limit = { OR = { has_technology = tech_rare_crystals has_technology = tech_mine_rare_crystals } }
            set_variable = { which = br_res_lock_9 value = 1 }
        }
        set_variable = { which = br_res_lock_10 value = 0 }
        if = {
            limit = { OR = { has_technology = tech_exotic_gases has_technology = tech_mine_exotic_gases } }
            set_variable = { which = br_res_lock_10 value = 1 }
        }
        set_variable = { which = br_res_lock_11 value = 0 }
        if = {
            limit = { has_technology = tech_mine_dark_matter }
            set_variable = { which = br_res_lock_11 value = 1 }
        }
        set_variable = { which = br_res_lock_12 value = 0 }
        if = {
            limit = { has_technology = tech_mine_living_metal }
            set_variable = { which = br_res_lock_12 value = 1 }
        }
        set_variable = { which = br_res_lock_13 value = 0 }
        if = {
            limit = { has_technology = tech_mine_zro }
            set_variable = { which = br_res_lock_13 value = 1 }
        }
        set_variable = { which = br_res_lock_14 value = 0 }
        if = {
            limit = { OR = { has_technology = tech_nanite_transmutation has_technology = tech_nanite_assemblers } }
            set_variable = { which = br_res_lock_14 value = 1 }
        }
        country_event = { id = reward.1101 }
    }
}
