# ========================
# 资源池 — reward.1001-1199
# 只能由 reward.1 总触发器调用。
# 每次从14种资源中随机选4种展示，玩家四选一。
# 战略资源（8-10）需 br_power_level >= 4 才入池。
# 稀有资源（11-14）需 br_power_level >= 8 才入池，基础值 0.2×等级。
# 增量 = @reward_monthly_base × br_power_level
# ========================

# 资源池入口
country_event = {
    id = reward.1001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        # 计算本次月产出增量
        set_variable = { which = br_res_inc value = @reward_monthly_base }
        multiply_variable = { which = br_res_inc value = br_power_level }
        # 预计算战略资源是否解锁（≥4级）
        if = {
            limit = { check_variable = { which = br_power_level value >= 4 } }
            set_variable = { which = br_res_strategic_ok value = 1 }
            else = { set_variable = { which = br_res_strategic_ok value = 0 } }
        }
        # 预计算稀有资源是否解锁（≥8级）
        if = {
            limit = { check_variable = { which = br_power_level value >= 8 } }
            set_variable = { which = br_res_rare_ok value = 1 }
            else = { set_variable = { which = br_res_rare_ok value = 0 } }
        }
        # 稀有资源基础值 0.2×等级（br_res_inc 的 0.1 倍）
        set_variable = { which = br_res_inc_rare value = br_res_inc }
        multiply_variable = { which = br_res_inc_rare value = 0.1 }
        country_event = { id = reward.1101 }
    }
}
