namespace = reward

# ========================
# 资源池 — reward.1001-1099
# 只能由 reward.1 总触发器调用。
# 统一基础值×战力等级，不再分级。
# ========================

# 资源池入口
country_event = {
    id = reward.1001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        set_variable = { which = br_resource_alloy value = @reward_alloy_base }
        set_variable = { which = br_resource_energy value = @reward_energy_base }
        set_variable = { which = br_resource_strategic value = @reward_strategic_base }
        multiply_variable = { which = br_resource_alloy value = br_power_level }
        multiply_variable = { which = br_resource_energy value = br_power_level }
        multiply_variable = { which = br_resource_strategic value = br_power_level }
        country_event = { id = reward.1101 }
    }
}

# 资源补给 — 五选一
country_event = {
    id = reward.1101
    is_triggered_only = yes
    title = REWARD_POOL_RESOURCE_TITLE
    desc = REWARD_POOL_RESOURCE_DESC
    picture = GFX_evt_mining_station

    option = { name = REWARD_OPTION_ALLOY    custom_tooltip = REWARD_OPTION_ALLOY_TT    add_resource = { alloys = @reward_alloy_base mult = br_power_level } }
    option = { name = REWARD_OPTION_ENERGY   custom_tooltip = REWARD_OPTION_ENERGY_TT   add_resource = { energy = @reward_energy_base mult = br_power_level } }
    option = { name = REWARD_OPTION_MOTES    custom_tooltip = REWARD_OPTION_MOTES_TT    add_resource = { volatile_motes = @reward_strategic_base mult = br_power_level } }
    option = { name = REWARD_OPTION_CRYSTALS custom_tooltip = REWARD_OPTION_CRYSTALS_TT add_resource = { rare_crystals = @reward_strategic_base mult = br_power_level } }
    option = { name = REWARD_OPTION_GASES    custom_tooltip = REWARD_OPTION_GASES_TT    add_resource = { exotic_gases = @reward_strategic_base mult = br_power_level } }
}
