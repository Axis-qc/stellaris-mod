# ========================
# 科技池 — reward.2001-2999
# 只能由 reward.1 总触发器调用。
# 五级制分流，每级拆为武器/防护/辅助(A槽)三个子事件。
# 具体奖励只做显示检测：已拥有科技不显示。
# TODO: 科技池可能被抽空，后续在入口或总触发器增加可用科技计数/重抽/转池兜底。
# ========================

# 科技池入口
country_event = {
    id = reward.2001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        country_event = { id = reward.2101 }
    }
}

# 科技池等级触发器
country_event = {
    id = reward.2101
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { check_variable = { which = br_power_level value < 3 } }
            random_list = {
                34 = { country_event = { id = reward.2111 } }   # T1 武器
                33 = { country_event = { id = reward.2121 } }   # T1 防护
                33 = { country_event = { id = reward.2131 } }   # T1 辅助
            }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 6 } }
            random_list = {
                34 = { country_event = { id = reward.2211 } }   # T2 武器
                33 = { country_event = { id = reward.2221 } }   # T2 防护
                33 = { country_event = { id = reward.2231 } }   # T2 辅助
            }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 9 } }
            random_list = {
                34 = { country_event = { id = reward.2311 } }   # T3 武器
                33 = { country_event = { id = reward.2321 } }   # T3 防护
                33 = { country_event = { id = reward.2331 } }   # T3 辅助
            }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 12 } }
            random_list = {
                34 = { country_event = { id = reward.2411 } }   # T4 武器
                33 = { country_event = { id = reward.2421 } }   # T4 防护
                33 = { country_event = { id = reward.2431 } }   # T4 辅助
            }
        }
        else = {
            random_list = {
                34 = { country_event = { id = reward.2511 } }   # T5 武器
                33 = { country_event = { id = reward.2521 } }   # T5 防护
                33 = { country_event = { id = reward.2531 } }   # T5 辅助
            }
        }
    }
}
