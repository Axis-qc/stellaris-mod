# ============================================================
# 星系探索奖励系统
# ID 区间：reward.7001-7099
# 玩家进入新星系时，在周边星系刷出探索奖励
# 流程：reward.7001 → reward.7002 → reward.7010/7020 → special project → reward.7011/7021
# ============================================================
# ============================================================
# reward.7001：入口（fleet_event）
# 由 on_entering_system_fleet 触发，与 br.804 刷敌并行
# 标记当前星系 → 找周边未标记星系 → 转 reward.7002 做随机
# ============================================================
fleet_event = {
    id = reward.7001
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = br_mode_pve
        owner = {
            is_ai = no
        }
        owner = {
            has_country_flag = br_player_country
        }
        owner = {
            NOT = {
                has_country_flag = br_exploration_cooldown
            }
        }
    }
    immediate = {
        # 标记当前星系为已探索，防止重复触发
        this = {
            solar_system = {
                if = {
                    limit = {
                        NOT = {
                            has_star_flag = br_explored
                        }
                    }
                    set_star_flag = br_explored
                }
            }
        }
        # 保存当前星系引用，找周边未标记星系
        this = {
            solar_system = {
                save_global_event_target_as = br_exploration_source@owner
                random_neighbor_system = {
                    limit = {
                        NOT = {
                            has_star_flag = br_explored
                        }
                        NOT = {
                            has_star_flag = br_player_spawn_system
                        }
                        NOT = {
                            any_ship_in_system = {
                                is_owned_by = root.owner
                            }
                        }
                    }
                    set_star_flag = br_exploration_marked
                    save_global_event_target_as = br_exploration_target@owner
                }
            }
        }
        # 转给 country_event 做随机（避免 fleet_event 深层作用域限制）
        owner = {
            country_event = {
                id = reward.7002
            }
        }
        # 设置2年冷却（730天=2年）
        owner = {
            set_timed_country_flag = {
                flag = br_exploration_cooldown
                days = 730
            }
        }
    }
}

# ============================================================
# reward.7002：随机路由器（country_event）
# 判断目标星系是否存在 → 按权重分配探索事件
# ============================================================
country_event = {
    id = reward.7002
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = br_mode_pve
    }
    immediate = {
        if = {
            limit = {
                exists = event_target:br_exploration_target@root
            }
            random_list = {
                # 测试阶段：100%出星门，上线前改回 50/40/10
                60 = { }
                30 = { country_event = { id = reward.7010 } }
                10 = {
                    country_event = {
                        # id = reward.7020
                    }
                }
                # 100 = {
                #     country_event = {
                #         id = reward.7010
                #     }
                # }
            }
        }
    }
}

# reward.7010：在目标星系刷出废弃星门（视觉标记），行星上激活调查特殊项目
# ============================================================
country_event = {
    id = reward.7010
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        has_global_flag = br_mode_pve
    }
    immediate = {
        event_target:br_exploration_target@root = {
            random_system_planet = {
                limit = {
                    br_is_valid_exploration_planet = yes
                }
                save_event_target_as = br_exploration_planet
            }
            create_ambient_object = {
                type = br_abandoned_gate
                location = event_target:br_exploration_planet
                scale = 1
                entity_offset = {
                    min = 1
                    max = 10
                }
                entity_offset_angle = {
                    min = 0
                    max = 10
                }
                effect = {
                    save_global_event_target_as = br_exploration_ao
                    enable_special_project = {
                        name = "br_exploration_wreck"
                        location = event_target:br_exploration_planet
                        owner = root
                    }
                }
            }
        }
    }
}
