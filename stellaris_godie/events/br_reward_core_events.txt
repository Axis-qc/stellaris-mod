# 二级流派选择后，按当前玩家帝国已有的先锋级设计直接发放一艘。
country_event = {
    id = br.6121
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { any_owned_fleet = { } }
            random_owned_design = {
                limit = { is_ship_size = br_vanguard }
                save_event_target_as = br_reward_vanguard_design
            }
            if = {
                limit = { exists = event_target:br_reward_vanguard_design }
                random_owned_fleet = { save_event_target_as = br_reward_vanguard_spawn_fleet }
                create_fleet = {
                    name = random
                    effect = {
                        set_owner = root
                        create_ship = {
                            name = random
                            design = event_target:br_reward_vanguard_design
                            graphical_culture = root
                        }
                        set_location = { target = event_target:br_reward_vanguard_spawn_fleet distance = 35 angle = random }
                    }
                }
            }
        }
    }
}
