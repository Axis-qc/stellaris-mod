# ── lev.3100 战斗成长 ──
# 由 on_ship_destroyed_perp 触发，逻辑与英雄舰 heroship.2 同构：
# 同星系内有利维坦舰船、且击杀的是敌对目标时，成长 += 被击杀舰船 max_hp ÷ 100（向上取整）
# 多只利维坦各自独立：谁在击杀星系内就给谁入账。
country_event = {
    id = lev.3100
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = no
        from = {
            is_hostile = root
        }
        exists = fromfromfrom
        any_owned_ship = {
            lev_is_leviathan_ship = yes
            solar_system = {
                is_same_value = fromfrom.solar_system
            }
        }
    }
    immediate = {
        # 船体 0.01 转成长
        fromfromfrom = {
            export_trigger_value_to_variable = {
                # trigger = fleet_power #死亡读不到
                trigger = has_max_hp
                variable = lev_tmp_kill_power
            }
        }
        set_variable = {
            which = lev_growth_add
            value = fromfromfrom.lev_tmp_kill_power
        }
        divide_variable = {
            which = lev_growth_add
            value = 100
        }
        ceiling_variable = lev_growth_add

        # ── 逐只利维坦判定：哪只在击杀星系内，就给哪只入账 ──
        # 新增一只利维坦时，在此照抄一个 if 并改 is_ship_size 与 TYPE
        if = {
            limit = {
                any_owned_ship = {
                    is_ship_size = lev_stellarite
                    solar_system = {
                        is_same_value = fromfrom.solar_system
                    }
                }
            }
            lev_growth_apply_combat = { TYPE = stellarite }
        }
        if = {
            limit = {
                any_owned_ship = {
                    is_ship_size = lev_drake
                    solar_system = {
                        is_same_value = fromfrom.solar_system
                    }
                }
            }
            lev_growth_apply_combat = { TYPE = drake }
        }
        if = {
            limit = {
                any_owned_ship = {
                    is_ship_size = lev_shard
                    solar_system = {
                        is_same_value = fromfrom.solar_system
                    }
                }
            }
            lev_growth_apply_combat = { TYPE = shard }
        }
        if = {
            limit = {
                any_owned_ship = {
                    is_ship_size = lev_wraith
                    solar_system = {

                    }

            }

        if = {

# ── lev.3101 营养月度结算 ──
# 由 on_monthly_pulse_country 触发。遍历所有已解锁利维坦，各按自己的档位扣资源换成长；
# 某只资源不足则它自己当月中止（不涨成长、不降档、不提示），不影响其它利维坦。
country_event = {
    id = lev.3101
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        is_ai = no
        OR = {
            has_country_flag = lev_stellarite_unlocked
            has_country_flag = lev_drake_unlocked
            has_country_flag = lev_shard_unlocked
            has_country_flag = lev_wraith_unlocked
            has_country_flag = lev_horror_unlocked
            has_country_flag = lev_toxic_god_unlocked
            has_country_flag = lev_dreadnought_unlocked
            has_country_flag = lev_scavenger_unlocked
            has_country_flag = lev_sphere_unlocked
            has_country_flag = lev_aquatic_unlocked
            has_country_flag = lev_voidspawn_unlocked
            has_country_flag = lev_fortress_unlocked
            has_country_flag = lev_hive_unlocked
            has_country_flag = lev_tiyanki_unlocked
        }
    }
    immediate = {
        lev_feed_settle_all = yes
    }
}
