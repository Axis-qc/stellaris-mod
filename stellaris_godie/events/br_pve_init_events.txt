# br.701：移动已有方舟舰和护卫舰队到出生点，设置房主，排程后续
country_event = {
	id = br.701
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_ai = no
		has_country_flag = br_player_replacement_scheduled
		NOT = {
			has_country_flag = br_player_replaced
		}
		has_origin = origin_br_battle_royale
	}
	immediate = {
		set_country_flag = br_player_replaced
		# 随机选择一个出生点
		random_system = {
			limit = {
				has_star_flag = br_player_spawn_system
			}
			save_event_target_as = br_chosen_spawn_system
			star = {
				save_event_target_as = br_chosen_spawn_star
			}
		}
		# 移动所有舰队到出生点
		every_owned_fleet = {
			set_location = {
				target = event_target:br_chosen_spawn_star
				distance = 50
				angle = random
			}
		}
		# 设置房主（仅首位玩家）
		if = {
			limit = {
				NOT = {
					has_global_flag = br_pve_alliance_leader_set
				}
			}
			set_global_flag = br_pve_alliance_leader_set
			save_global_event_target_as = br_pve_alliance_leader
		}
		#发遗珍
		add_relic = br_reward_relics
		# 排程PVE后续初始化
		country_event = {
			id = br.507
			days = 1
		}
	}
}

# br.702：PVE模式初始化，建立联盟与创建敌对势力
country_event = {
	id = br.702
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_country_flag = br_player_country
	}
	immediate = {
		country_event = {
			id = br.1100
		}
		every_country = {
			limit = {
				NOT = {
					is_same_value = root
				}
				has_country_flag = br_player_country
			}
			establish_communications_no_message = root
			add_trust = {
				amount = 100
				who = root
			}
			add_intel = {
				amount = 100
				who = root
			}
			root = {
				add_trust = {
					amount = 100
					who = prev
				}
				add_intel = {
					amount = 100
					who = prev
				}
			}
		}
		if = {
			limit = {
				exists = event_target:br_pve_alliance_leader
				NOT = {
					is_same_value = event_target:br_pve_alliance_leader
				}
			}
			join_alliance = {
				who = event_target:br_pve_alliance_leader
				name = BR_PVE_FEDERATION_NAME
				override_requirements = yes
			}
		}
		if = {
			limit = {
				has_federation = yes
			}
			federation = {
				add_cohesion = 1000
				add_federation_experience = 600
			}
		}
		# 虚空工具国加入联邦，使其舰队与玩家协同作战
		event_target:br_void_country = {
			join_alliance = {
				who = event_target:br_pve_alliance_leader
				name = BR_PVE_FEDERATION_NAME
				override_requirements = yes
			}
		}
		# PVE模式：创建可见但不可外交的天灾式敌对势力
		create_country = {
			name = "BR_PVE_ENEMY_NAME"
			type = br_pve_enemy
			flag = {
				icon = {
					category = "ornate"
					file = "flag_ornate_8.dds"
				}
				background = {
					category = "backgrounds"
					file = "00_solid.dds"
				}
				colors = {
					"red"
					"black"
					"null"
					"null"
				}
			}
			effect = {
				save_global_event_target_as = br_pve_enemy_country
				set_global_flag = br_pve_enemy_created
				set_country_flag = br_pve_enemy
				set_country_flag = br_pve_crisis_country
				set_graphical_culture = mammalian_01
			}
		}
		every_country = {
			limit = {
				has_country_flag = br_player_country
			}
			establish_communications_no_message = event_target:br_pve_enemy_country
			start_situation = { type = situation_br_pve_intensity }
		}
		set_global_flag = br_pve_enemy_created
		country_event = {
			id = br.1
		}
		country_event = {
			id = br.3
		}
		# 正常初始化：统一变量初始化
		country_event = {
			id = br.799
		}
		# [后备保障] 敌人已存在时不生效
	}
}
