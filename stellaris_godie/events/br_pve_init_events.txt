# br.702：PVE模式初始化，建立联盟与创建敌对势力喵
country_event = {
    id = br.702
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { is_country_type = br_player_pve }
            country_event = { id = br.1100 }

            every_country = {
                limit = {
                    NOT = { is_same_value = root }
                    is_country_type = br_player_pve
                }
                establish_communications_no_message = root
                add_trust = {
                    amount = 100
                    who = root
                }
                add_intel = {
                    amount = 100
                    who = root
                }
                root = {
                    add_trust = {
                        amount = 100
                        who = prev
                    }
                    add_intel = {
                        amount = 100
                        who = prev
                    }
                }
            }

            if = {
                limit = {
                    exists = event_target:br_pve_alliance_leader
                    NOT = { is_same_value = event_target:br_pve_alliance_leader }
                }
                join_alliance = {
                    who = event_target:br_pve_alliance_leader
                    name = BR_PVE_FEDERATION_NAME
                    override_requirements = yes
                }
            }

            if = {
                limit = { has_federation = yes }
                federation = {
                    add_cohesion = 1000
                    add_federation_experience = 600
                }
            }

            # PVE模式：创建可见但不可外交的天灾式敌对势力
            if = {
                limit = {
                    has_global_flag = br_mode_pve
                    NOT = { exists = event_target:br_pve_enemy_country }

                create_country = {
                    name = "BR_PVE_ENEMY_NAME"
                    type = br_pve_enemy
                    flag = {
                        icon = {
                            category = "ornate"
                            file = "flag_ornate_8.dds"
                        }
                        background = {
                            category = "backgrounds"
                            file = "00_solid.dds"
                        }

                            "red"
                            "black"

                    effect = {

                    }
                }

                limit = { exists = event_target:br_pve_enemy_country }
