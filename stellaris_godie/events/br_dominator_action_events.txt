# Dominator 战地再生力场（由 on_br_dominator_repair_activated 触发）
# 消耗充能层数，为母舰所在星系内所有友军舰队施加再生力场。
# 持续天数 = 30 × 充能层数。
fleet_event = {
    id = br.1200
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_dominator
        }
    }

    immediate = {
        owner = {
            save_event_target_as = br_dominator_regen_owner
            set_variable = { which = br_dominator_regen_charge_day value = 30 }
            multiply_variable = { which = br_dominator_regen_charge_day value = br_dominator_regen_charge }
        }
        solar_system = {
            every_fleet_in_system = {
                limit = {
                    exists = controller
                    controller = {
                        NOT = { is_hostile = event_target:br_dominator_regen_owner }
                    }
                }
                add_modifier = {
                    modifier = br_dominator_regen_field
                    days = 1
                    time_multiplier = owner.br_dominator_regen_charge_day
                }
            }
        }

        owner = {
            set_variable = { which = br_dominator_regen_charge value = 0 }
            set_variable = { which = br_dominator_regen_charge_day value = 0 }
            country_event = { id = br.1205 days = 60 }
        }
    }
}

# Dominator 全体召回（由 on_br_dominator_recall_activated 触发）
# 将所有玩家舰队直接传送至母舰位置。
fleet_event = {
    id = br.1201
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_dominator
        }
    }

    immediate = {
        save_event_target_as = br_mothership_recall_target

        owner = {
            every_owned_fleet = {
                limit = {
                    NOT = { is_same_value = event_target:br_mothership_recall_target }
                }
                set_location = event_target:br_mothership_recall_target
            }
        }
    }
}

# Dominator 再生力场充能（由 br.1200 启动，每 60 天 +1 层，满 3 层停止自触发）
country_event = {
    id = br.1205
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_ai = no
        check_variable = { which = br_dominator_regen_charge value < 3 }
    }

    immediate = {
        change_variable = { which = br_dominator_regen_charge value = 1 }
        set_variable = { which = br_dominator_regen_charge_day value = 30 }
        multiply_variable = { which = br_dominator_regen_charge_day value = br_dominator_regen_charge }
        if = {
            limit = {
                check_variable = { which = br_dominator_regen_charge value < 3 }
            }
            country_event = { id = br.1205 days = 60 }
        }
    }
}
