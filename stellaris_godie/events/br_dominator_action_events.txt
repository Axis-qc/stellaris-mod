# Dominator 一次性维修（由 on_br_dominator_repair_activated 触发）
fleet_event = {
    id = br.1200
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_dominator
        }
    }

    immediate = {
        owner = {
            save_event_target_as = br_dominator_mode_owner
        }

        solar_system = {
            every_fleet_in_system = {
                limit = {
                    exists = controller
                    controller = {
                        NOT = { is_hostile = event_target:br_dominator_mode_owner }
                    }
                }
                repair_percentage = 0.10
                repair_armor_percentage = 0.10
            }
        }
    }
}

# Dominator 模式切换光环（由 on_br_dominator_mode_toggle_activated 触发）
fleet_event = {
    id = br.1201
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_dominator
        }
    }

    immediate = {
        owner = {
            save_event_target_as = br_dominator_mode_owner
        }

        if = {
            limit = { has_fleet_flag = br_dominator_mode_targeting }

            remove_fleet_flag = br_dominator_mode_targeting
            remove_fleet_flag = br_dominator_mode_repair_aura
            set_fleet_flag = br_dominator_mode_repair_aura

            solar_system = {
                every_fleet_in_system = {
                    limit = {
                        exists = controller
                        controller = {
                            NOT = { is_hostile = event_target:br_dominator_mode_owner }
                        }
                    }
                    remove_modifier = br_dominator_mode_targeting
                    remove_modifier = br_dominator_mode_repair_aura
                    add_modifier = {
                        modifier = br_dominator_mode_repair_aura
                        days = 30
                    }
                }
            }
        }

        else = {
            remove_fleet_flag = br_dominator_mode_targeting
            remove_fleet_flag = br_dominator_mode_repair_aura
            set_fleet_flag = br_dominator_mode_targeting

            solar_system = {
                every_fleet_in_system = {
                    limit = {
                        exists = controller
                        controller = {
                            NOT = { is_hostile = event_target:br_dominator_mode_owner }
                        }
                    }
                    remove_modifier = br_dominator_mode_targeting
                    remove_modifier = br_dominator_mode_repair_aura
                    add_modifier = {
                        modifier = br_dominator_mode_targeting
                        days = 30
                    }
                }
            }
        }
    }
}

# 周期刷新母舰模式的临时星系范围 modifier。
country_event = {
    id = br.1204
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_fleet = {
            any_owned_ship = {
                is_ship_size = br_dominator
            }
            OR = {
                has_fleet_flag = br_dominator_mode_targeting
                has_fleet_flag = br_dominator_mode_repair_aura
            }
        }
    }

    immediate = {
        every_owned_fleet = {
            limit = {
                any_owned_ship = {
                    is_ship_size = br_dominator
                }
                OR = {
                    has_fleet_flag = br_dominator_mode_targeting
                    has_fleet_flag = br_dominator_mode_repair_aura
                }
            }

            if = {
                limit = { has_fleet_flag = br_dominator_mode_targeting }
                solar_system = {
                    every_fleet_in_system = {
                        limit = {
                            exists = controller
                            controller = {
                                NOT = { is_hostile = root }
                            }
                        }
                        remove_modifier = br_dominator_mode_targeting
                        remove_modifier = br_dominator_mode_repair_aura
                        add_modifier = {
                            modifier = br_dominator_mode_targeting
                            days = 30
                        }
                    }
                }
            }

            if = {
                limit = { has_fleet_flag = br_dominator_mode_repair_aura }
                solar_system = {
                    every_fleet_in_system = {
                        limit = {
                            exists = controller
                            controller = {
                                NOT = { is_hostile = root }
                            }
                        }
                        remove_modifier = br_dominator_mode_targeting
                        remove_modifier = br_dominator_mode_repair_aura
                        add_modifier = {
                            modifier = br_dominator_mode_repair_aura
                            days = 30
                        }
                    }
                }
            }
        }
    }
}
