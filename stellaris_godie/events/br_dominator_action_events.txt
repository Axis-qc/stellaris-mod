namespace = br

# 由 on_br_dominator_repair_activated 触发。
fleet_event = {
    id = br.1200
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_dominator
        }
    }

    immediate = {
        owner = {
            save_event_target_as = br_dominator_mode_owner
        }

        solar_system = {
            every_fleet_in_system = {
                limit = {
                    exists = controller
                    controller = {
                        NOT = { is_hostile = event_target:br_dominator_mode_owner }
                    }
                }
                repair_percentage = 0.25
                repair_armor_percentage = 0.25
                repair_shield_percentage = 0.25
            }
        }
    }
}

# 能量超频激活
fleet_event = {
    id = br.1214
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_vanguard
            has_component = BR_VANGUARD_CORE_POWER
        }
        owner = { has_country_flag = br_skill_energy_overclock_unlocked }
        NOT = { has_fleet_flag = br_overclock_active }
    }

    immediate = {
        set_fleet_flag = br_overclock_active
        set_variable = { which = br_overclock_timer value = 10 }

        every_owned_ship = {
            limit = {
                is_ship_size = br_vanguard
                has_component = BR_VANGUARD_CORE_POWER
            }
            add_modifier = {
                modifier = br_vanguard_power_energy_overclock
                days = 11
            }
        }

        fleet_event = { id = br.1215 days = 1 }
    }
}

# 能量超频每日 tick
fleet_event = {
    id = br.1215
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = {
                check_variable = { which = br_overclock_timer value > 0 }
            }
            subtract_variable = { which = br_overclock_timer value = 1 }

            every_owned_ship = {
                limit = {
                    is_ship_size = br_vanguard
                    has_component = BR_VANGUARD_CORE_POWER
                }
                remove_modifier = br_vanguard_power_energy_overclock
                add_modifier = {
                    modifier = br_vanguard_power_energy_overclock
                    days = 2
                }
            }

            fleet_event = { id = br.1215 days = 1 }
        }
        else = {
            remove_fleet_flag = br_overclock_active
            every_owned_ship = {
                limit = {
                    is_ship_size = br_vanguard
                    has_component = BR_VANGUARD_CORE_POWER
                }
                remove_modifier = br_vanguard_power_energy_overclock
            }
        }
    }
}

# 越战越勇被动：每击杀 +1% 武器伤害，上限 50%
country_event = {
    id = br.1213
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_country_flag = br_skill_kill_stack_unlocked
        check_variable = { which = br_vanguard_kill_count value < 50 }
    }

    immediate = {
        change_variable = { which = br_vanguard_kill_count value = 1 }

        remove_modifier = br_vanguard_power_kill_stack_damage
        if = {
            limit = {
                check_variable = { which = br_vanguard_kill_count value >= 50 }
            }
            set_variable = { which = br_vanguard_kill_count value = 50 }
            add_modifier = {
                modifier = br_vanguard_power_kill_stack_damage
                days = -1
                mult = 50
            }
        }
        else = {
            add_modifier = {
                modifier = br_vanguard_power_kill_stack_damage
                days = -1
                mult = br_vanguard_kill_count
            }
        }
    }
}

# 力量核心原型：接战后每日叠加伤害，脱战清空。
fleet_event = {
    id = br.1211
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_vanguard
            has_component = BR_VANGUARD_CORE_POWER
        }
        owner = { has_country_flag = br_skill_combat_stack_unlocked }
    }

    immediate = {
        if = {
            limit = {
                NOT = { has_fleet_flag = br_vanguard_power_combat_stack_active }
            }
            set_fleet_flag = br_vanguard_power_combat_stack_active
            clear_variable = br_vanguard_power_combat_stack
            fleet_event = { id = br.1212 days = 1 }
        }
    }
}

fleet_event = {
    id = br.1212
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = {
                is_in_combat = yes
                any_owned_ship = {
                    is_ship_size = br_vanguard
                    has_component = BR_VANGUARD_CORE_POWER
                }
            }

            if = {
                limit = {
                    NOT = { is_variable_set = br_vanguard_power_combat_stack }
                }
                set_variable = { which = br_vanguard_power_combat_stack value = 0 }
            }

            if = {
                limit = {
                    check_variable = { which = br_vanguard_power_combat_stack value < 100 }
                }
                change_variable = { which = br_vanguard_power_combat_stack value = 1 }
            }

            remove_modifier = br_vanguard_power_combat_stack_damage
            add_modifier = {
                modifier = br_vanguard_power_combat_stack_damage
                days = 2
                mult = br_vanguard_power_combat_stack
            }

            fleet_event = { id = br.1212 days = 1 }
        }

        else = {
            remove_modifier = br_vanguard_power_combat_stack_damage
            clear_variable = br_vanguard_power_combat_stack
            remove_fleet_flag = br_vanguard_power_combat_stack_active
        }
    }
}

# 由 on_br_dominator_mode_toggle_activated 触发。
fleet_event = {
    id = br.1201
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_dominator
        }
    }

    immediate = {
        owner = {
            save_event_target_as = br_dominator_mode_owner
        }

        if = {
            limit = { has_fleet_flag = br_dominator_mode_targeting }

            remove_fleet_flag = br_dominator_mode_targeting
            remove_fleet_flag = br_dominator_mode_repair_aura
            set_fleet_flag = br_dominator_mode_repair_aura

            solar_system = {
                every_fleet_in_system = {
                    limit = {
                        exists = controller
                        controller = {
                            NOT = { is_hostile = event_target:br_dominator_mode_owner }
                        }
                    }
                    remove_modifier = br_dominator_mode_targeting
                    remove_modifier = br_dominator_mode_repair_aura
                    add_modifier = {
                        modifier = br_dominator_mode_repair_aura
                        days = 30
                    }
                }
            }
        }

        else = {
            remove_fleet_flag = br_dominator_mode_targeting
            remove_fleet_flag = br_dominator_mode_repair_aura
            set_fleet_flag = br_dominator_mode_targeting

            solar_system = {
                every_fleet_in_system = {
                    limit = {
                        exists = controller
                        controller = {
                            NOT = { is_hostile = event_target:br_dominator_mode_owner }
                        }
                    }
                    remove_modifier = br_dominator_mode_targeting
                    remove_modifier = br_dominator_mode_repair_aura
                    add_modifier = {
                        modifier = br_dominator_mode_targeting
                        days = 30
                    }
                }
            }
        }
    }
}

# 周期刷新母舰模式的临时星系范围 modifier。
country_event = {
    id = br.1204
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_fleet = {
            any_owned_ship = {
                is_ship_size = br_dominator
            }
            OR = {
                has_fleet_flag = br_dominator_mode_targeting
                has_fleet_flag = br_dominator_mode_repair_aura
            }
        }
    }

    immediate = {
        every_owned_fleet = {
            limit = {
                any_owned_ship = {
                    is_ship_size = br_dominator
                }
                OR = {
                    has_fleet_flag = br_dominator_mode_targeting
                    has_fleet_flag = br_dominator_mode_repair_aura
                }
            }

            if = {
                limit = { has_fleet_flag = br_dominator_mode_targeting }
                solar_system = {
                    every_fleet_in_system = {
                        limit = {
                            exists = controller
                            controller = {
                                NOT = { is_hostile = root }
                            }
                        }
                        remove_modifier = br_dominator_mode_targeting
                        remove_modifier = br_dominator_mode_repair_aura
                        add_modifier = {
                            modifier = br_dominator_mode_targeting
                            days = 30
                        }
                    }
                }
            }

            if = {
                limit = { has_fleet_flag = br_dominator_mode_repair_aura }
                solar_system = {
                    every_fleet_in_system = {
                        limit = {
                            exists = controller
                            controller = {
                                NOT = { is_hostile = root }
                            }
                        }
                        remove_modifier = br_dominator_mode_targeting
                        remove_modifier = br_dominator_mode_repair_aura
                        add_modifier = {
                            modifier = br_dominator_mode_repair_aura
                            days = 30
                        }
                    }
                }
            }
        }
    }
}
