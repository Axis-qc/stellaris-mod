namespace = heroship

# ========================
# 火力核心技能 — heroship.1100-1199
# ========================

# 能量超频激活（由 on_br_hero_active_skill_activated 触发）
fleet_event = {
    id = heroship.1100
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        br_has_power_vanguard = yes
        owner = {
            has_country_flag = br_skill_energy_overclock_unlocked
            br_vanguard_power_skill_cd_ready = yes
        }
    }

    immediate = {
        owner = {
            set_variable = { which = br_vanguard_power_skill_cd value = 90 }
            if = {
                limit = { NOT = { has_country_flag = br_vanguard_skill_cd_tick_running } }
                set_country_flag = br_vanguard_skill_cd_tick_running
                country_event = { id = heroship.1014 days = 1 }
            }
        }
        every_owned_ship = {
            limit = {
                is_ship_size = br_vanguard
                has_component = BR_VANGUARD_CORE_POWER
            }
            add_modifier = {
                modifier = br_vanguard_power_energy_overclock
                days = 10
            }
        }
    }
}

# 越战越勇被动：每击杀 +1% 武器伤害，上限 50%（仅先锋舰）
country_event = {
    id = heroship.1102
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_country_flag = br_skill_kill_stack_unlocked
        check_variable = { which = br_vanguard_kill_count value < value:br_vanguard_kill_cap }
    }

    immediate = {
        change_variable = { which = br_vanguard_kill_count value = 1 }
        if = {
            limit = {
                check_variable = { which = br_vanguard_kill_count value > value:br_vanguard_kill_cap }
            }
            set_variable = { which = br_vanguard_kill_count value = value:br_vanguard_kill_cap }
        }

        every_owned_fleet = {
            every_owned_ship = {
                limit = {
                    is_ship_size = br_vanguard
                    has_component = BR_VANGUARD_CORE_POWER
                }
                remove_modifier = br_vanguard_power_kill_stack_damage
                add_modifier = {
                    modifier = br_vanguard_power_kill_stack_damage
                    days = -1
                    mult = owner.br_vanguard_kill_count
                }
            }
        }
    }
}

# 战意过载（接战后每日叠加伤害，脱战清空）— 初始化
fleet_event = {
    id = heroship.1103
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_owned_ship = {
            is_ship_size = br_vanguard
            has_component = BR_VANGUARD_CORE_POWER
        }
        owner = { has_country_flag = br_skill_combat_stack_unlocked }
    }

    immediate = {
        if = {
            limit = {
                NOT = { has_fleet_flag = br_vanguard_power_combat_stack_active }
            }
            set_fleet_flag = br_vanguard_power_combat_stack_active
            clear_variable = br_vanguard_power_combat_stack
            fleet_event = { id = heroship.1104 days = 1 }
        }
    }
}

# 战意过载每日 tick
fleet_event = {
    id = heroship.1104
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = {
                is_in_combat = yes
                any_owned_ship = {
                    is_ship_size = br_vanguard
                    has_component = BR_VANGUARD_CORE_POWER
                }
            }

            if = {
                limit = {
                    NOT = { is_variable_set = br_vanguard_power_combat_stack }
                }
                set_variable = { which = br_vanguard_power_combat_stack value = 0 }
            }

            if = {
                limit = {
                    check_variable = { which = br_vanguard_power_combat_stack value < 100 }
                }
                change_variable = { which = br_vanguard_power_combat_stack value = 1 }
            }

            remove_modifier = br_vanguard_power_combat_stack_damage
            add_modifier = {
                modifier = br_vanguard_power_combat_stack_damage
                days = 2
                mult = br_vanguard_power_combat_stack
            }

            fleet_event = { id = heroship.1104 days = 1 }
        }

        else = {
            remove_modifier = br_vanguard_power_combat_stack_damage
            clear_variable = br_vanguard_power_combat_stack
            remove_fleet_flag = br_vanguard_power_combat_stack_active
        }
    }
}
