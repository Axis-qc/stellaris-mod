namespace = lev

# 利维坦·虚空孳孽 技能事件（lev.270~279，不得越界）
# on_action 触发时 THIS = 施放舰队、from = 施放舰队
# 数值动态：add_modifier 的 mult = root.owner.value:lev_sk_*（等级驱动，
# 见 common/script_values/lev_skill_values.txt）；同技能内曲线不同的字段
# 按维度拆成多个 modifier 分别挂（一个 add_modifier 只有一个 mult）。
# 本期不做直接伤害，敌舰一律用减益（modifier）替代

# sk1 虚空触须：本星系敌舰伤害/护盾回复减益
fleet_event = {
    id = lev.270
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        from.owner = { save_event_target_as = lev_caster }

        solar_system = {
            every_ship_in_system = {
                limit = {
                    owner = {
                        is_hostile = event_target:lev_caster
                    }
                }
                add_modifier = { modifier = lev_voidspawn_sk1_dmg mult = root.owner.value:lev_sk_void_1_dmg days = 30 }
                add_modifier = { modifier = lev_voidspawn_sk1_sr mult = root.owner.value:lev_sk_void_1_sr days = 30 }
            }
        }
    }
}

# sk2 孳孽孕育：本舰船体再生与护盾增益
fleet_event = {
    id = lev.271
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        every_owned_ship = {
            limit = { is_ship_size = lev_voidspawn }
            add_modifier = { modifier = lev_voidspawn_sk2_hr mult = root.owner.value:lev_sk_void_2_hr days = 25 }
            add_modifier = { modifier = lev_voidspawn_sk2_sh mult = root.owner.value:lev_sk_void_2_sh days = 25 }
        }
    }
}
