namespace = reward

# ========================
# 奖励系统核心 — 总触发器
# reward.1 是全部普通奖励的唯一入口。
# 只负责随机选五个普通奖励池中的一个，不做等级/流派检测，不发奖励。
# 调用方：战斗击杀(br.1102)、爆星(br.1103)。
# ========================

country_event = {
    id = reward.1
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        random_list = {
            20 = { country_event = { id = reward.1001 } }   # 资源池
            25 = { country_event = { id = reward.2001 } }   # 科技池
            25 = { country_event = { id = reward.3001 } }   # BUFF池
            15 = { country_event = { id = reward.4001 } }   # 舰船池
            15 = { country_event = { id = reward.5001 } }   # 领袖池
        }
    }
}

# 英雄舰发放
country_event = {
    id = reward.6102
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { any_owned_fleet = { } }
            random_owned_design = {
                limit = { is_ship_size = br_vanguard }
                save_event_target_as = br_reward_vanguard_design
            }
            if = {
                limit = { exists = event_target:br_reward_vanguard_design }
                random_owned_fleet = { save_event_target_as = br_reward_vanguard_spawn_fleet }
                create_fleet = {
                    name = random
                    effect = {
                        set_owner = root
                        create_ship = {
                            name = random
                            design = event_target:br_reward_vanguard_design
                            graphical_culture = root
                        }
                        set_location = { target = event_target:br_reward_vanguard_spawn_fleet distance = 35 angle = random }
                    }
                }
            }
        }

        # 英雄舰等级初始化
        set_country_flag = br_hero_level_initialized
        set_variable = { which = br_hero_level value = 1 }
        set_variable = { which = br_hero_xp value = 0 }
        set_variable = { which = br_hero_xp_next value = 10 }
    }
}

# ========================
# 英雄舰独立等级系统 — reward.6200-6299
# reward.6200：后备初始化（主初始化在 reward.6102 发放英雄舰时）
# reward.6201：杀敌经验（由 on_ship_destroyed_perp 直接调用）
# reward.6202：等级检测与升级
# ========================

country_event = {
    id = reward.6200
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_country_type = br_player_pve
        NOT = { has_country_flag = br_hero_level_initialized }
    }

    immediate = {
        set_country_flag = br_hero_level_initialized
        set_variable = { which = br_hero_level value = 1 }
        set_variable = { which = br_hero_xp value = 0 }
        set_variable = { which = br_hero_xp_next value = 10 }
    }
}

# 英雄舰杀敌经验系统（由 on_ship_destroyed_perp 直接调用，fromfromfrom = 被击杀舰船）
country_event = {
    id = reward.6201
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_country_type = br_player_pve
        from = { is_country_type = br_pve_enemy }
        exists = fromfromfrom
        any_owned_ship = {
            is_ship_size = br_vanguard
            solar_system = {
                is_same_value = fromfrom.solar_system
            }
        }
    }

    immediate = {

        if = {
            limit = { fromfromfrom = { is_ship_size = br_pve_boss } }
            change_variable = { which = br_hero_xp value = 500 }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = battleship } }
            change_variable = { which = br_hero_xp value = 40 }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = cruiser } }
            change_variable = { which = br_hero_xp value = 20 }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = destroyer } }
            change_variable = { which = br_hero_xp value = 10 }
        }
        else_if = {
            limit = { fromfromfrom = { is_ship_size = corvette } }
            change_variable = { which = br_hero_xp value = 5 }
        }
        else = {
            change_variable = { which = br_hero_xp value = 3 }
        }

        country_event = { id = reward.6202 }
    }
}

# 英雄舰等级检测升级
country_event = {
    id = reward.6202
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { check_variable = { which = br_hero_xp value >= 6830 } }
