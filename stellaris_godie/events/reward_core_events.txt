namespace = reward

# ========================
# 奖励系统核心 — 总触发器
# reward.1 是全部普通奖励的唯一入口。
# 只负责随机选五个普通奖励池中的一个，不做等级/流派检测，不发奖励。
# 调用方：战斗击杀(br.1102)、爆星(br.1103)。
# ========================

country_event = {
    id = reward.1
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { any_owned_ship = { is_in_combat = yes } }
            change_variable = { which = br_pending_rewards value = 1 }
        }
        else = {
            random_list = {
                30 = { country_event = { id = reward.1001 } }   # 资源池
                25 = { country_event = { id = reward.2001 } }   # 科技池
                25 = { country_event = { id = reward.3001 } }   # BUFF池
                15 = { country_event = { id = reward.4001 } }   # 舰船池
                5 = { country_event = { id = reward.5001 } }   # 领袖池
            }
        }
    }
}

# ========================
# 奖励补发 — reward.2
# 每月触发，检测无战斗且 pending > 0 时补发
# ========================

country_event = {
    id = reward.2
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        check_variable = { which = br_pending_rewards value > 0 }
        NOT = { any_owned_ship = { is_in_combat = yes } }
    }

    immediate = {
        if = {
            limit = { check_variable = { which = br_pending_rewards value > 0 } }
            random_list = {
                30 = { country_event = { id = reward.1001 } }   # 资源池
                25 = { country_event = { id = reward.2001 } }   # 科技池
                25 = { country_event = { id = reward.3001 } }   # BUFF池
                5 = { country_event = { id = reward.4001 } }   # 舰船池
                5 = { country_event = { id = reward.5001 } }   # 领袖池
            }
            subtract_variable = { which = br_pending_rewards value = 1 }
        }
    }
}
