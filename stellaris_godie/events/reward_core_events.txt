namespace = reward

# ========================
# 奖励系统核心 — 总触发器
# reward.1 是全部普通奖励的唯一入口。
# 只负责随机选五个普通奖励池中的一个，不做等级/流派检测，不发奖励。
# 调用方：战斗击杀(br.1102)、爆星(br.1103)。
# ========================

country_event = {
    id = reward.1
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { any_owned_ship = { is_in_combat = yes } }
            change_variable = { which = br_pending_rewards value = 1 }
        }
        else = {
            random_list = {
                30 = { country_event = { id = reward.1001 } }   # 资源池
                25 = { country_event = { id = reward.2001 } }   # 科技池
                25 = { country_event = { id = reward.3001 } }   # BUFF池
                5 = { country_event = { id = reward.4001 } }   # 舰船池
                5 = { country_event = { id = reward.5001 } }   # 领袖池
            }
        }
    }
}

# ========================
# 奖励补发 — reward.2
# 每月触发，检测无战斗且 pending > 0 时补发
# ========================

country_event = {
    id = reward.2
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        check_variable = { which = br_pending_rewards value > 0 }
        NOT = { any_owned_ship = { is_in_combat = yes } }
    }

    immediate = {
        if = {
            limit = { check_variable = { which = br_pending_rewards value > 0 } }
            random_list = {
                30 = { country_event = { id = reward.1001 } }   # 资源池
                25 = { country_event = { id = reward.2001 } }   # 科技池
                25 = { country_event = { id = reward.3001 } }   # BUFF池
                5 = { country_event = { id = reward.4001 } }   # 舰船池
                5 = { country_event = { id = reward.5001 } }   # 领袖池
            }
            subtract_variable = { which = br_pending_rewards value = 1 }
        }
    }
}

# ========================
# 英雄舰发放 — reward.6101
# 由 br_reward_level_events.txt 在玩家达到 2 级时调用。
# 核心选择直接内置于本事件：选择核心后初始化等级系统，60天后先锋舰折跃抵达。
# ========================

country_event = {
    id = reward.6101
    title = reward.6101_title
    desc = reward.6101_desc
    is_triggered_only = yes

    trigger = {
        NOT = { has_country_flag = br_hero_ship_received }
    }

    option = {
        name = REWARD_HERO_CORE_POWER
        custom_tooltip = REWARD_HERO_CORE_POWER_TT
        set_country_flag = br_hero_ship_received
        set_country_flag = br_hero_level_initialized
        set_variable = { which = br_hero_level value = 1 }
        set_variable = { which = br_hero_xp value = 0 }
        set_variable = { which = br_hero_xp_next value = 10 }
        set_country_flag = br_hero_core_selected
        set_country_flag = br_hero_core_power
        country_event = { id = reward.6102 days = 60 }
    }
    option = {
        name = REWARD_HERO_CORE_DEFENSE
        custom_tooltip = REWARD_HERO_CORE_DEFENSE_TT
        set_country_flag = br_hero_ship_received
        set_country_flag = br_hero_level_initialized
        set_variable = { which = br_hero_level value = 1 }
        set_variable = { which = br_hero_xp value = 0 }
        set_variable = { which = br_hero_xp_next value = 10 }
        set_country_flag = br_hero_core_selected
        set_country_flag = br_hero_core_defense
        country_event = { id = reward.6102 days = 60 }
    }
    option = {
        name = REWARD_HERO_CORE_SUPPORT
        custom_tooltip = REWARD_HERO_CORE_SUPPORT_TT
        set_country_flag = br_hero_ship_received
        set_country_flag = br_hero_level_initialized
        set_variable = { which = br_hero_level value = 1 }
        set_variable = { which = br_hero_xp value = 0 }
        set_variable = { which = br_hero_xp_next value = 10 }
        set_country_flag = br_hero_core_selected
        set_country_flag = br_hero_core_support
        country_event = { id = reward.6102 days = 60 }
    }
}

# 先锋英雄舰折跃抵达（reward.6101 的延迟落地事件）
country_event = {
    id = reward.6102
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        random_owned_design = {
            limit = { is_ship_size = br_vanguard }
            save_event_target_as = br_reward_vanguard_design
        }
        owner = {
            random_owned_ship = {
                limit = { is_ship_size = br_dominator }
                fleet = { save_event_target_as = br_vanguard_arrival_fleet }
            }
        }
        create_fleet = {
            name = "BR_VANGUARD_ARRIVAL"
            effect = {
                set_owner = root
                create_ship = {
                    name = random
                    design = event_target:br_reward_vanguard_design
                    graphical_culture = root
                }
                set_location = { target = event_target:br_vanguard_arrival_fleet distance = 35 angle = random }
            }
        }
    }
}
