namespace = br

# br.501：首次进入游戏时打开大逃杀开局设置窗口喵
country_event = {
	id = br.501
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		NOT = { has_global_flag = br_start_settings_opened }
		NOR = {
			is_country_type = br_player
			is_country_type = br_player_pvp
			is_country_type = br_player_pve
		}
	}

	immediate = {
		set_global_flag = br_start_settings_opened
		country_event = { id = br.500 }
	}
}

# br.502：扩圈完成后按模式创建标准玩家国家，并直接在出生点生成初始舰队喵
country_event = {
	id = br.502
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		NOR = {
			is_country_type = br_player
			is_country_type = br_player_pvp
			is_country_type = br_player_pve
		}
	}

	immediate = {
		capital_star = {
			save_event_target_as = br_player_spawn_star
			solar_system = {
				save_event_target_as = br_player_spawn_system
			}
		}

		if = {
			limit = { has_global_flag = br_mode_pve }
			country_event = { id = br.530 }
		}

		# PVP开局后续会在这里接入自己的出生点选择规则喵

		if = {
			limit = { has_global_flag = br_mode_pve }
			create_country = {
				name = "BR_PLAYER_PVE_COUNTRY_NAME"
				type = br_player_pve
				species = random
				authority = auth_democratic
				ethos = {
					ethic = ethic_egalitarian
					ethic = ethic_xenophile
					ethic = ethic_materialist
				}
				civics = {
					civic = civic_beacon_of_liberty
					civic = civic_idealistic_foundation
				}
				origin = origin_default
				ignore_initial_colony_error = yes
				day_zero_contact = no
				flag = {
					icon = {
						category = "ornate"
						file = "flag_ornate_8.dds"
					}
					background = {
						category = "backgrounds"
						file = "00_solid.dds"
					}
					colors = {
						"blue"
						"white"
						"null"
						"null"
					}
				}
				effect = {
					save_event_target_as = br_new_player_country
					set_country_flag = br_player_country
				}
			}
		}
		else = {
			create_country = {
				name = "BR_PLAYER_PVP_COUNTRY_NAME"
				type = br_player_pvp
				species = random
				authority = auth_democratic
				ethos = {
					ethic = ethic_egalitarian
					ethic = ethic_xenophile
					ethic = ethic_materialist
				}
				civics = {
					civic = civic_beacon_of_liberty
					civic = civic_idealistic_foundation
				}
				origin = origin_default
				ignore_initial_colony_error = yes
				day_zero_contact = no
				flag = {
					icon = {
						category = "ornate"
						file = "flag_ornate_8.dds"
					}
					background = {
						category = "backgrounds"
						file = "00_solid.dds"
					}
					colors = {
						"red"
						"black"
						"null"
						"null"
					}
				}
				effect = {
					save_event_target_as = br_new_player_country
					set_country_flag = br_player_country
				}
			}
		}

		if = {
			limit = {
				has_global_flag = br_mode_pve
				NOT = { has_global_flag = br_pve_alliance_leader_set }
			}
			set_global_flag = br_pve_alliance_leader_set
			event_target:br_new_player_country = {
				save_global_event_target_as = br_pve_alliance_leader
			}
		}

		event_target:br_new_player_country = {
			create_fleet = {
				name = "BR_MOTHERSHIP_FLEET"
				effect = {
					set_owner = event_target:br_new_player_country
					create_ship = {
						name = "NAME_BR_Dominator"
						design = "NAME_BR_Dominator"
						prefix = no
					}
					set_location = {
						target = event_target:br_player_spawn_star
						distance = 50
						angle = random
					}
				}
			}

			create_leader = {
				class = commander
				species = owner_main_species
				name = "BR_ESCORT_COMMANDER"
				skill = 3
			}

			create_fleet = {
				name = "BR_ESCORT_FLEET"
				effect = {
					set_owner = event_target:br_new_player_country
					while = {
						count = 3
						create_ship = {
							name = random
							design = NAME_Dagger
						}
					}
					assign_leader = last_created_leader
					set_location = {
						target = event_target:br_player_spawn_star
						distance = 80
						angle = random
					}
				}
			}

			set_player = root
			if = {
				limit = { has_global_flag = br_mode_pve }
				country_event = { id = br.770 }
			}
			country_event = { id = br.510 days = 5 }
		}

		if = {
			limit = {
				NOT = { has_global_flag = br_start_map_cleaned }
			}

			set_global_flag = br_start_map_cleaned

			every_system = {
				limit = {
					exists = starbase
				}

				starbase = {
					fleet = {
						destroy_fleet = this
					}
				}
			}

			every_system = {
				every_ship_in_system = {
					limit = {
						OR = {
							NOT = { exists = owner }
							owner = {
								NOR = {
									is_country_type = br_player
									is_country_type = br_player_pvp
									is_country_type = br_player_pve
								}
							}
						}
					}

					destroy_ship = this
				}
			}

			every_country = {
				every_owned_planet = {
					destroy_colony = yes
				}
			}

			every_country = {
				limit = {
					is_ai = yes
					is_country_type = default
				}

				destroy_country = yes
			}
		}

		country_event = { id = br.520 }

		destroy_country = yes
	}
}

# br.510：完成开局后的PVP/PVE关系、敌方国家创建和PVE刷怪处理喵
country_event = {
	id = br.510
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { is_country_type = br_player_pvp }
			every_country = {
				limit = {
					NOT = { is_same_value = root }
					OR = {
						is_country_type = br_player_pvp
						is_country_type = br_player_pve
						is_country_type = br_player
					}
				}
				establish_communications_no_message = root
			}
		}

		if = {
			limit = { is_country_type = br_player_pve }
			country_event = { id = br.770 }

			if = {
				limit = {
					NOR = {
						has_global_flag = br_pve_template_artillery
						has_global_flag = br_pve_template_brawler
						has_global_flag = br_pve_template_mixed
					}
				}
				set_global_flag = br_pve_template_mixed
			}

			every_country = {
				limit = {
					NOT = { is_same_value = root }
					is_country_type = br_player_pve
				}
				establish_communications_no_message = root
				add_trust = {
					amount = 100
					who = root
				}
				add_intel = {
					amount = 100
					who = root
				}
				root = {
					add_trust = {
						amount = 100
						who = prev
					}
					add_intel = {
						amount = 100
						who = prev
					}
				}
			}

			if = {
				limit = {
					exists = event_target:br_pve_alliance_leader
					NOT = { is_same_value = event_target:br_pve_alliance_leader }
				}
				join_alliance = {
					who = event_target:br_pve_alliance_leader
					name = BR_PVE_FEDERATION_NAME
					override_requirements = yes
				}
			}

			if = {
				limit = { has_federation = yes }
				federation = {
					set_federation_type = default_federation
					add_cohesion = 1000
					add_federation_experience = 600
				}
			}

			# PVE模式：创建敌对势力国家，仅执行一次喵
			if = {
				limit = {
					has_global_flag = br_mode_pve
					NOT = { has_global_flag = br_pve_enemy_created }
				}
				set_global_flag = br_pve_enemy_created
				create_country = {
					name = "BR_PVE_ENEMY_NAME"
					type = br_pve_enemy
					species = random
					authority = auth_democratic
					ethos = {
						ethic = ethic_egalitarian
						ethic = ethic_xenophile
						ethic = ethic_materialist
					}
					civics = {
						civic = civic_beacon_of_liberty
						civic = civic_idealistic_foundation
					}
					origin = origin_default
					ignore_initial_colony_error = yes
					day_zero_contact = no
					flag = {
						icon = {
							category = "ornate"
							file = "flag_ornate_8.dds"
						}
						background = {
							category = "backgrounds"
							file = "00_solid.dds"
						}
						colors = {
							"red"
							"black"
							"null"
							"null"
						}
					}
					effect = {
						save_global_event_target_as = br_pve_enemy_country
						set_country_flag = br_pve_enemy
						set_graphical_culture = mammalian_01
					}
				}
			}

			if = {
				limit = {
					has_global_flag = br_mode_pve
					NOT = { has_global_flag = br_pve_initialized }
					exists = event_target:br_pve_enemy_country
				}
				country_event = { id = br.531 }
			}

		}
	}
}
