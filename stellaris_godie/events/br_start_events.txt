namespace = br

country_event = {
	id = br.501
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		NOT = { is_country_type = br_player }
	}

	immediate = {
		if = {
			limit = {
				NOT = { has_global_flag = br_start_map_cleaned }
			}

			set_global_flag = br_start_map_cleaned

			every_system = {
				limit = {
					exists = starbase
				}

				starbase = {
					fleet = {
						destroy_fleet = this
					}
				}
			}

			every_system = {
				every_ship_in_system = {
					limit = {
						OR = {
							is_ship_size = mining_station
							is_ship_size = research_station
							is_ship_size = observation_station
							is_ship_size = military_station_small
							is_ship_size = military_station_medium
							is_ship_size = military_station_large
							is_ship_size = ion_cannon
						}
					}

					destroy_ship = this
				}
			}

			every_country = {
				every_owned_planet = {
					destroy_colony = yes
				}
			}

			every_country = {
				limit = {
					is_ai = yes
					is_country_type = default
				}

				destroy_country = yes
			}
		}

		create_country = {
			name = random
			type = br_player
			species = owner_main_species
			ignore_initial_colony_error = yes
			day_zero_contact = no
			flag = random
			effect = {
				save_event_target_as = br_new_player_country
				set_country_flag = br_player_country
			}
		}

		event_target:br_new_player_country = {
			set_player = root
			country_event = { id = br.100 }

			create_fleet = {
				name = "BR_STARTING_FLEET"
				effect = {
					set_owner = event_target:br_new_player_country
					while = {
						count = 3
						create_ship = {
							name = random
							design = NAME_Dagger
						}
					}
					set_location = {
						target = event_target:battle_royale_center.star
						distance = 50
						angle = random
					}
				}
			}
		}

		destroy_country = yes
	}
}
