# br.500：首次进入游戏时打开大逃杀开局设置窗口喵
country_event = {
    id = br.500
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_ai = no
        NOT = { has_global_flag = br_start_settings_opened }
        NOR = {
            is_country_type = br_player
            is_country_type = br_player_pvp
            is_country_type = br_player_pve
        }
    }

    immediate = {
        set_global_flag = br_start_settings_opened
        country_event = { id = br.501 }
    }
}

# br.504：创建虚空工具国家，启动扩圈，为所有原版玩家排程模式初始化喵
country_event = {
    id = br.504
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        # 每次游戏开始时创建虚空工具国家，承载全局变量和定时事件触发喵
        create_country = {
            name = "BR_VOID_COUNTRY_NAME"
            type = br_void
            species = random
            authority = auth_democratic
            ethos = {
                ethic = ethic_egalitarian
                ethic = ethic_xenophile
                ethic = ethic_materialist
            }
            civics = {
                civic = civic_beacon_of_liberty
                civic = civic_idealistic_foundation
            }
            origin = origin_default
            ignore_initial_colony_error = yes
            day_zero_contact = no
            flag = {
                icon = {
                    category = "ornate"
                    file = "flag_ornate_8.dds"
                }
                background = {
                    category = "backgrounds"
                    file = "00_solid.dds"
                }
                colors = {
                    "black"
                    "black"
                    "null"
                    "null"
                }
            }
            effect = {
                save_global_event_target_as = br_void_country
                set_country_flag = br_start_init
            }
        }

        # 排程缩圈和圈外伤害（每局首次创建时排程）喵
        if = {
            limit = {
                event_target:br_void_country = {
                    has_country_flag = br_start_init
                }
            }
            event_target:br_void_country = {
                country_event = { id = br.300 days = 1079 }
                country_event = { id = br.1000 days = 1079 }
            }
        }

        # 启动圈层框架扩圈喵
        country_event = { id = br.100 }

        # 为所有需要替换的原版玩家国家排程模式初始化喵
        every_country = {
            limit = {
                is_ai = no
                NOT = { has_country_flag = br_player_replacement_scheduled }
                NOR = {
                    is_country_type = br_player
                    is_country_type = br_player_pvp
                    is_country_type = br_player_pve
                }
            }
            set_country_flag = br_player_replacement_scheduled
            country_event = { id = br.505 days = 1 }
        }
    }
}

# br.505：按模式分流，排程对应初始化流程喵
country_event = {
    id = br.505
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_ai = no
        NOR = {
            is_country_type = br_player
            is_country_type = br_player_pvp
            is_country_type = br_player_pve
        }
    }

    immediate = {
        if = {
            limit = { has_global_flag = br_mode_pve }
            if = {
                limit = { NOT = { has_global_flag = br_pve_player_transfer_started } }
                set_global_flag = br_pve_player_transfer_started
                if = {
                    limit = { exists = event_target:br_void_country }
                    event_target:br_void_country = {
                        country_event = { id = br.700 }
                    }
                }
                else = {
                    country_event = { id = br.700 }
                }
            }
        }
        else = {
            country_event = { id = br.600 }
        }
    }
}

# br.507：地图清理与后续排程，在玩家国家创建完成后执行喵
country_event = {
    id = br.507
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        OR = {
            is_country_type = br_player_pve
            is_country_type = br_player_pvp
        }
    }

    immediate = {
        # 清理地图上的原版国家和舰队（每局首次执行时清理）喵
        if = {
            limit = {
                exists = event_target:br_void_country
                event_target:br_void_country = {
                    has_country_flag = br_start_init
                }
            }

            event_target:br_void_country = {
                remove_country_flag = br_start_init
            }

            every_system = {
                limit = {
                    exists = starbase
                }

                starbase = {
                    fleet = {
                        destroy_fleet = this
                    }
                }
            }

            every_system = {
                every_ship_in_system = {
                    limit = {
                        OR = {
                            NOT = { exists = owner }
                            owner = {
                                NOR = {
                                    is_country_type = br_player
                                    is_country_type = br_player_pvp
                                    is_country_type = br_player_pve
                                }
                            }
                        }
                    }
                    destroy_ship = this
                }
            }

            every_country = {
                every_owned_planet = {
                    destroy_colony = yes
                }
            }

            every_country = {
                limit = {
                    is_ai = yes
                    NOT = {
                        is_country_type = br_void
                        is_country_type = br_player
                        is_country_type = br_player_pvp
                        is_country_type = br_player_pve
                    }
                }
                destroy_country = yes
            }
        }

        # 排程下一阶段：PVE敌对势力 或 PVP通信喵
        if = {
            limit = { has_global_flag = br_mode_pve }
            country_event = { id = br.702 days = 4 }
        }
        else = {
            country_event = { id = br.601 days = 4 }
        }
    }
}
