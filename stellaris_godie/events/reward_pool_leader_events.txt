namespace = reward

# ========================
# 领袖池 — reward.5001-5999
# 只能由 reward.1 总触发器调用。
# 五级制分流，具体奖励只展示指挥官/科学家自选。
# ========================

# 领袖池入口
country_event = {
    id = reward.5001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        country_event = { id = reward.5101 }
    }
}

# 领袖池等级触发器
country_event = {
    id = reward.5101
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        if = {
            limit = { check_variable = { which = br_power_level value < 3 } }
            country_event = { id = reward.5102 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 6 } }
            country_event = { id = reward.5103 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 9 } }
            country_event = { id = reward.5104 }
        }
        else_if = {
            limit = { check_variable = { which = br_power_level value < 12 } }
            country_event = { id = reward.5105 }
        }
        else = {
            country_event = { id = reward.5106 }
        }
    }
}
