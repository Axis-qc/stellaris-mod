namespace = reward

# ========================
# 领袖池 — reward.5001-5099
# 只能由 reward.1 总触发器调用。
# 经验制：已有该类型领袖则注入经验，无则招募新领袖。
# 经验值由 script_value br_leader_xp 动态计算（500×战力等级）。
# ========================

# 领袖池入口
country_event = {
    id = reward.5001
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        set_variable = { which = br_leader_xp value = 500 }
        multiply_variable = { which = br_leader_xp value = br_power_level }
        country_event = { id = reward.5101 }
    }
}

# 领袖奖励 — 六选一（三类领袖各分经验/招募）
country_event = {
    id = reward.5101
    is_triggered_only = yes
    title = REWARD_POOL_LEADER_TITLE
    desc = REWARD_POOL_LEADER_DESC
    picture = GFX_evt_ship_in_orbit

    option = {
        name = REWARD_LEADER_COMMANDER_XP
        custom_tooltip = REWARD_LEADER_COMMANDER_XP_TT
        allow = {
            any_owned_leader = {
                leader_class = commander
                NOT = { has_base_skill >= 10 }
            }
        }
        random_owned_leader = {
            limit = {
                leader_class = commander
                NOT = { has_base_skill >= 10 }
            }
            add_experience = owner.value:br_leader_xp
        }
    }
    option = {
        name = REWARD_LEADER_COMMANDER_RECRUIT
        custom_tooltip = REWARD_LEADER_COMMANDER_RECRUIT_TT
        create_leader = {
            class = commander
            species = owner_main_species
            name = random
            skill = 1
        }
    }
    option = {

        create_leader = {
            class = scientist
            species = owner_main_species

    }
    option = {
        name = REWARD_LEADER_OFFICIAL_XP
        custom_tooltip = REWARD_LEADER_OFFICIAL_XP_TT

            any_owned_leader = {

        create_leader = {
            class = official
