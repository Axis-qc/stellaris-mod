# 火力核心
utility_component_template = {
	# 组件ID
	key = "BR_VANGUARD_CORE_POWER"
	# 组件大小
	size = small
	# 图标
	icon = "GFX_ship_part_aura_targeting_grid"
	icon_frame = 1
	# 能耗
	power = 0
	# 限定舰船类型
	size_restriction = {
		br_vanguard
	}
	# 所属组件槽位
	component_set = "br_hero_ship_Core"
	# 解锁条件：需选择火力核心
	potential = {
		from = {
			has_country_flag = br_hero_core_power
		}
	}
	# 属性加成：每英雄等级 +500 船体/装甲/护盾
	triggered_ship_modifier = {
		modifier = {
			ship_hull_add = 500
			ship_armor_add = 500
			ship_shield_add = 500
			ship_damage_reduction_add = 1
			ship_weapon_damage = 0.1
			ship_hull_regen_add_perc = 0.01
			ship_armor_regen_add_perc = 0.01
		}
		mult = owner.br_hero_level
	}
	scripted_action = {
		br_vanguard_gather
		br_vanguard_power_skill
		br_vanguard_power_ultimate
	}
	# AI权重
	ai_weight = {
		weight = 999
	}
}
