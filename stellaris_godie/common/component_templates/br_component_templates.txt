        # button_visible:   按钮是否显示在舰队面板上 → 用 flag 控制哪个技能可见

            custom_tooltip = {
                fail_text = br_dominator_repair_fail_tt

                    check_variable = { which = br_dominator_regen_charge value > 0 }

                }
            }
        }
