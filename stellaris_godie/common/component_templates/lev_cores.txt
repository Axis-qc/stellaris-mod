################################### 噬星者 ###################################

utility_component_template = {
    key = "LEV_STELLARITE_CORE"
    size = small
    icon = "GFX_ship_part_aura_electromagnetic_disruptor"
    icon_frame = 1
    power = 0
    potential = {
        is_ship_size = lev_stellarite
    }
    component_set = "lev_stellarite_Core"
    triggered_ship_modifier = {
        modifier = {
            ship_weapon_damage = 0.05
            ship_shield_add = 300
            ship_shield_regen_add_static = 4
            ship_hull_regen_add_perc = 0.01
        }
        mult = owner.lev_stellarite_growth_level
    }
    scripted_action = {
        br_dominator_level_panel        # 打开等级界面（技能栏第 1 位）
        # 吞噬恒星已改由 planet_killer 槽位武器接管，不再是技能按钮
        lev_stellarite_ignite
        lev_stellarite_beam
        lev_stellarite_support          # 行星支援（slot 4，点目标行星）
    }
    ai_weight = {
        weight = 999
    }
}
