# 利维坦·噬星者 武器（复制自原版 STELLARITE_*）
# 限定舰种一律用 potential，别改回 size_restriction

# 噬星之光 — XL 槽（迷你恒星弹丸）
weapon_component_template = {
	key = "LEV_STELLARITE_BEAM"
	size = extra_large
	type = instant
	prio_projectile = yes

	icon = "GFX_ship_part_energy_lance_2"
	icon_frame = 1

	firing_arc = 35.0
	min_range = 10.0

	tags = { weapon_type_energy }
	component_set = "LEV_STELLARITE_BEAM"
	projectile_gfx = "lev_stellarite_star_shot"

	potential = {
		is_ship_size = lev_stellarite
	}

	damage = {
		min = 400
		max = 1200
	}
	# 基础伤害压到开局水平；命中后按成长等级追加（见下方 on_hit）
	hull_damage = 2
	shield_damage = 0.75
	shield_penetration = 0
	armor_damage = 2
	armor_penetration = 0
	windup = {
		min = 12
		max = 47
	}
	total_fire_time = 70
	range = 120
	accuracy = 0.85
	tracking = 0

	on_hit = {
		# THIS = 被击中的船，from = 射击方（噬星者），动态伤害 = (成长等级 − 1) × 750
		hidden_effect = {
			from.owner = { save_event_target_as = lev_beam_caster }
			from.owner = {
				# 等级未初始化按 0 处理，防负数回血
				if = {
					limit = { NOT = { is_variable_set = lev_stellarite_growth_level } }
					set_variable = { which = lev_beam_hit_dmg value = 0 }
				}
				else = {
					set_variable = { which = lev_beam_hit_dmg value = lev_stellarite_growth_level }
					subtract_variable = { which = lev_beam_hit_dmg value = 1 }
					multiply_variable = { which = lev_beam_hit_dmg value = 750 }
				}
			}
			if = {
				limit = {
					event_target:lev_beam_caster = {
						check_variable = { which = lev_beam_hit_dmg value > 0 }
					}
				}
				damage_ship = {
					amount = event_target:lev_beam_caster.lev_beam_hit_dmg
					attacker = event_target:lev_beam_caster
				}
			}
		}
		custom_tooltip = "LEV_STELLARITE_BEAM_ON_HIT"
	}

}
