# ========================
# 英雄舰 on_actions — 从 br_on_actions.txt 独立
# Stellaris 会合并同目录下同 key 的 on_action，所以拆分后仍正常工作
# ========================

# ── 年度脉冲（英雄舰发放入口）──
on_yearly_pulse_country = {
	events = {
		heroship.50		# 英雄舰发放路由（开局1年后）
	}
}

# ── 每月脉冲（英雄舰部分）──
on_monthly_pulse_country = {
	events = {
		heroship.1207		# 防御核心大招显示数值更新
		heroship.1203		# 协同结构力场·每月刷新（自带 flag 检测）
		heroship.1202		# 结构增幅·每月刷新（等级变化自动更新）
		heroship.1331		# 无人机·月度链起点（全局初始化→计算变量→补船→位置/buff）
	}
}

# ── 击杀敌舰（英雄舰部分）──
on_ship_destroyed_perp = {
	events = {
		heroship.2			# 英雄舰杀敌经验
		heroship.1102		# 火力被动·越战越勇击杀叠加
	}
}
