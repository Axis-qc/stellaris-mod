# 利维坦·噬星者 数值

# 噬星之光 on_hit 追加伤害 = (成长等级 − 1) × 750；1 级不追加
lev_beam_hit_dmg = {
    base = 0
    add = owner.lev_stellarite_growth_level
    subtract = 1
    mult = 750
}

# 恒星光束单次伤害 = 500 + 成长等级 × 100
lev_beam_damage = {
    base = 500
    add = value:lev_beam_growth_bonus
}

# 引用的是该利维坦自己的成长等级变量（每只利维坦一套），
# 不是累积成长值——后者是无上限累积值（12 级需 2200 点），直接乘会算出几万几十万。
lev_beam_growth_bonus = {
    base = 0
    add = owner.lev_stellarite_growth_level
    mult = 100
}

# 腹中日轮单次伤害 = 1200 + 成长等级 × 120
lev_sun_damage = {
    base = 1200
    add = value:lev_sun_growth_bonus
}

lev_sun_growth_bonus = {
    base = 0
    add = owner.lev_stellarite_growth_level
    mult = 120
}

# 腹中日轮行星破坏 = 20 + 成长等级 × 2
lev_sun_devastation = {
    base = 20
    add = value:lev_sun_dev_step
}

lev_sun_dev_step = {
    base = 0
    add = owner.lev_stellarite_growth_level
    mult = 2
}

# 饥饿态（能量 ≤ 2）：两门伤害技能上浮 25%。
# mult 对累计值相乘，base = 0 即「取主值后整体 ×1.25」。
lev_beam_damage_hungry = {
    base = 0
    add = value:lev_beam_damage
    mult = 1.25
}

lev_sun_damage_hungry = {
    base = 0
    add = value:lev_sun_damage
    mult = 1.25
}
