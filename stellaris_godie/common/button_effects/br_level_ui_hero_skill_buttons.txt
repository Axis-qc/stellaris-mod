# 英雄舰技能切换按钮 — 国家作用域
# allow 查对方技能的 _unlocked flag，点过一次后两边互通

# ── 主动技能 slot 1 ──
br_level_ui_active_slot1 = {
    potential = {
        from = {
            check_variable = { which = br_level_ui_tab value = 1 }
        }
    }
    allow = {
        custom_tooltip = {
            fail_text = BR_LEVEL_UI_SKILL_FAIL_TT
            from = { NOT = { has_country_flag = br_skill_active_slot_1 } }
        }
        custom_tooltip = {
            fail_text = BR_LEVEL_UI_SKILL_LOCKED
            from = {
                OR = {
                    AND = { has_country_flag = br_hero_core_defense has_country_flag = br_skill_barrier_unlocked }
                    AND = { has_country_flag = br_hero_core_power has_country_flag = br_skill_weakness_scan_unlocked }
                }
            }
        }
    }
    effect = {
        hidden_effect = {
            from = {
                remove_country_flag = br_skill_active_slot_2
                set_country_flag = br_skill_active_slot_1
                if = { limit = { has_country_flag = br_hero_core_defense } remove_country_flag = br_skill_barrier_unlocked set_country_flag = br_skill_phase_shift_unlocked }
                if = { limit = { has_country_flag = br_hero_core_power } remove_country_flag = br_skill_weakness_scan_unlocked set_country_flag = br_skill_energy_overclock_unlocked }
            }
        }
    }
}
