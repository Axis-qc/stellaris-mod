br_dominator_repair = {
	user_scope = fleet
	scope = self
	possible = {
		custom_tooltip = {
			fail_text = br_dominator_repair_fail_tt
			always = yes
			owner = {
				check_variable = {
					which = br_dominator_regen_charge
					value > 0
				}
			}
		}
	}
	button_visible = {
		owner = {
			OR = {
				any_owned_ship = {
					is_ship_size = br_dominator
				}
				any_owned_ship = {
					is_ship_size = br_civilian_arkship_tier_1
				}
				any_owned_ship = {
					is_ship_size = br_civilian_arkship_tier_2
				}
				any_owned_ship = {
					is_ship_size = br_civilian_arkship_tier_3
				}
				any_owned_ship = {
					is_ship_size = br_science_arkship_tier_1
				}
				any_owned_ship = {
					is_ship_size = br_science_arkship_tier_2
				}
				any_owned_ship = {
					is_ship_size = br_science_arkship_tier_3
				}
				any_owned_ship = {
					is_ship_size = br_military_arkship_tier_1
				}
				any_owned_ship = {
					is_ship_size = br_military_arkship_tier_2
				}
				any_owned_ship = {
					is_ship_size = br_military_arkship_tier_3
				}
			}
		}
	}
	button_clickable = {
		custom_tooltip = {
			fail_text = br_dominator_repair_fail_tt
			always = yes
			owner = {
				check_variable = {
					which = br_dominator_regen_charge
					value > 0
				}
			}
		}
	}
	on_completed = on_br_dominator_repair_activated
	tooltip = br_dominator_repair_ship_button
	icon = GFX_br_dominator_repair
	icon_selected = GFX_br_dominator_repair
	activity_key = br_dominator_repair_activity_key
	context_menu_name = br_dominator_repair_menu_title
	progress_activity_key = br_dominator_repair_progress_activity_key
	required_progress = 4
	wait = no
	slot = 1
}
