# 利维坦·位面之魇 技能（恐惧召唤）
# possible：This = 舰队自身、From = 施放舰队（国家用 from.owner）；button_clickable 是舰队作用域，用 owner
# 本期无资源门槛，possible / button_clickable 直接放行

# 恐惧压制（槽1）：本星系敌舰武器伤害与射速降低
lev_horror_fear = {
    user_scope = fleet
    scope = self

    possible = {
        always = yes
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_horror }
    }

    button_clickable = {
        always = yes
    }

    on_completed = on_lev_horror_sk1_completed
    tooltip = lev_horror_fear_button
    icon = GFX_fleet_action_button_behemoth_damage
    icon_selected = GFX_fleet_action_button_behemoth_damage
    activity_key = lev_horror_fear_activity_key
    context_menu_name = lev_horror_fear_menu_title
    progress_activity_key = lev_horror_fear_progress_key

    wait = no
    required_progress = 5
    cooldown = 240
    slot = 1
}
