# 利维坦·毒圣 技能（腐蚀）
# possible：This = 舰队自身、From = 施放舰队；button_clickable 是舰队作用域
# 本期无资源门槛，possible / button_clickable 直接放行

# 腐蚀之雾（槽1）：本星系敌舰护甲与武器伤害削弱
lev_toxic_god_corrosion = {
    user_scope = fleet
    scope = self

    possible = {
        always = yes
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_toxic_god }
    }

    button_clickable = {
        always = yes
    }

    on_completed = on_lev_toxic_god_sk1_completed
    tooltip = lev_toxic_god_corrosion_button
    icon = GFX_fleet_action_button_behemoth_damage
    icon_selected = GFX_fleet_action_button_behemoth_damage
    activity_key = lev_toxic_god_corrosion_activity_key
    context_menu_name = lev_toxic_god_corrosion_menu_title
    progress_activity_key = lev_toxic_god_corrosion_progress_key

    wait = no
    required_progress = 5
    cooldown = 240
    slot = 1
}
