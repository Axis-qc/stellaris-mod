# 利维坦·噬星者 技能
# possible 在目标恒星作用域（国家用 from.owner），button_clickable 在舰队作用域

# 吞噬恒星走 planet_killer 组件，不在这里
# lev_absorb_fail_* 被 scripted_triggers/lev_stellarite_planet_killers.txt 复用，不要删

# 重燃恒星（槽1）
lev_stellarite_ignite = {
    user_scope = fleet
    scope = planet

    possible = {
        custom_tooltip = {
            fail_text = lev_fail_not_star
            is_star = yes
        }
        custom_tooltip = {
            fail_text = lev_ignite_fail_no_energy
            from.owner = {
                check_variable = { which = lev_stellarite_energy value > 0 }
            }
        }
        custom_tooltip = {
            fail_text = lev_ignite_fail_edge
            NOT = { is_star_class = sc_b }
        }
        custom_tooltip = {
            fail_text = lev_ignite_fail_multi_star
            NOR = { is_binary_star = yes  is_trinary_star = yes }
        }
        custom_tooltip = {
            fail_text = lev_ignite_fail_special_star
            NOR = {
                is_star_class = sc_black_hole
                is_star_class = sc_neutron_star
                is_star_class = sc_pulsar
                is_star_class = sc_toxoid_star
                is_star_class = sc_rift_star
            }
        }
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_stellarite }
    }

    button_clickable = {
        custom_tooltip = {
            fail_text = lev_ignite_fail_no_energy
            owner = {
                check_variable = { which = lev_stellarite_energy value > 0 }
            }
        }
    }

    automation = {
        priority = 3
        default_on = no
        tooltip = lev_auto_tt_ignite
    }

    on_completed = on_lev_ignite_completed
    tooltip = lev_stellarite_ignite_button
    icon = GFX_fleet_action_button_arkship_stellar_ignition
    icon_selected = GFX_fleet_action_button_arkship_stellar_ignition
    activity_key = lev_stellarite_ignite_activity_key
    context_menu_name = lev_stellarite_ignite_menu_title
    progress_activity_key = lev_stellarite_ignite_progress_key

    begin_anim_state = "breath_begin"
    progress_anim_state = "breath_loop"
    finish_anim_state = "breath_end"

    wait = no
    required_progress = 5
    cooldown = 360
    slot = 1

    ai = {
        # This = 目标恒星，From = 施放舰队
        weight = {
            base = 10
            # 订单在飞期间封权
            inline_script = scripted_actions/lev_stellarite_absorb_guard
            # 只在自家星系重燃
            modifier = {
                factor = 0
                solar_system = {
                    NOT = { space_owner? = { is_same_value = FROM.controller } }
                }
            }
            # 蓝星不再重燃
            modifier = {
                factor = 0
                is_star_class = sc_b
            }
        }
        max_jumps = 6
        good_enough_weight = 10
    }
}

# 恒星光束（槽2）：星系选择器，订单只负责选目标，其余走 on_lev_beam_targeted → lev.102 → lev.106
lev_stellarite_beam = {
    user_scope = fleet
    scope = galactic_object

    possible = {
        custom_tooltip = {
            fail_text = lev_beam_fail_no_energy
            from.owner = {
                check_variable = { which = lev_stellarite_energy value > 0 }
            }
        }
        # 射程即自身传感器范围
        custom_tooltip = {
            fail_text = lev_beam_fail_out_of_range
            is_in_sensor_range_of_country = from.owner
        }
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_stellarite }
    }

    button_clickable = {
        custom_tooltip = {
            fail_text = lev_beam_fail_no_energy
            owner = {
                check_variable = { which = lev_stellarite_energy value > 0 }
            }
        }
    }

    automation = {
        priority = 1
        default_on = yes
        tooltip = lev_auto_tt_beam
    }

    on_started = on_lev_beam_targeted
    tooltip = lev_stellarite_beam_button
    icon = GFX_fleet_action_button_behemoth_superweapon
    icon_selected = GFX_fleet_action_button_behemoth_superweapon
    activity_key = lev_stellarite_beam_activity_key
    context_menu_name = lev_stellarite_beam_menu_title
    click_to_act_key = lev_beam_click_to_act

    wait = no
    cooldown = 110
    slot = 2

    ai = {
        # This = 目标星系，From = 施放舰队
        weight = {
            base = 0
            # 星系内有敌对舰队才值得开火
            modifier = {
                weight = 20
                any_fleet_in_system = {
                    exists = owner
                    owner = { is_hostile_to = FROM.controller }
                }
            }
            # 封权须排在本块最后，上方是加法项
            inline_script = scripted_actions/lev_stellarite_absorb_guard
        }
    }
}

# 潜入恒星不设按钮：走吞噬恒星的灭星器链路（见 lev.107/108 与 leviathan_on_actions.txt）

# 自动吞噬：控制面动作，不出按钮，只给舰队自动化菜单用
lev_stellarite_auto_absorb = {
    user_scope = fleet
    scope = planet

    automation = {
        priority = 4
        default_on = no
        tooltip = lev_auto_tt_absorb
        options = {
            inline_script = scripted_actions/lev_stellarite_automation_options
        }
    }

    # 与 can_destroy_planet_with_LEV_STELLARITE_ABSORB_BEAM 同条件，免得选到灭星器会拒的目标
    possible = {
        custom_tooltip = {
            fail_text = lev_fail_not_star
            is_star = yes
        }
        custom_tooltip = {
            fail_text = lev_absorb_fail_no_energy
            from.owner = {
                check_variable = { which = lev_stellarite_energy value < 10 }
            }
        }
        custom_tooltip = {
            fail_text = lev_absorb_fail_edge
            NOT = { is_star_class = sc_m_giant }
        }
        custom_tooltip = {
            fail_text = lev_absorb_fail_multi_star
            NOR = { is_binary_star = yes  is_trinary_star = yes }
        }
        custom_tooltip = {
            fail_text = lev_absorb_fail_special_star
            NOR = {
                is_star_class = sc_black_hole
                is_star_class = sc_neutron_star
                is_star_class = sc_pulsar
                is_star_class = sc_toxoid_star
                is_star_class = sc_rift_star
            }
        }
    }

    # 控制面动作，不出按钮也不进右键菜单
    button_visible = { always = no }
    hide_from_context_menu = { always = yes }

    on_completed = on_lev_auto_absorb_completed
    tooltip = lev_stellarite_auto_absorb_tt
    icon = GFX_fleet_action_button_behemoth_superweapon
    icon_selected = GFX_fleet_action_button_behemoth_superweapon
    activity_key = lev_stellarite_auto_absorb_activity_key
    context_menu_name = lev_stellarite_auto_absorb_menu_title
    progress_activity_key = lev_stellarite_auto_absorb_progress_key

    wait = no
    required_progress = 5
    cooldown = 30

    ai = {
        # This = 目标恒星，From = 施放舰队
        weight = {
            base = 10
            # 订单在飞期间封权
            inline_script = scripted_actions/lev_stellarite_absorb_guard
            inline_script = scripted_actions/lev_stellarite_ai_filters
        }
        max_jumps = 8
        good_enough_weight = 10
    }
}
