# 重燃恒星（槽1）
lev_stellarite_ignite = {
    user_scope = fleet
    scope = planet

    possible = {
        custom_tooltip = {
            fail_text = lev_fail_not_star
            is_star = yes
        }
        custom_tooltip = {
            fail_text = lev_ignite_fail_no_energy
            from.owner = {
                check_variable = { which = lev_stellarite_energy value > 0 }
            }
        }
        custom_tooltip = {
            fail_text = lev_ignite_fail_edge
            NOT = { is_star_class = sc_b }
        }
        custom_tooltip = {
            fail_text = lev_ignite_fail_multi_star
            NOR = { is_binary_star = yes  is_trinary_star = yes }
        }
        custom_tooltip = {
            fail_text = lev_ignite_fail_special_star
            NOR = {
                is_star_class = sc_black_hole
                is_star_class = sc_neutron_star
                is_star_class = sc_pulsar
                is_star_class = sc_toxoid_star
                is_star_class = sc_rift_star
            }
        }
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_stellarite }
    }

    button_clickable = {
        custom_tooltip = {
            fail_text = lev_ignite_fail_no_energy
            owner = {
                check_variable = { which = lev_stellarite_energy value > 0 }
            }
        }
    }

    on_completed = on_lev_ignite_completed
    tooltip = lev_stellarite_ignite_button
    icon = GFX_fleet_action_button_arkship_stellar_ignition
    icon_selected = GFX_fleet_action_button_arkship_stellar_ignition
    activity_key = lev_stellarite_ignite_activity_key
    context_menu_name = lev_stellarite_ignite_menu_title
    progress_activity_key = lev_stellarite_ignite_progress_key

    begin_anim_state = "breath_begin"
    progress_anim_state = "breath_loop"
    finish_anim_state = "breath_end"

    wait = no
    required_progress = 30
    cooldown = 360
    slot = 1
}
