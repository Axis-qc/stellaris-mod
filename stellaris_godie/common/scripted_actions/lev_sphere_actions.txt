# 利维坦·无限神机 技能（槽 1/2/3）
# 本期技能无资源门槛：possible / button_clickable 直接 always = yes
# 不写 begin_anim_state / progress_anim_state / finish_anim_state，避免引用未注册动画

# 引力透镜（槽1）：自身武器伤害 +40%、射速 +15%
lev_sphere_gravitational_lens = {
    user_scope = fleet
    scope = self

    possible = {
        always = yes
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_sphere }
    }

    button_clickable = {
        always = yes
    }

    on_completed = on_lev_sphere_sk1_completed
    tooltip = lev_sphere_gravitational_lens_button
    icon = GFX_fleet_action_button_behemoth_damage
    icon_selected = GFX_fleet_action_button_behemoth_damage
    activity_key = lev_sphere_gravitational_lens_activity_key
    context_menu_name = lev_sphere_gravitational_lens_menu_title
    progress_activity_key = lev_sphere_gravitational_lens_progress_key

    wait = no
    required_progress = 5
    cooldown = 180
    slot = 1
}
