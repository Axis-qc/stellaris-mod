# 利维坦·天龙 技能（槽 1/2/3）
# 本期技能无资源门槛：possible / button_clickable 直接 always = yes
# 不写 begin_anim_state / progress_anim_state / finish_anim_state，避免引用未注册动画

# 潮汐吐息（槽1）：本星系敌舰武器伤害 -25%、射速 -15%
lev_aquatic_tide_breath = {
    user_scope = fleet
    scope = self

    possible = {
        always = yes
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_aquatic }
    }

    button_clickable = {
        always = yes
    }

    on_completed = on_lev_aquatic_sk1_completed
    tooltip = lev_aquatic_tide_breath_button
    icon = GFX_fleet_action_button_behemoth_damage
    icon_selected = GFX_fleet_action_button_behemoth_damage
    activity_key = lev_aquatic_tide_breath_activity_key
    context_menu_name = lev_aquatic_tide_breath_menu_title
    progress_activity_key = lev_aquatic_tide_breath_progress_key

    wait = no
    required_progress = 5
    cooldown = 180
    slot = 1
}
