# 利维坦·自动无畏战舰 技能（炮击）
# possible：This = 舰队自身、From = 施放舰队；button_clickable 是舰队作用域
# 本期无资源门槛，possible / button_clickable 直接放行

# 净化齐射（槽1）：自身武器伤害与射速强化
lev_dreadnought_volley = {
    user_scope = fleet
    scope = self

    possible = {
        always = yes
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_dreadnought }
    }

    button_clickable = {
        always = yes
    }

    on_completed = on_lev_dreadnought_sk1_completed
    tooltip = lev_dreadnought_volley_button
    icon = GFX_fleet_action_button_behemoth_damage
    icon_selected = GFX_fleet_action_button_behemoth_damage
    activity_key = lev_dreadnought_volley_activity_key
    context_menu_name = lev_dreadnought_volley_menu_title
    progress_activity_key = lev_dreadnought_volley_progress_key

    wait = no
    required_progress = 5
    cooldown = 180
    slot = 1
}
