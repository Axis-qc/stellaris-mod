# 利维坦行星支援：主动支援行动，slot 4，scope = planet
# possible 在目标行星作用域（国家用 from.controller），button_visible/clickable 在舰队作用域

# ── 噬星者：恒星暖流（目标行星能量岗位产出强化）──
lev_stellarite_support = {
	user_scope = fleet
	scope = planet

	possible = {
		custom_tooltip = {
			fail_text = lev_support_requires_own_colony_tt
			exists = owner
			exists = from.controller
			owner = { is_same_value = from.controller }
		}
		custom_tooltip = {
			fail_text = lev_support_no_conflict_tt
			NOT = { has_modifier = lev_orbit_stellarite_active }
		}
	}

	button_visible = {
		any_owned_ship = { is_ship_size = lev_stellarite }
	}

	button_clickable = {
		always = yes
	}

	# 自动托管整体禁用中
	# automation = {
	# 	priority = 5
	# 	default_on = no
	# 	tooltip = lev_auto_tt_support
	# }

	on_completed = on_lev_stellarite_support_completed
	tooltip = lev_stellarite_support_ship_button
	icon = GFX_fleet_action_button_arkship_stellar_ignition
	icon_selected = GFX_fleet_action_button_arkship_stellar_ignition
	activity_key = lev_stellarite_support_activity_key
	context_menu_name = lev_stellarite_support_menu_title
	progress_activity_key = lev_stellarite_support_progress_key

	wait = no
	required_progress = 5
	cooldown = 480
	slot = 4

	ai = {
		# This = 目标行星，From = 施放舰队
		weight = {
			base = 10
			# 订单在飞期间封权
			inline_script = scripted_actions/lev_stellarite_absorb_guard
			# 只支援自家殖民地
			modifier = {
				factor = 0
				NOT = { owner = { is_same_value = FROM.controller } }
			}
		}
		max_jumps = 6
		good_enough_weight = 10
	}
}
