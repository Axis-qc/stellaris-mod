# 通用：集结舰队至母舰（槽位1，三个核心共享）

br_vanguard_gather = {
    user_scope = fleet
    scope = self
    possible = {
        always = yes
    }
    button_visible = {
        owner = {
            OR = {
                is_country_type = br_player_pve
                is_country_type = br_player_pvp
            }
        }
        any_owned_ship = {
            is_ship_size = br_vanguard
        }
    }
    button_clickable = {
        custom_tooltip = {
            fail_text = br_hero_active_skill_fail_tt
            always = yes
        }
    }
    on_completed = on_br_gather_fleets_to_mothership
    tooltip = br_vanguard_skill_gather
    icon = GFX_fleet_action_button_paladin_recall
    icon_selected = GFX_fleet_action_button_paladin_recall
    activity_key = br_hero_gather_skill_activity_key
    context_menu_name = br_hero_gather_skill_menu_title
    progress_activity_key = br_hero_gather_skill_progress_activity_key

    wait = no
    cooldown = 90
    slot = 1
}
