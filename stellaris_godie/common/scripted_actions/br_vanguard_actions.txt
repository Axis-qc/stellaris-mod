# 先锋级英雄舰 scripted_actions
# 从 br_vanguard_auras.txt 内嵌定义迁移而来

# ─────────────────────────────────
# 通用：集结舰队至母舰（槽位1，三个核心共享）
# ─────────────────────────────────
br_vanguard_gather = {
    user_scope = fleet
    scope = self
    possible = {
        always = yes
    }
    button_visible = {
        OR = {
            has_global_flag = br_mode_pve
            has_global_flag = br_mode_pvp
        }
        owner = {
            has_country_flag = br_player_country
        }
        any_owned_ship = {
            is_ship_size = br_vanguard
        }
    }
    button_clickable = {
        custom_tooltip = {
            fail_text = br_hero_active_skill_fail_tt
            always = yes
        }
    }
    on_completed = on_br_gather_fleets_to_mothership
    tooltip = br_vanguard_skill_gather
    icon = GFX_fleet_action_button_paladin_recall
    icon_selected = GFX_fleet_action_button_paladin_recall
    activity_key = br_hero_gather_skill_activity_key
    context_menu_name = br_hero_gather_skill_menu_title
    progress_activity_key = br_hero_gather_skill_progress_activity_key

    wait = no
    cooldown = 90
    slot = 1
}

# 变形技能（共用槽位 4，靠船 flag 互斥，同一时刻只显示一个）
# 进入走 working → working_looping，退出走 stow → idle，见 br_vanguard_star_eater.asset。
br_vanguard_deploy = {
    user_scope = fleet
    scope = self

    possible = {
        always = yes
    }
    button_visible = {
        any_owned_ship = {
            is_ship_size = br_vanguard
            NOT = { has_ship_flag = br_vanguard_deployed }
        }
    }
    button_clickable = {
        custom_tooltip = {
            fail_text = br_vanguard_deploy_fail_tt
            always = yes
        }
    }

    on_completed = on_br_vanguard_deploy
    tooltip = br_vanguard_deploy_button
    icon = GFX_fleet_action_button_paladin_super_shield
    icon_selected = GFX_fleet_action_button_paladin_super_shield
    activity_key = br_vanguard_deploy_activity_key
    context_menu_name = br_vanguard_deploy_menu_title
    progress_activity_key = br_vanguard_deploy_progress_key

    begin_anim_state = "working"

    wait = no
    required_progress = 10
    slot = 4
}
