# 利维坦·拾荒机器人 技能（槽 1/2/3）
# 本期技能无资源门槛：possible / button_clickable 直接 always = yes
# 不写 begin_anim_state / progress_anim_state / finish_anim_state，避免引用未注册动画

# 碎甲重组（槽1）：自身护甲 +45%、护甲再生 +3/天
lev_scavenger_reassembly = {
    user_scope = fleet
    scope = self

    possible = {
        always = yes
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_scavenger }
    }

    button_clickable = {
        always = yes
    }

    on_completed = on_lev_scavenger_sk1_completed
    tooltip = lev_scavenger_reassembly_button
    icon = GFX_fleet_action_button_paladin_field_repairs
    icon_selected = GFX_fleet_action_button_paladin_field_repairs
    activity_key = lev_scavenger_reassembly_activity_key
    context_menu_name = lev_scavenger_reassembly_menu_title
    progress_activity_key = lev_scavenger_reassembly_progress_key

    wait = no
    required_progress = 5
    cooldown = 180
    slot = 1
}
