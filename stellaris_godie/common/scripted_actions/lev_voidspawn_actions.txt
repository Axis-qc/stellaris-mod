# 利维坦·虚空孳孽 技能（槽 1/2/3）
# 本期技能无资源门槛：possible / button_clickable 直接 always = yes
# 不写 begin_anim_state / progress_anim_state / finish_anim_state，避免引用未注册动画

# 虚空触须（槽1）：本星系敌舰武器伤害 -25%、护盾回复 -40%
lev_voidspawn_void_tentacle = {
    user_scope = fleet
    scope = self

    possible = {
        always = yes
    }

    button_visible = {
        any_owned_ship = { is_ship_size = lev_voidspawn }
    }

    button_clickable = {
        always = yes
    }

    on_completed = on_lev_voidspawn_sk1_completed
    tooltip = lev_voidspawn_void_tentacle_button
    icon = GFX_fleet_action_button_behemoth_damage
    icon_selected = GFX_fleet_action_button_behemoth_damage
    activity_key = lev_voidspawn_void_tentacle_activity_key
    context_menu_name = lev_voidspawn_void_tentacle_menu_title
    progress_activity_key = lev_voidspawn_void_tentacle_progress_key

    wait = no
    required_progress = 5
    cooldown = 240
    slot = 1
}
