# ----------------------------
# 方舟舰勘探
# 作用域: 舰队→星球
# 功能: 复侧行政舰的勘探命令,需要领袖在舰上
# 自动化: 优先级1,默认开启
# ----------------------------
br_arkship_survey = {
	user_scope = fleet
	scope = planet

	# 行为模式: 复侧行政舰的勘探命令
	specific_behavior = scripted_survey_order

	automation_priority = 1
	automation_default_on = yes
	slot = 1
	icon = GFX_fleet_order_button_survey_small
	icon_selected = GFX_fleet_order_button_survey_small_selected

	hide_from_context_menu = {
		is_surveyed = {
			who = from.controller
			status = yes
		}
	}

	possible = {
		custom_tooltip = {
			fail_text = arkship_has_leader_tt_fail
			success_text = arkship_has_leader_tt_success
			exists = from.leader
		}
		is_surveyed = {
			who = from.controller
			status = no
		}
		NOT = {
			is_being_surveyed = { who = from }
		}
	}

	button_visible = {
		always = yes
	}

	button_clickable = {
		custom_tooltip = {
			fail_text = arkship_has_leader_tt_fail
			success_text = arkship_has_leader_tt_success
			exists = leader
		}
	}

	ai = {
		#This - Target Planet
		#From - Acting Fleet
		weight = {
			base = @ai_action_weight_10
			inline_script = {
				script = scripted_actions/ai_distance_modifiers
				LOCAL_SYSTEM_CHECK = "any_system_planet = { is_same_value = ROOT }"
			}
		}
		max_jumps = @ai_action_weight_5
		good_enough_weight = @ai_action_weight_30
		ai_minister = interior
	}

	context_menu_name = br_arkship_survey
}

# ----------------------------
# 方舟舰深度扫描
# 作用域: 舰队→星球
# 功能: 对星球进行深度扫描,发现隐藏资源和异常
# 前置条件: 需要领袖、传感器阵列/系统扫描仪(或T2+方舟舰)
# 排除: 已有异常、宜居世界(需自然博物学家特质)、人造/环形世界
# 自动化: 优先级3
# ----------------------------
br_arkship_deep_scan = {
	user_scope = fleet
	scope = planet

	automation_priority = 3
	on_completed = on_br_arkship_deep_scan

			success_text = arkship_has_leader_tt_success

		}

			limit = {
				from = {
					OR = {
						has_starbase_size = br_science_arkship_tier_1
						has_starbase_size = br_science_arkship_tier_2

					success_text = has_arkship_system_scanner_pass_tt

						has_starbase_module = br_arkship_system_scanner

				}

				custom_tooltip = {
					fail_text = has_arkship_sensor_array_fail_tt
					success_text = has_arkship_sensor_array_pass_tt
					from.starbase? = {
						has_starbase_module = br_arkship_sensor_array

				}

					fail_text = has_arkship_system_scanner_fail_tt
					success_text = has_arkship_system_scanner_pass_tt
					from.starbase? = {
						has_starbase_module = br_arkship_system_scanner

					has_leader_flag = nomadic_naturalist

			fail_text = arkship_has_leader_tt_fail
			success_text = arkship_has_leader_tt_success
			exists = leader

					has_starbase_size = br_science_arkship_tier_3
				}
			}
			OR = {
				custom_tooltip = {
					fail_text = is_tier_2_or_3_arkship_fail_tt
					success_text = is_tier_2_or_3_arkship_pass_tt
					NOR = {
						has_starbase_size = br_civilian_arkship_tier_1
						has_starbase_size = br_science_arkship_tier_1
						has_starbase_size = br_military_arkship_tier_1
					}
				}
				custom_tooltip = {
					fail_text = has_arkship_sensor_array_fail_tt
					success_text = has_arkship_sensor_array_pass_tt
					starbase? = {
						has_starbase_module = br_arkship_sensor_array
					}
				}
				custom_tooltip = {
					fail_text = has_arkship_system_scanner_fail_tt
					success_text = has_arkship_system_scanner_pass_tt
					starbase? = {
						has_starbase_module = br_arkship_system_scanner
					}
				}
			}
		}
		else = {
			OR = {
				custom_tooltip = {
					fail_text = has_arkship_sensor_array_fail_tt
					success_text = has_arkship_sensor_array_pass_tt
					starbase? = {
						has_starbase_module = br_arkship_sensor_array
					}
				}
				custom_tooltip = {
					fail_text = has_arkship_system_scanner_fail_tt
					success_text = has_arkship_system_scanner_pass_tt
					starbase? = {
						has_starbase_module = br_arkship_system_scanner
					}
				}
			}
		}
	}

	activity_key = arkship_deep_scan_activity_key
	context_menu_name = arkship_deep_scan
	required_progress = value:arkship_deep_scan_required_progress_value
	required_progress_key = arkship_deep_scan_required_progress_key
	progress_activity_key = arkship_deep_scan_progress_activity_key
	cooldown = @arkship_deep_scan_cooldown

	uses_vfx_type = survey_effect

	cost = {
		energy = @arkship_deep_scan_cost
	}
	slot = 2

	ai = {
		#This - Target Planet
		#From - Acting Fleet
		weight = {
			base = @ai_action_weight_15
			modifier = {
				factor = 0 #cannot afford
				FROM.controller = {
					resource_stockpile_compare = {
						resource = energy
						value < @arkship_deep_scan_cost
						mult = 1.5
					}
				}
			}
			inline_script = {
				script = scripted_actions/ai_distance_modifiers
				LOCAL_SYSTEM_CHECK = "any_system_planet = { is_same_value = ROOT }"
			}
			modifier = {
				factor = @ai_action_weight_100 #strongly prioritize planets that are targets of a deep scan contract
				is_point_of_interest = {
					id = deep_scan_poi@this
					event_chain = contract_deep_scan_chain
					owner = FROM.controller
				}
			}
			modifier = {
				factor = @ai_action_weight_factor_0_1 #already deep scanned and not a contract target
				is_variable_set = times_deep_scanned
				NOT = {
					is_point_of_interest = {
						id = deep_scan_poi@this
						event_chain = contract_deep_scan_chain
						owner = FROM.controller
					}
				}
			}
		}
		max_jumps = @ai_action_weight_5
		good_enough_weight = @ai_action_weight_30
		ai_minister = interior
	}
}

# ----------------------------
# 方舟舰采集资源
# 作用域: 舰队→星球
# 功能: 从未被开采的星球上采集资源
# 排除: 已被开采、有异常的已勘探星球
# 自动化: 优先级2,默认开启
# ----------------------------
br_arkship_harvest_resources = {
	user_scope = fleet
	scope = planet

	automation_priority = 2
	automation_default_on = yes
	on_completed = on_br_harvest_resources
	tooltip = arkship_harvest_resources_ship_button
	icon = GFX_fleet_action_button_arkship_harvest_resources
	icon_selected = GFX_fleet_action_button_arkship_harvest_resources

	possible = {
		NOT = {
			has_modifier = "harvested_resources_mining"
		}
		if = {
			limit = {
				is_surveyed = {
					who = from
					status = yes
				}
				has_anomaly = yes
			}
			custom_tooltip = {
				fail_text = arkship_target_has_anomaly_fail_tt
				always = no
			}
		}
		is_valid_harvest_target = yes
	}

	button_visible = {
		always = yes
	}

	button_clickable = {
		always = yes
	}

	activity_key = arkship_harvest_resources_activity_key
	context_menu_name = arkship_harvest_resources
	required_progress = value:arkship_harvest_resources_required_progress_value
	progress_activity_key = arkship_harvest_resources_progress_activity_key
	required_progress_key = arkship_harvest_resources_required_progress_key
	cooldown = @harvest_resources_cooldown
	category = harvest_resources
	category_limit = 1

	uses_vfx_type = harvest_effect

	cost = {
		energy = @harvest_resources_cost
	}
	slot = 3
	overrides_gained_resources = yes
	completion_message_type = TOAST_HARVEST_RESOURCES_COMPLETE
	completion_message_localization = TOAST_HARVEST_RESOURCES_COMPLETE_DESC

	ai = {
		#This - Target Planet
		#From - Acting Fleet
		weight = {
			base = @ai_action_weight_15
			inline_script = {
				script = scripted_actions/ai_distance_modifiers
				LOCAL_SYSTEM_CHECK = "any_system_planet = { is_same_value = ROOT }"
			}
			# In Operational Reserves crisis (Stage 1), prioritize planets that yield energy or minerals
			modifier = {
				factor = @ai_low_resource_harvest_factor
				FROM.controller = {
					any_situation = {
						is_situation_type = situation_nomad_economy
						current_stage = situation_nomad_economy_stage_1_struggle
					}
				}
				OR = {
					is_planet_class = pc_ice_asteroid		# primary energy, secondary exotic gases
					is_planet_class = pc_asteroid			# primary minerals, secondary volatile motes
					is_planet_class = pc_barren				# primary minerals, secondary minerals
					is_planet_class = pc_barren_cold		# primary minerals, secondary minerals
					is_planet_class = pc_broken				# primary minerals, secondary alloys
					is_planet_class = pc_shattered			# primary minerals, secondary alloys
					is_planet_class = pc_shattered_2		# primary minerals, secondary alloys
					has_resource = {						# energy deposits
						type = energy
						amount > 0
					}
					has_resource = {						# mineral deposits
						type = minerals
						amount > 0
					}
				}
			}
			inline_script = scripted_actions/ai_harvest_resources_modifiers
		}
		max_jumps = @ai_action_weight_5
		good_enough_weight = @ai_action_weight_30
		ai_minister = interior
	}
}

br_arkship_stellar_ignition = {
	user_scope = fleet
	scope = planet

	on_completed = on_br_arkship_stellar_ignition

	icon_selected = GFX_fleet_action_button_arkship_stellar_ignition

	possible = {
		custom_tooltip = {
			fail_text = has_arkship_stellar_igniter_fail_tt
			success_text = has_arkship_stellar_igniter_pass_tt
			from.starbase? = {
				has_starbase_module = br_arkship_stellar_igniter
			}
		}
		has_anomaly = no
		# If you have a Stellar Ignition contract in this system, only allow that gas giant
		custom_tooltip = {
			fail_text = arkship_target_contract_target_fail_tt
			OR = {
				# Either there's no contract POI in this system...
				NOT = {
					solar_system = {
						any_system_planet = {
							is_point_of_interest = {
								id = stellar_ignition_poi@this
								event_chain = contract_stellar_ignition_chain
								owner = from.owner
							}
						}
					}
				}
				# ... or this planet is the contract POI
				is_point_of_interest = {
					id = stellar_ignition_poi@this
					event_chain = contract_stellar_ignition_chain
					owner = from.owner
				}
			}
		}
		# Either the target is a Gas Giant..
		if = {
			limit = {
				is_planet_class = pc_gas_giant
				is_moon = no
			}
			# If a moon is a colony of a space-faring empire, we must be hostile to them.
			custom_tooltip = {
				fail_text = arkship_target_moon_is_colony_fail_tt
				count_moon = {
					count = 0
					limit = {
						exists = space_owner
						exists = colony
						colony.owner = {
							is_primitive = no
						}
						space_owner = {
							NOR = {
								is_at_war_with = from.owner
								is_country_type = swarm
								is_country_type = extradimensional
								is_country_type = extradimensional_2
								is_country_type = extradimensional_3
								is_country_type = ai_empire
							}
						}
					}
				}
			}
			# If a moon is home to a pre-FTL civilization
			# it must either be in hostile or unowned space...
			custom_tooltip = {
				fail_text = arkship_target_moon_is_pre_ftl_colony_fail_tt
				count_moon = {
					count = 0
					limit = {
						exists = space_owner
						exists = colony
						colony.owner = {
							is_primitive = yes
						}
						space_owner = {
							OR = {
								is_at_war_with = from.owner
								is_country_type = swarm
								is_country_type = extradimensional
								is_country_type = extradimensional_2
								is_country_type = extradimensional_3
								is_country_type = ai_empire
							}
						}
					}
				}
			}
			# And our pre-FTL policy must allow thier extermination
			custom_tooltip = {
				fail_text = arkship_target_pre_ftl_planet_interference_fail_tt
				if = {
					limit = {
						from.owner = {
							NOT = { has_policy_flag = interference_aggressive }
						}
					}
					count_moon = {
						count = 0
						limit = {
							exists = colony
							colony.owner = {
								is_primitive = yes
							}
						}
					}
				}
			}
		}
		# Or a Star...
		else_if = {
			limit = {
				is_star = yes
			}
			NOT = {
				has_modifier = stellar_ignition_mod
			}
			custom_tooltip = {
				fail_text = arkship_target_contract_target_fail_tt
				OR = {
					# Either there's no contract POI in this system...
					NOT = {
						solar_system = {
							any_system_planet = {
								is_point_of_interest = {
									id = stellar_ignition_poi@this
									event_chain = contract_stellar_ignition_chain
									owner = from.owner
								}
							}
						}
					}
					# ... or this planet is the contract POI
					is_point_of_interest = {
						id = stellar_ignition_poi@this
						event_chain = contract_stellar_ignition_chain
						owner = from.owner
					}
				}
			}
		}
		# And we inform the player if otherwise
		else = {
			custom_tooltip = {
				fail_text = arkship_target_not_gas_giant_or_star_fail_tt
				always = no
			}
		}
	}

	button_visible = {
		custom_tooltip = {
			fail_text = has_arkship_stellar_igniter_fail_tt
			success_text = has_arkship_stellar_igniter_pass_tt
			starbase? = {
				has_starbase_module = br_arkship_stellar_igniter
			}
		}
	}

	button_clickable = {
		custom_tooltip = {
			fail_text = has_arkship_stellar_igniter_fail_tt
			success_text = has_arkship_stellar_igniter_pass_tt
			starbase? = {
				has_starbase_module = br_arkship_stellar_igniter
			}
		}
	}

	activity_key = arkship_stellar_ignition_activity_key
	context_menu_name = arkship_stellar_ignition
	required_progress = value:arkship_stellar_ignition_required_progress_value
	progress_activity_key = arkship_stellar_ignition_progress_activity_key
	required_progress_key = arkship_stellar_ignition_required_progress_key
	cooldown = @arkship_stellar_ignition_cooldown
	category = stellar_ignition
	category_limit = 1

	cost = {

	}

# ----------------------------
# 收集中转站库存
# 作用域: 舰队→星系
# 功能: 从中转站星堡收集存储的资源,支持远程收集网络
# 适用于方舟舰和后勤舰
# 自动化: 优先级1,默认开启
# ----------------------------
# 从中转站收集库存(方舟舰和后勤舰)
br_collect_waystation_stockpile = {
	user_scope = fleet
	scope = galactic_object

	also_automate = { logistic_drop_off_stockpile }
	automation_priority = 1
	automation_default_on = yes

	possible = {
		# This - Target System
		# From - Acting Fleet
		any_starbase_in_system = {
			is_waystation_starbase = yes
			owner? = {
				is_same_value = from.controller
			}
			OR = {
				used_stockpile_capacity_percentage > 0
				AND = {
					waystation_has_remote_collection = yes
					waystation_network_stockpile_used > 0
				}
			}
		}
		FROM = {
			OR = {
				is_arkship_fleet = yes
				is_br_civilian_arkship = yes
				any_owned_ship = {
					used_stockpile_capacity_percentage < 1
				}
			}
		}
	}

	context_menu_name = collect_waystation_stockpile

	hide_from_context_menu = {
		always = yes
	}

	button_visible = {
		always = no
	}

	button_clickable = {
		always = yes
	}

	activity_key = collect_waystation_stockpile
	required_progress = @collect_waystation_stockpile_required_progress
	required_progress_key = collect_waystation_stockpile_key
	progress_activity_key = collect_waystation_stockpile_progress_activity_key

	cooldown = @collect_waystation_stockpile_cooldown
	category = collect_waystation_stockpile
	category_limit = 1

	ai = {
		# This - Target System
		# From - Acting Fleet
		weight = {
			base = @ai_action_weight_5 # network has no remote collection, visit individual waystations
			modifier = {
				factor = @ai_action_weight_10 # has remote collection and network has some resources in stockpile
				any_starbase_in_system = {
					is_waystation_starbase = yes
					owner = {
						is_same_value = from.controller
					}
					waystation_has_remote_collection = yes
					waystation_network_stockpile_used > 0
				}
			}
			modifier = {
				factor = @ai_action_weight_20 # local stockpile above threshold and network has no remote collection
				any_starbase_in_system = {
					is_waystation_starbase = yes
					owner = {
						is_same_value = from.controller
					}
					waystation_network_has_remote_collection = no
					used_stockpile_capacity_percentage >= 0.5
				}
			}
			modifier = {
				factor = @ai_action_weight_30 # has remote collection and network has stockpile above threshold
				any_starbase_in_system = {
					is_waystation_starbase = yes
					owner = {
						is_same_value = from.controller
					}
					waystation_has_remote_collection = yes
					waystation_network_stockpile_used >= 0.5
				}
			}
			modifier = {
				factor = @ai_action_weight_100 # always collect local stockpile before moving on if no hub elsewhere to go to
				FROM = {
					solar_system? = {
						is_same_value = ROOT
					}
				}
				any_starbase_in_system = {
					is_waystation_starbase = yes
					owner = {
						is_same_value = from.controller
					}
					used_stockpile_capacity_percentage > 0
					OR = {
						waystation_has_remote_collection = yes			#logistic hub in this system
						waystation_network_has_remote_collection = no	#no logistic hub anywhere in the network
					}
				}
			}
		}
		target_owned_starbases = yes
		ai_minister = interior
	}
}
