				}
			}
			modifier = {
				factor = @ai_action_weight_factor_0_1 #already deep scanned and not a contract target

					is_point_of_interest = {
						id = deep_scan_poi@this

					}
				}
			}
		}
		max_jumps = @ai_action_weight_5
		good_enough_weight = @ai_action_weight_30
		ai_minister = interior
	}

			}
		}
		is_valid_harvest_target = yes
	}

	button_visible = {
		always = yes
	}

	button_clickable = {
		always = yes
	}

	activity_key = br_arkship_harvest_resources_activity_key
	context_menu_name = br_arkship_harvest_resources
	required_progress = value:br_arkship_harvest_resources_required_progress_value
	progress_activity_key = br_arkship_harvest_resources_progress_activity_key
	required_progress_key = br_arkship_harvest_resources_required_progress_key
	cooldown = @harvest_resources_cooldown
	category = harvest_resources
	category_limit = 1

	uses_vfx_type = harvest_effect

	cost = {
		energy = @harvest_resources_cost
	}
	slot = 3
	overrides_gained_resources = yes
	completion_message_type = TOAST_HARVEST_RESOURCES_COMPLETE
	completion_message_localization = TOAST_HARVEST_RESOURCES_COMPLETE_DESC

	ai = {
		#This - Target Planet
		#From - Acting Fleet
		weight = {
			base = @ai_action_weight_15
			inline_script = {
				script = scripted_actions/ai_distance_modifiers
				LOCAL_SYSTEM_CHECK = "any_system_planet = { is_same_value = ROOT }"
			}
			# In Operational Reserves crisis (Stage 1), prioritize planets that yield energy or minerals
			modifier = {
				factor = @ai_low_resource_harvest_factor
				FROM.controller = {
					any_situation = {
						is_situation_type = situation_nomad_economy
						current_stage = situation_nomad_economy_stage_1_struggle
					}
				}
				OR = {
					is_planet_class = pc_ice_asteroid		# primary energy, secondary exotic gases
					is_planet_class = pc_asteroid			# primary minerals, secondary volatile motes
					is_planet_class = pc_barren				# primary minerals, secondary minerals
					is_planet_class = pc_barren_cold		# primary minerals, secondary minerals
					is_planet_class = pc_broken				# primary minerals, secondary alloys
					is_planet_class = pc_shattered			# primary minerals, secondary alloys
					is_planet_class = pc_shattered_2		# primary minerals, secondary alloys
					has_resource = {						# energy deposits
						type = energy
						amount > 0
					}
					has_resource = {						# mineral deposits
						type = minerals
						amount > 0
					}
				}
			}
			inline_script = scripted_actions/ai_harvest_resources_modifiers
		}
		max_jumps = @ai_action_weight_5
		good_enough_weight = @ai_action_weight_30
		ai_minister = interior
	}
}

				}
			}
			# In Operational Reserves crisis (Stage 1), prioritize planets that yield energy or minerals
			modifier = {
				factor = @ai_low_resource_harvest_factor
				FROM.controller = {
					any_situation = {
						is_situation_type = situation_nomad_economy
						current_stage = situation_nomad_economy_stage_1_struggle
					}
				}
				OR = {
					is_planet_class = pc_ice_asteroid		# primary energy, secondary exotic gases
					is_planet_class = pc_asteroid			# primary minerals, secondary volatile motes
					is_planet_class = pc_barren				# primary minerals, secondary minerals
					is_planet_class = pc_barren_cold		# primary minerals, secondary minerals
					is_planet_class = pc_broken				# primary minerals, secondary alloys
					is_planet_class = pc_shattered			# primary minerals, secondary alloys
					is_planet_class = pc_shattered_2		# primary minerals, secondary alloys
					has_resource = {						# energy deposits
						type = energy
						amount > 0
					}
					has_resource = {						# mineral deposits
						type = minerals
						amount > 0
					}
				}
			}
			inline_script = scripted_actions/ai_harvest_resources_modifiers
		}
		max_jumps = @ai_action_weight_10
		good_enough_weight = @ai_action_weight_100
		ai_minister = interior
	}
}

# Collect Stockpile from a Waystation for Arkships and Logistic Ships
collect_waystation_stockpile = {
	user_scope = fleet
	scope = galactic_object

	also_automate = { logistic_drop_off_stockpile }
	automation_priority = 1
	automation_default_on = yes

	possible = {
		# This - Target System
		# From - Acting Fleet
		any_starbase_in_system = {
			is_waystation_starbase = yes
			owner? = {
				is_same_value = from.controller
			}
			OR = {
				used_stockpile_capacity_percentage > 0
				AND = {
					waystation_has_remote_collection = yes
					waystation_network_stockpile_used > 0
				}
			}
		}
		FROM = {
			OR = {
				is_br_arkship_fleet = yes
				any_owned_ship = {
					used_stockpile_capacity_percentage < 1
				}
			}
		}
	}

	context_menu_name = collect_waystation_stockpile

	hide_from_context_menu = {
		always = yes
	}

	button_visible = {
		always = no
	}

	button_clickable = {
		always = yes
	}

	activity_key = collect_waystation_stockpile
	required_progress = @collect_waystation_stockpile_required_progress
	required_progress_key = collect_waystation_stockpile_key
	progress_activity_key = collect_waystation_stockpile_progress_activity_key

	cooldown = @collect_waystation_stockpile_cooldown
	category = collect_waystation_stockpile
	category_limit = 1

	ai = {
		# This - Target System
		# From - Acting Fleet
		weight = {
			base = @ai_action_weight_5 # network has no remote collection, visit individual waystations
			modifier = {
				factor = @ai_action_weight_10 # has remote collection and network has some resources in stockpile
				any_starbase_in_system = {
					is_waystation_starbase = yes
					owner = {
						is_same_value = from.controller
					}
					waystation_has_remote_collection = yes
					waystation_network_stockpile_used > 0
				}
			}
			modifier = {
				factor = @ai_action_weight_20 # local stockpile above threshold and network has no remote collection
				any_starbase_in_system = {
					is_waystation_starbase = yes
					owner = {
						is_same_value = from.controller
					}
					waystation_network_has_remote_collection = no
					used_stockpile_capacity_percentage >= 0.5
				}
			}
			modifier = {
				factor = @ai_action_weight_30 # has remote collection and network has stockpile above threshold
				any_starbase_in_system = {
					is_waystation_starbase = yes
					owner = {
						is_same_value = from.controller
					}
					waystation_has_remote_collection = yes
					waystation_network_stockpile_used >= 0.5
				}
			}
			modifier = {
				factor = @ai_action_weight_100 # always collect local stockpile before moving on if no hub elsewhere to go to
				FROM = {
					solar_system? = {
						is_same_value = ROOT
					}
				}
				any_starbase_in_system = {
					is_waystation_starbase = yes
					owner = {
						is_same_value = from.controller
					}
					used_stockpile_capacity_percentage > 0
					OR = {
						waystation_has_remote_collection = yes			#logistic hub in this system
						waystation_network_has_remote_collection = no	#no logistic hub anywhere in the network
					}
				}
			}
		}
		target_owned_starbases = yes
		ai_minister = interior
	}
}
