# 英雄舰等级/经验显示。作用域：舰船（Root），读取 owner 国家变量。
defined_text = {
    name = GetBRHeroXPState
    random = no
    text = {
        localization_key = BR_HERO_XP_MAX
        trigger = { owner = { check_variable = { which = br_hero_level value >= 12 } } }
    }
    text = {
        localization_key = BR_HERO_XP_STATE_value
        trigger = { owner = { has_country_flag = br_hero_level_initialized } }
    }
    default = BR_HERO_XP_NO_DATA
}
