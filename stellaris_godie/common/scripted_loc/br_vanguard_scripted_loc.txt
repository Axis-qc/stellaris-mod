# ========================
# CD 动态显示
# 作用域：舰船，按核心类型匹配对应 CD 文本
# ========================

defined_text = {
    name = GetBRVanguardActiveCD
    random = no

    text = {
        localization_key = br_vanguard_active_cd_power
        trigger = {
            br_has_power_vanguard = yes
            owner = { br_vanguard_power_skill_cd_running = yes }
        }
    }
    text = {
        localization_key = br_vanguard_active_cd_defense
        trigger = {
            br_has_defense_vanguard = yes
            owner = { br_vanguard_defense_skill_cd_running = yes }
        }
    }
    text = {
        localization_key = br_vanguard_active_cd_support
        trigger = {
            br_has_support_vanguard = yes
            owner = { br_vanguard_support_skill_cd_running = yes }
        }
    }
    text = {
        localization_key = br_vanguard_active_cd_ready
        trigger = {
            OR = {
                br_has_power_vanguard = yes
                br_has_defense_vanguard = yes
                br_has_support_vanguard = yes
            }
        }
    }
    default = ""
}

# 按已解锁技能匹配对应的菜单名
# ========================

    name = GetBRVanguardContextMenu
    random = no

    text = {
        localization_key = br_vanguard_skill_locked
        trigger = {
            owner = {
                NOR = {
                    has_country_flag = br_skill_phase_shift_unlocked

                    has_country_flag = br_skill_weakness_scan_unlocked
                }
            }
        }
    }
    text = {
        localization_key = br_context_menu_overclock
        trigger = { owner = { has_country_flag = br_skill_energy_overclock_unlocked } }
    }
    text = {
        localization_key = br_context_menu_phase_shift
        trigger = { owner = { has_country_flag = br_skill_phase_shift_unlocked } }
    }
    text = {
        localization_key = br_context_menu_scan
        trigger = { owner = { has_country_flag = br_skill_weakness_scan_unlocked } }
    }
    text = {
        localization_key = br_context_menu_fullfire
        trigger = { owner = { has_country_flag = br_skill_fullfire_unlocked } }
    }
    text = {
        localization_key = br_context_menu_final_volley
        trigger = { owner = { has_country_flag = br_skill_final_volley_unlocked } }
    }
    default = br_hero_active_skill_menu_title
}
