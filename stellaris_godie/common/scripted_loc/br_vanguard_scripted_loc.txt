# ========================
# CD 动态显示
# 作用域：舰船，按核心类型匹配对应 CD 文本
# ========================

defined_text = {
    name = GetBRVanguardActiveCD
    random = no

    text = {
        localization_key = br_vanguard_active_cd_power
        trigger = {
            br_has_power_vanguard = yes
            owner = { br_vanguard_power_skill_cd_running = yes }
        }
    }
    text = {
        localization_key = br_vanguard_active_cd_defense
        trigger = {
            br_has_defense_vanguard = yes
            owner = { br_vanguard_defense_skill_cd_running = yes }
        }
    }
    text = {
        localization_key = br_vanguard_active_cd_carrier
        trigger = {
            br_has_carrier_vanguard = yes
            owner = { br_vanguard_carrier_skill_cd_running = yes }
        }
    }
    text = {
        localization_key = br_vanguard_active_cd_ready
        trigger = {
            OR = {
                br_has_power_vanguard = yes
                br_has_defense_vanguard = yes
                br_has_carrier_vanguard = yes
            }
        }
    }
    default = ""
}
