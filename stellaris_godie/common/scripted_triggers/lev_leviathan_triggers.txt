# 利维坦成长值系统 — 触发器
# ============================================
# 新增一只利维坦时，只需在 lev_is_leviathan_ship 的 OR 里追加一行 is_ship_size，
# 战斗成长（lev.20）即自动对该利维坦生效，无需改动其它文件。

# 是否为利维坦英雄舰（ship 作用域）
lev_is_leviathan_ship = {
    OR = {
        is_ship_size = lev_stellarite
        is_ship_size = lev_drake
        is_ship_size = lev_shard
        is_ship_size = lev_wraith
        is_ship_size = lev_horror
        is_ship_size = lev_toxic_god
        is_ship_size = lev_dreadnought
        is_ship_size = lev_scavenger
        is_ship_size = lev_sphere
        is_ship_size = lev_aquatic
        is_ship_size = lev_voidspawn
        is_ship_size = lev_fortress
        is_ship_size = lev_hive
        is_ship_size = lev_tiyanki
    }
}

# 舰队是否含拥有行星支援技能的利维坦（fleet 作用域，6 只支援型）
# 军事型（drake/shard/horror/dreadnought/voidspawn）无支援技能，不进列表
lev_has_support_leviathan = {
    any_owned_ship = {
        OR = {
            is_ship_size = lev_stellarite
            is_ship_size = lev_sphere
            is_ship_size = lev_wraith
            is_ship_size = lev_scavenger
            is_ship_size = lev_toxic_god
            is_ship_size = lev_aquatic
        }
    }
}
