#火力核心主动CD 作用域：国家
br_vanguard_power_skill_cd_ready = {
	OR = {
		NOT = {
			is_variable_set = br_vanguard_power_skill_cd
		}
		check_variable = {
			which = br_vanguard_power_skill_cd
			value <= 0
		}
	}
}
