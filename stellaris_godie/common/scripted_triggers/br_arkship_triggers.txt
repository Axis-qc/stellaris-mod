# 方舟舰星港类型检查触发器
# 用于限制建筑和模块只能在指定方舟舰上使用
is_br_civilian_arkship = {
	OR = {
		has_starbase_size = br_civilian_arkship_tier_1
		has_starbase_size = br_civilian_arkship_tier_2
		has_starbase_size = br_civilian_arkship_tier_3
		has_starbase_size = br_science_arkship_tier_1
		has_starbase_size = br_science_arkship_tier_2
		has_starbase_size = br_science_arkship_tier_3
		has_starbase_size = br_military_arkship_tier_1
		has_starbase_size = br_military_arkship_tier_2
		has_starbase_size = br_military_arkship_tier_3
	}
}
