# =========================================================================
# PVE 年度裂隙刷敌链
# 调用链: br.920(on_yearly_pulse_country) → br_spawn_rift_fleet
#          → br_crisis_small/medium/large_spawn
# 每年在 BOSS 位置刷 1 支 + 每道虚空裂隙(br_boss_void_rift)各刷 1 支
# 数量基于 br_total_intensity：small=/100, medium=/200(>=200), large=/400(>=400)
# 校准目标（2026-08-26）：intensity=500 时单支约 50 艘
# =========================================================================

br_spawn_rift_fleet = {
	#复制变量进行检查
	set_variable = { which = br_total_intensity_effect value = event_target:br_pve_enemy_country.br_total_intensity }
    # small = intensity / 100
    set_variable = { which = br_crisis_small value = event_target:br_pve_enemy_country.br_total_intensity }
    divide_variable = { which = br_crisis_small value = 100 }
    # medium = intensity / 200（intensity >= 200 才触发）
    if = {
        limit = { check_variable = { which = br_total_intensity_effect value >= 200 } }
        set_variable = { which = br_crisis_medium value = event_target:br_pve_enemy_country.br_total_intensity }
        divide_variable = { which = br_crisis_medium value = 200 }
    }
    else = {
        set_variable = { which = br_crisis_medium value = 0 }
    }
    # large = intensity / 400（intensity >= 400 才触发）
    if = {
        limit = { check_variable = { which = br_total_intensity_effect value >= 400 } }
        set_variable = { which = br_crisis_large value = event_target:br_pve_enemy_country.br_total_intensity }
        divide_variable = { which = br_crisis_large value = 400 }
    }
    else = {
        set_variable = { which = br_crisis_large value = 0 }
    }
    # 生成三阶段混编天灾舰队（每种随机编队类型）
    br_crisis_small_spawn = yes
    br_crisis_medium_spawn = yes
    br_crisis_large_spawn = yes
    # 清理临时变量
    clear_variable = br_crisis_small
    clear_variable = br_crisis_medium
    clear_variable = br_crisis_large
}
