# 刷 N 艘指定设计（随机二选一）
# $COUNT$ = 艘数, $DESIGN_A$ / $DESIGN_B$ = 完整设计名
br_spawn_n_ships = {
    while = {
        count = $COUNT$
        random_list = {
            50 = { create_ship = { name = random design = $DESIGN_A$ } }
            50 = { create_ship = { name = random design = $DESIGN_B$ } }
        }
    }
}

# 纯编队效果 — 总容量 = power_level × 8，全出一种船
# ============================================================

# 纯护卫：容量 ÷ 1 = level × 8 艘
br_spawn_pure_corvette = {
    set_variable = { which = br_reinf_count value = root.br_power_level }
    multiply_variable = { which = br_reinf_count value = 8 }
    while = {
        count = br_reinf_count
        create_ship = { name = random design = event_target:br_player_corvette }
    }
    clear_variable = br_reinf_count
}

# 纯驱逐：容量 ÷ 2 = level × 4 艘，需 level≥3
br_spawn_pure_destroyer = {
    if = {
        limit = { root = { check_variable = { which = br_power_level value >= 3 } } }
        set_variable = { which = br_reinf_count value = root.br_power_level }
        multiply_variable = { which = br_reinf_count value = 4 }
        while = {
            count = br_reinf_count
            create_ship = { name = random design = event_target:br_player_destroyer }
        }
        clear_variable = br_reinf_count
    }
    else = { br_spawn_mixed = yes }
}

# 混合编队（原增援逻辑）
# ============================================================

br_spawn_mixed = {
    # 护卫
    set_variable = { which = br_reinf_cv value = root.br_power_level }
    while = {
        count = br_reinf_cv
        create_ship = { name = random design = event_target:br_player_corvette }
    }
    clear_variable = br_reinf_cv

    # 驱逐：level≥3
    if = {
        limit = { root = { check_variable = { which = br_power_level value >= 3 } } }
        set_variable = { which = br_reinf_dd value = root.br_power_level }
        divide_variable = { which = br_reinf_dd value = 2 }
        while = {
            count = br_reinf_dd
            create_ship = { name = random design = event_target:br_player_destroyer }
        }
        clear_variable = br_reinf_dd
    }

    # 巡洋：level≥5
    if = {
        limit = { root = { check_variable = { which = br_power_level value >= 5 } } }
        set_variable = { which = br_reinf_cr value = root.br_power_level }
        divide_variable = { which = br_reinf_cr value = 4 }
        while = {
            count = br_reinf_cr
            create_ship = { name = random design = event_target:br_player_cruiser }
        }
        clear_variable = br_reinf_cr
    }

    # 战列：level≥8
    if = {
        limit = { root = { check_variable = { which = br_power_level value >= 8 } } }
        set_variable = { which = br_reinf_bb value = root.br_power_level }
        divide_variable = { which = br_reinf_bb value = 8 }
        while = {
            count = br_reinf_bb
            create_ship = { name = random design = event_target:br_player_battleship }
        }
        clear_variable = br_reinf_bb
    }
}
