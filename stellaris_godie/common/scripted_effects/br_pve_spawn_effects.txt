# =========================================================================
# PVE 年度突击刷敌链
# 调用链: br.850(on_yearly_pulse_country, 保存玩家舰船设计)
#          → 延迟触发 br.830/br.840 → br_pve_reinforce_events.txt:96
#          → br_reinforce_pve_fleet → br_spawn_mixed(40%) / br_spawn_pure_*(各15%)
# 舰船设计来自 event_target:br_player_*（玩家舰船保存，含生物船兼容修复）
# 数量由 br_total_intensity 驱动：护卫=intensity/10 保底1，驱逐/巡洋/战列逐级减半
# =========================================================================

# ============================================================
# 年度增援调度 — 随机编队类型
# 在已有舰队作用域内调用
# ============================================================
br_reinforce_pve_fleet = {
    random_list = {
        40 = { br_spawn_mixed = yes }
        15 = { br_spawn_pure_corvette = yes }
        15 = { br_spawn_pure_destroyer = yes }
        15 = { br_spawn_pure_cruiser = yes }
        15 = { br_spawn_pure_battleship = yes }
    }
}

# ============================================================
# 混合编队 — 四舰种混出
# ============================================================

br_spawn_mixed = {
    # 数量由 br_total_intensity 驱动（替代累计计数 br_reinf_1），保底 1 艘护卫
    #复制变量进行检查
    set_variable = { which = br_total_intensity_effect value = event_target:br_pve_enemy_country.br_total_intensity }
    set_variable = { which = br_reinf_1 value = event_target:br_pve_enemy_country.br_total_intensity }
    divide_variable = { which = br_reinf_1 value = 10 }
    if = {
        limit = { check_variable = { which = br_reinf_1 value < 1 } }
        set_variable = { which = br_reinf_1 value = 1 }
    }

    # 护卫
    while = {
        count = br_reinf_1
        create_ship = { name = random design = event_target:br_player_corvette }
    }

    # 驱逐：intensity >= 200 时出现，数量 = 护卫 / 2
    if = {
        limit = { check_variable = { which = br_total_intensity_effect value >= 200 } }
        set_variable = { which = br_reinf_2 value = br_reinf_1 }
        divide_variable = { which = br_reinf_2 value = 2 }
        while = {
            count = br_reinf_2
            create_ship = { name = random design = event_target:br_player_destroyer }
        }
    }

    # 巡洋：intensity >= 400 时出现，数量 = 驱逐 / 2
    if = {
        limit = { check_variable = { which = br_total_intensity_effect value >= 400 } }
        set_variable = { which = br_reinf_3 value = br_reinf_2 }
        divide_variable = { which = br_reinf_3 value = 2 }
        while = {
            count = br_reinf_3
            create_ship = { name = random design = event_target:br_player_cruiser }
        }
    }

    # 战列：intensity >= 600 时出现，数量 = 巡洋 / 2
    if = {
        limit = { check_variable = { which = br_total_intensity_effect value >= 600 } }
        set_variable = { which = br_reinf_4 value = br_reinf_3 }
        divide_variable = { which = br_reinf_4 value = 2 }
        while = {
            count = br_reinf_4
            create_ship = { name = random design = event_target:br_player_battleship }
        }
    }

    clear_variable = br_reinf_1
    clear_variable = br_reinf_2
    clear_variable = br_reinf_3
    clear_variable = br_reinf_4
}

# ============================================================
# 纯编队效果 — 累计模式，每次调用+1，全出一种船
# 计数器挂在 event_target:br_pve_enemy_country 上
# 无对应玩家设计时 fallback 到混合编队
# ============================================================

# 纯护卫：累计模式，上限 100（变量挂 br_pve_enemy_country）
br_spawn_pure_corvette = {
    if = {
        limit = { exists = event_target:br_player_corvette }
        event_target:br_pve_enemy_country = {
            change_variable = { which = br_pure_corvette_count value = 1 }
            if = {
                limit = { check_variable = { which = br_pure_corvette_count value > 100 } }
                set_variable = { which = br_pure_corvette_count value = 100 }
            }
            set_variable = { which = br_tmp_spawn value = br_pure_corvette_count }
        }
        set_variable = { which = br_tmp_spawn value = event_target:br_pve_enemy_country.br_tmp_spawn }
        while = {
            count = br_tmp_spawn
            create_ship = { name = random design = event_target:br_player_corvette }
        }
        clear_variable = br_tmp_spawn
        event_target:br_pve_enemy_country = { clear_variable = br_tmp_spawn }
    }
    else = {
        br_spawn_mixed = yes
    }
}
