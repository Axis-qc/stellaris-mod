# =========================================================================
# PVE天灾杂兵舰队 — 外圈使用，不出女王
# 调用链: br_spawn_crisis_grunt_fleet → br_crisis_create_grunt_fleet
#          → br_crisis_grunt_small/medium/large_spawn
# 数量由 br_pve_enemy_country.br_grunt_spawn_count 控制，每次四次调用+1
# 与 br_crisis_spawn_effects.txt 的区别：无纯种/女王分支，混合编队中无女王
# =========================================================================

# [调度器]刷船（br_fleet_created 防重复）
br_crisis_create_grunt_fleet = {
    if = {
        limit = { not = { has_fleet_flag = br_fleet_created } }
        event_target:br_pve_enemy_country = {
            change_variable = { which = br_grunt_spawn_count value = 0.25 }
            if = {
                limit = { check_variable = { which = br_grunt_spawn_count value > 100 } }
                set_variable = { which = br_grunt_spawn_count value = 100 }
            }
        }
        set_variable = { which = br_crisis_small value = event_target:br_pve_enemy_country.br_grunt_spawn_count }
        set_variable = { which = br_crisis_medium value = 0 }
        set_variable = { which = br_crisis_large value = 0 }
        br_crisis_grunt_small_spawn = yes
        if = {
            limit = {
                check_variable = { which = br_crisis_small value > 10 }
            }
            divide_variable = { which = br_crisis_small value = 10 }
            change_variable = { which = br_crisis_medium value = br_crisis_small }
        }
        br_crisis_grunt_medium_spawn = yes
        br_crisis_grunt_large_spawn = yes
        set_fleet_flag = br_fleet_created
    }
}
