# 利维坦·噬星者 技能效果（THIS = 目标恒星，from = 施放舰队）
# 施放国先存成事件目标 lev_caster，避免嵌套作用域把 from 链挤偏

# 光谱降 1 级
lev_stellarite_spectrum_down = {
    if = { limit = { is_star_class = sc_b }
        solar_system = { set_star_class = sc_a }
        change_pc = pc_a_star
    }
    else_if = { limit = { is_star_class = sc_a }
        solar_system = { set_star_class = sc_f }
        change_pc = pc_f_star
    }
    else_if = { limit = { is_star_class = sc_f }
        solar_system = { set_star_class = sc_g }
        change_pc = pc_g_star
    }
    else_if = { limit = { is_star_class = sc_g }
        solar_system = { set_star_class = sc_k }
        change_pc = pc_k_star
    }
    else_if = { limit = { is_star_class = sc_k }
        solar_system = { set_star_class = sc_m }
        change_pc = pc_m_star
    }
    else_if = { limit = { is_star_class = sc_m }
        solar_system = { set_star_class = sc_m_giant }
        change_pc = pc_m_giant_star
    }
}

# $VALUE$ 常态值、$VALUE_HUNGRY$ 饥饿态值、$VAR$ 写入的国家变量名
lev_stellarite_shot_prepare = {
    if = {
        limit = { owner = { check_variable = { which = lev_stellarite_energy value <= 2 } } }
        set_variable = { which = lev_stellarite_shot_tmp value = value:$VALUE_HUNGRY$ }
    }
    else = {
        set_variable = { which = lev_stellarite_shot_tmp value = value:$VALUE$ }
    }
    owner = {
        set_variable = { which = $VAR$ value = prev.lev_stellarite_shot_tmp }
    }
}

# 吞噬恒星
lev_stellarite_absorb_apply = {
    from.owner = { save_event_target_as = lev_caster }

    lev_stellarite_spectrum_down = yes

    solar_system = {
        every_system_planet = {
            limit = { is_star = no  is_asteroid = no }
            spawn_planet_temperature_effect = yes
            decrease_planet_temperature = yes
            add_modifier = { modifier = stellarite_low_temp  days = -1 }
        }
    }
    spawn_sun_temperature_effect = yes

    event_target:lev_caster = {
        if = { limit = { check_variable = { which = lev_stellarite_energy value < 10 } }
            change_variable = { which = lev_stellarite_energy value = 1 }
        }
    }

    # 对他国星系：该星系所有者对我们 -100 观感
    solar_system = {
        if = {
            limit = {
                exists = space_owner
                NOT = { space_owner = { is_same_value = event_target:lev_caster } }
            }
            space_owner = {
                add_opinion_modifier = {
                    who = event_target:lev_caster
                    modifier = lev_star_devoured
                }
            }
        }
    }
}

# 重燃恒星
lev_stellarite_ignite_apply = {
    from.owner = { save_event_target_as = lev_caster }

    lev_stellarite_spectrum_up = yes

    # 冰封星救赎须在升温前，否则 pc_frozen 会先被升温链改掉
    solar_system = {
        every_system_planet = {
            limit = { is_planet_class = pc_frozen  is_colony = yes }
            change_pc = pc_gaia
        }
        every_system_planet = {
            limit = { is_star = no  is_asteroid = no }
            spawn_planet_temperature_effect = yes
            increase_planet_temperature = yes
            remove_modifier = stellarite_low_temp
        }
    }
    spawn_sun_temperature_effect = yes

    event_target:lev_caster = {
        if = { limit = { check_variable = { which = lev_stellarite_energy value > 0 } }
            change_variable = { which = lev_stellarite_energy value = -1 }
        }
    }
}

# 恒星光束：目标星系内全部敌舰（远程打击，与舰队自身所在星系无关）
# THIS = 施放舰队，目标星系取全局事件目标 lev_beam_target
# 由 lev.106 在蓄能期满时调用；旧契约（打 solar_system 即舰队所在地）随远程化废弃
lev_stellarite_beam_apply = {
    owner = { save_event_target_as = lev_caster }

    lev_stellarite_shot_prepare = {
        VALUE = lev_beam_damage
        VALUE_HUNGRY = lev_beam_damage_hungry
        VAR = lev_stellarite_shot_damage
    }

    # 打的是当初选定的目标星系，不是舰队此刻所在地
    event_target:lev_beam_target = {
        every_fleet_in_system = {
            limit = {
                controller = { is_hostile = event_target:lev_caster }
            }
            every_owned_ship = {
                damage_ship = {
                    amount = event_target:lev_caster.lev_stellarite_shot_damage
                    attacker = event_target:lev_caster
                }
            }
        }
    }

    event_target:lev_caster = {
        if = { limit = { check_variable = { which = lev_stellarite_energy value > 0 } }
            change_variable = { which = lev_stellarite_energy value = -1 }
        }
    }
}

# 腹中日轮：本星系敌舰伤害与行星破坏
lev_stellarite_sunburst_apply = {
    from.owner = { save_event_target_as = lev_caster }

    lev_stellarite_shot_prepare = {
        VALUE = lev_sun_damage
        VALUE_HUNGRY = lev_sun_damage_hungry
        VAR = lev_stellarite_shot_damage
    }
    # 行星破坏不吃饥饿加成
    lev_stellarite_shot_prepare = {
        VALUE = lev_sun_devastation
        VALUE_HUNGRY = lev_sun_devastation
        VAR = lev_stellarite_shot_dev
    }

    solar_system = {
        every_fleet_in_system = {
            limit = {
                controller = { is_hostile = event_target:lev_caster }
            }
            every_owned_ship = {
                damage_ship = {
                    amount = event_target:lev_caster.lev_stellarite_shot_damage
                    attacker = event_target:lev_caster
                }
            }
        }
        every_system_planet = {
            limit = { is_star = no  is_asteroid = no }
            add_planet_devastation = event_target:lev_caster.lev_stellarite_shot_dev
        }
    }

    lev_stellarite_sunburst_vfx = yes

    event_target:lev_caster = {
        if = { limit = { check_variable = { which = lev_stellarite_energy value >= 5 } }
            change_variable = { which = lev_stellarite_energy value = -5 }
        }
        else = {
            set_variable = { which = lev_stellarite_energy value = 0 }
        }
    }
}

# 潜入恒星
lev_stellarite_submerge_apply = {
    # THIS = 噬星者舰队
    # 订单在飞标记：期间封掉自动化动作的选目标权重
    set_timed_fleet_flag = { flag = lev_stellarite_absorbing days = 190 }
    every_owned_ship = {
        limit = { is_ship_size = lev_stellarite }
        add_modifier = { modifier = lev_stellarite_submerged_regen days = 190 }
    }
}

# 上浮：订单取消或结算时收回回复
lev_stellarite_surface_apply = {
    remove_fleet_flag = lev_stellarite_absorbing
    every_owned_ship = {
        limit = { is_ship_size = lev_stellarite }
        remove_modifier = lev_stellarite_submerged_regen
    }
}
