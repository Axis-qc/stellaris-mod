# [调度器] 建舰队 + random_list 选编队类型
br_crisis_create_fleet = {
    # 编队类型随机：10纯肃正 / 10纯恶魔 / 10纯虫子 / 10纯女王 / 40混合
    if = {
        limit = {
            not = {
                has_fleet_flag = br_crisis_spawn_pure_contigency
                has_fleet_flag = br_crisis_spawn_pure_extradimensional
                has_fleet_flag = br_crisis_spawn_pure_swarm
                has_fleet_flag = br_crisis_spawn_pure_queen
                has_fleet_flag = br_crisis_spawn_mixed
            }
        }
        random_list = {
            10 = { set_fleet_flag = br_crisis_spawn_pure_contigency }
            10 = { set_fleet_flag = br_crisis_spawn_pure_extradimensional }
            10 = { set_fleet_flag = br_crisis_spawn_pure_swarm }
            10 = { set_fleet_flag = br_crisis_spawn_pure_queen }
            40 = { set_fleet_flag = br_crisis_spawn_mixed }
        }
    }
    #启动三次，先刷大再刷小
    # ==================== Phase 3：刷 Large ====================
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_contigency_2 }
        set_variable = { which = br_crisis_small value = 0 }
        set_variable = { which = br_crisis_medium value = 0 }
        multiply_variable = { which = br_crisis_large value = 17 }
        br_crisis_spawn_pure_contigency = yes
        remove_fleet_flag = br_crisis_spawn_pure_contigency_2
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_extradimensional_2 }
        set_variable = { which = br_crisis_small value = 0 }
        set_variable = { which = br_crisis_medium value = 0 }
        multiply_variable = { which = br_crisis_large value = 6 }
        br_crisis_spawn_pure_extradimensional = yes
        remove_fleet_flag = br_crisis_spawn_pure_extradimensional_2
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_swarm_2 }
        set_variable = { which = br_crisis_small value = 0 }
        set_variable = { which = br_crisis_medium value = 0 }
        multiply_variable = { which = br_crisis_large value = 15 }
        br_crisis_spawn_pure_swarm = yes
        remove_fleet_flag = br_crisis_spawn_pure_swarm_2
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_queen_2 }
        set_variable = { which = br_crisis_small value = 0 }
        set_variable = { which = br_crisis_medium value = 0 }
        # queen large ×1
        br_crisis_spawn_pure_queen = yes
        remove_fleet_flag = br_crisis_spawn_pure_queen_2
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_mixed_2 }
        set_variable = { which = br_crisis_small value = 0 }
        set_variable = { which = br_crisis_medium value = 0 }
        # large_spawn 内部自己算倍率
        br_crisis_large_spawn = yes
        remove_fleet_flag = br_crisis_spawn_mixed_2
    }
    # # ==================== Phase 2：刷 Medium ====================
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_contigency_1 }
        set_variable = { which = br_crisis_small value = 0 }
        multiply_variable = { which = br_crisis_medium value = 3 }
        set_variable = { which = br_crisis_large value = 0 }
        br_crisis_spawn_pure_contigency = yes
        set_fleet_flag = br_crisis_spawn_pure_contigency_2
        remove_fleet_flag = br_crisis_spawn_pure_contigency_1
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_extradimensional_1 }
        set_variable = { which = br_crisis_small value = 0 }
        multiply_variable = { which = br_crisis_medium value = 2 }
        set_variable = { which = br_crisis_large value = 0 }
        br_crisis_spawn_pure_extradimensional = yes
        set_fleet_flag = br_crisis_spawn_pure_extradimensional_2
        remove_fleet_flag = br_crisis_spawn_pure_extradimensional_1
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_swarm_1 }
        set_variable = { which = br_crisis_small value = 0 }
        multiply_variable = { which = br_crisis_medium value = 3 }
        set_variable = { which = br_crisis_large value = 0 }
        br_crisis_spawn_pure_swarm = yes
        set_fleet_flag = br_crisis_spawn_pure_swarm_2
        remove_fleet_flag = br_crisis_spawn_pure_swarm_1
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_queen_1 }
        set_variable = { which = br_crisis_small value = 0 }
        # queen medium ×1
        set_variable = { which = br_crisis_large value = 0 }
        br_crisis_spawn_pure_queen = yes
        set_fleet_flag = br_crisis_spawn_pure_queen_2
        remove_fleet_flag = br_crisis_spawn_pure_queen_1
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_mixed_1 }
        set_variable = { which = br_crisis_small value = 0 }
        # medium_spawn 内部自己算倍率
        set_variable = { which = br_crisis_large value = 0 }
        br_crisis_medium_spawn = yes
        set_fleet_flag = br_crisis_spawn_mixed_2
        remove_fleet_flag = br_crisis_spawn_mixed_1
    }
    # ==================== Phase 1：刷 Small ====================
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_contigency }
        multiply_variable = { which = br_crisis_small value = 10 }
        set_variable = { which = br_crisis_medium value = 0 }
        set_variable = { which = br_crisis_large value = 0 }
        br_crisis_spawn_pure_contigency = yes
        set_fleet_flag = br_crisis_spawn_pure_contigency_1
        remove_fleet_flag = br_crisis_spawn_pure_contigency
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_extradimensional }
        multiply_variable = { which = br_crisis_small value = 6 }
        multiply_variable = { which = br_crisis_medium value = 0 }
        multiply_variable = { which = br_crisis_large value = 0 }
        br_crisis_spawn_pure_extradimensional = yes
        set_fleet_flag = br_crisis_spawn_pure_extradimensional_1
        remove_fleet_flag = br_crisis_spawn_pure_extradimensional
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_swarm }
        multiply_variable = { which = br_crisis_small value = 6 }
        multiply_variable = { which = br_crisis_medium value = 0 }
        multiply_variable = { which = br_crisis_large value = 0 }
        br_crisis_spawn_pure_swarm = yes
        set_fleet_flag = br_crisis_spawn_pure_swarm_1
        remove_fleet_flag = br_crisis_spawn_pure_swarm
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_pure_queen }
        multiply_variable = { which = br_crisis_medium value = 0 }
        multiply_variable = { which = br_crisis_large value = 0 }
        br_crisis_spawn_pure_queen = yes
        set_fleet_flag = br_crisis_spawn_pure_queen_1
        remove_fleet_flag = br_crisis_spawn_pure_queen
    }
    if = {
        limit = { has_fleet_flag = br_crisis_spawn_mixed }
        multiply_variable = { which = br_crisis_medium value = 0 }
        multiply_variable = { which = br_crisis_large value = 0 }
        br_crisis_small_spawn = yes
        set_fleet_flag = br_crisis_spawn_mixed_1
        remove_fleet_flag = br_crisis_spawn_mixed
    }
}
