# ============================================================

# 
# 作用域链：
#   event_target:br_pve_enemy_country = { br_spawn_crisis_fleet = yes }
#     → 国家作用域 调用 br_spawn_crisis_fleet
#     → 读 event_target:br_spawn_tmp 的 ring flag
#     → 通过参数($small$/$medium$/$large$)传给 br_crisis_create_fleet
#     → create_fleet 内部刷船
# 
# 所有数量全部通过 scripted_effect 参数(文本替换)传递，不依赖运行时变量
# 避免跨作用域读变量的问题
# 
# ── 触发顺序 ──
# 1. br_spawn_crisis_fleet         ← 入口：按圈层算数量
# 2. br_crisis_create_fleet        ← 调度器：建舰队 + random_list 选编队
# 3. 编队效果（被 random_list 调用）
#    ├─ br_crisis_spawn_pure_contigency       (10%)
#    ├─ br_crisis_spawn_pure_extradimensional (10%)
#    ├─ br_crisis_spawn_pure_swarm            (10%)
#    ├─ br_crisis_spawn_pure_queen            (10%)
#    └─ 混合编队 → 基础 spawn (60%)
#       ├─ br_crisis_small_spawn
#       ├─ br_crisis_medium_spawn
#       └─ br_crisis_large_spawn
# ============================================================

# ============================================================
# [调度器] 创建舰队 + 编队类型选择
# 
# 被 br_spawn_crisis_fleet 调用
# 在国家作用域下执行
# $small$/$medium$/$large$ 是文本替换参数，不受作用域影响
# ============================================================
br_crisis_create_fleet = {
    create_fleet = {
        name = "BR_PVE_CRISIS_FLEET"
        effect = {
            set_owner = event_target:br_pve_enemy_country
            set_location = { target = event_target:br_spawn_tmp distance = 50 angle = random }

            # 编队类型随机：10%纯肃正 / 10%纯恶魔 / 10%纯虫子 / 10%纯女王 / 60%混合
            random_list = {
                10 = { br_crisis_spawn_pure_contigency = { small = $small$ medium = $medium$ large = $large$ } }
                10 = { br_crisis_spawn_pure_extradimensional = { small = $small$ medium = $medium$ large = $large$ } }
                10 = { br_crisis_spawn_pure_swarm = { small = $small$ medium = $medium$ large = $large$ } }
                10 = { br_crisis_spawn_pure_queen = { small = $small$ medium = $medium$ large = $large$ } }
                60 = {
                    # 混合编队：各势力按比例混抽
                    br_crisis_small_spawn = { COUNT = $small$ }
                    br_crisis_medium_spawn = { COUNT = $medium$ }
                    br_crisis_large_spawn = { COUNT = $large$ }
                }
            }
        }
    }
}

# ============================================================
# [纯肃正编队] 10%
# 肃正只有 Corvette 和 Battleship 两种尺寸
# 中型配额并入大型，全出 Battleship
# ============================================================
br_crisis_spawn_pure_contigency = {
    while = { count = $small$ create_ship = { name = random design = "NAME_BR_AI_Corvette" } }
    while = { count = $medium$ create_ship = { name = random design = "NAME_BR_AI_Battleship" } }
    while = { count = $large$ create_ship = { name = random design = "NAME_BR_AI_Battleship" } }
}

# ============================================================
# [纯恶魔编队] 10%
# 三色恶魔均分
# ============================================================
br_crisis_spawn_pure_extradimensional = {
    while = { count = $small$
        random_list = {
            33 = { create_ship = { name = random design = "NAME_BR_Wraith" } }
            33 = { create_ship = { name = random design = "NAME_BR_Predator" } }
            34 = { create_ship = { name = random design = "NAME_BR_Obliterator" } }
        }
    }
    while = { count = $medium$
        random_list = {
            33 = { create_ship = { name = random design = "NAME_BR_Phantom" } }
            33 = { create_ship = { name = random design = "NAME_BR_Assassin" } }
            34 = { create_ship = { name = random design = "NAME_BR_Annihilator" } }
        }
    }
    while = { count = $large$
        random_list = {
            33 = { create_ship = { name = random design = "NAME_BR_Revenant" } }
            33 = { create_ship = { name = random design = "NAME_BR_Huntress" } }
            34 = { create_ship = { name = random design = "NAME_BR_Eradicator" } }
        }
    }
}

# ============================================================
# [纯虫子编队] 10%
# 单色三尺寸
# ============================================================
br_crisis_spawn_pure_swarm = {
    while = { count = $small$ create_ship = { name = random design = "NAME_BR_Swarm_Small" } }
    while = { count = $medium$ create_ship = { name = random design = "NAME_BR_Swarm_Large" } }
    while = { count = $large$ create_ship = { name = random design = "NAME_BR_Swarm_Carrier" } }
}

# ============================================================
# [纯女王编队] 10%
# 小型配额也造 Harrier，防止边缘星系抽出女王编队变空
# ============================================================
br_crisis_spawn_pure_queen = {
    while = { count = $small$ create_ship = { name = random design = "NAME_BR_Synth_Harrier" } }
    while = { count = $medium$ create_ship = { name = random design = "NAME_BR_Synth_Harrier" } }
    while = { count = $large$ create_ship = { name = random design = "NAME_BR_Synth_Royal_Guard" } }
}

# ============================================================
# [基础 spawn] 各尺寸按势力混抽 — 用于混合编队 (60%)
# 保留原 random_list 比例不变
# ============================================================

# ---------- 小型舰 ----------
# 外圈：虫群小 / 恶魔三色小 / 肃正小(AI)
# 不出女王（Harrier 带X槽前期太强）
br_crisis_small_spawn = {
    while = {
        count = $COUNT$
        random_list = {
            33 = { create_ship = { name = random design = "NAME_BR_Swarm_Small" } }
            33 = {
                random_list = {
                    33 = { create_ship = { name = random design = "NAME_BR_Wraith" } }
                    33 = { create_ship = { name = random design = "NAME_BR_Predator" } }
                    34 = { create_ship = { name = random design = "NAME_BR_Obliterator" } }
                }
            }
            34 = { create_ship = { name = random design = "NAME_BR_AI_Corvette" } }
        }
    }
}

# ---------- 中型舰 ----------
# 中圈：虫群大(算中型) / 恶魔三色中 / 肃正中 / 女王Harrier
br_crisis_medium_spawn = {
    while = {
        count = $COUNT$
        random_list = {
            25 = { create_ship = { name = random design = "NAME_BR_Swarm_Large" } }
            25 = {
                random_list = {
                    33 = { create_ship = { name = random design = "NAME_BR_Phantom" } }
                    33 = { create_ship = { name = random design = "NAME_BR_Assassin" } }
                    34 = { create_ship = { name = random design = "NAME_BR_Annihilator" } }
                }
            }
            25 = { create_ship = { name = random design = "NAME_BR_AI_Battleship" } }
            25 = { create_ship = { name = random design = "NAME_BR_Synth_Harrier" } }
        }
    }
}
