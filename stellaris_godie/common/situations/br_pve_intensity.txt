# ====================================================================
# PVE 天灾强度局势（每个玩家独立）
# 每月 +0.1 自动增长，永久不消失
# 所有玩家的进度累加 = 最终天灾强度，用于计算刷敌数量
# 玩家可通过事件/决议影响自己的进度（加速或压制）
# ====================================================================
situation_br_pve_intensity = {
	permanent = yes
	show_in_outliner = yes
	picture = GFX_evt_wormhole
	situation_log_category = empire_concerns
	category = neutral
	initial_progress = 0
	# 动态上限：敌方存在感同步抬高局势终点（裂隙×100 + 场上敌船×3）
	total_progress = {
		base = 10000
		add = value:br_pve_max_from_enemy
	}
	override_finisher_title = br_situation_title
	override_finisher_desc = br_pve_desc
	# override_progress_current_desc = br_pve_intensity_progress_current

	abort_trigger = {
		NOT = {
			exists = owner
		}
	}
	on_start = {
		# 首次 approach 豁免：局内第一次选中（含开局默认）不设切换冷却
		owner = {
			set_country_flag = br_approach_first_select
		}
	}
	on_fail = {
	}
	on_progress_complete = {
	}
	# 月度增长：底盘1.5 + 裂隙(每道×0.3) + 敌船(每艘×0.02) + BOSS降临+3
	# 姿态为总流速百分比乘法：压制×0.7 / 催化×1.5（置于所有加项之后执行）
	monthly_progress = {
		base = 1
		modifier = {
			add = value:br_pve_rift_growth
			desc = br_pve_intensity_rifts
		}
		modifier = {
			add = value:br_pve_enemy_ship_growth
			desc = br_pve_intensity_ships
		}
		modifier = {
			add = 3
			desc = br_pve_intensity_boss
			has_global_flag = br_boss_spawned
		}
		modifier = {
			desc = br_pve_intensity_approach_suppress
			mult = 0.5
			current_situation_approach = br_pve_intensity_suppress
		}
		modifier = {
			desc = br_pve_intensity_approach_accelerate
			mult = 2
			current_situation_approach = br_pve_intensity_accelerate
		}
	}
	approach = {
		name = br_pve_intensity_suppress
		icon = GFX_situation_approach_nomad_neutral
		icon_background = GFX_situation_approach_bg_yellow
		default = yes
		on_select = {
			custom_tooltip = "br_pve_intensity_suppress_tt"
			owner = {
				if = {
					limit = {
						has_country_flag = br_approach_first_select
					}
					remove_country_flag = br_approach_first_select
				}
				else = {
					set_timed_country_flag = {
						flag = br_approach_switch_cooldown
						days = 360
					}
				}
			}
		}
		# 压制：低流速 + 稳定虚空消耗（资源随局势阶段升级）
		resources = {
			category = situations
			upkeep = {
				energy = 20
			}
			upkeep = {
				trigger = {
					current_stage = stage_high
				}
				alloys = 15
			}
			upkeep = {
				trigger = {
					current_stage = stage_extreme
				}
				volatile_motes = 3
			}
		}
		allow = {
			custom_tooltip = {
				fail_text = br_pve_intensity_cooldown_tt
				OR = {
					current_situation_approach = br_pve_intensity_suppress
					owner = {
						NOT = {
							has_country_flag = br_approach_switch_cooldown
						}
					}
				}
			}
		}
	}
	approach = {
		name = br_pve_intensity_neutral
		icon = GFX_situation_approach_nomad_good
		icon_background = GFX_situation_approach_bg_green
		on_select = {
			custom_tooltip = "br_pve_intensity_neutral_tt"
			owner = {
				if = {
					limit = {
						has_country_flag = br_approach_first_select
					}
					remove_country_flag = br_approach_first_select
				}
				else = {
					set_timed_country_flag = {
						flag = br_approach_switch_cooldown
						days = 360
					}
				}
			}
		}
		# 均衡：不增不减，纯基线
		allow = {
			custom_tooltip = {
				fail_text = br_pve_intensity_cooldown_tt
				OR = {
					current_situation_approach = br_pve_intensity_neutral
					owner = {
						NOT = {
							has_country_flag = br_approach_switch_cooldown
						}
					}
				}
			}
		}
	}
	approach = {
		name = br_pve_intensity_accelerate
		icon = GFX_situation_approach_nomad_very_good
		icon_background = GFX_situation_approach_bg_green
		on_select = {
			custom_tooltip = "br_pve_intensity_accelerate_tt"
			add_situation_progress = 200
			owner = {
				if = {
					limit = {
						has_country_flag = br_approach_first_select
					}
					remove_country_flag = br_approach_first_select
				}
				else = {
					set_timed_country_flag = {
						flag = br_approach_switch_cooldown
						days = 360
					}
				}
			}
		}
		# 催化：高流速 + 选中当场推进进度（回报=战斗系统早已给出的更多刷怪）
		allow = {
			custom_tooltip = {
				fail_text = br_pve_intensity_cooldown_tt
				OR = {
					current_situation_approach = br_pve_intensity_accelerate
					owner = {
						NOT = {
							has_country_flag = br_approach_switch_cooldown
						}
					}
				}
			}
		}
	}
	# 纯视觉分阶段
	stages = {
		stage_low = {
			icon = GFX_situation_stage_1
			icon_background = GFX_situation_stage_frame_green
			section_weight = 25
			custom_tooltip = stage_low_desc
		}
		stage_medium = {
			icon = GFX_situation_stage_2
			icon_background = GFX_situation_stage_frame_yellow
			section_weight = 25
		}
		stage_high = {
			icon = GFX_situation_stage_3
			icon_background = GFX_situation_stage_frame_orange
			section_weight = 25
		}
		stage_extreme = {
			icon = GFX_situation_stage_4
			icon_background = GFX_situation_stage_frame_red
			section_weight = 25
		}
	}
}
