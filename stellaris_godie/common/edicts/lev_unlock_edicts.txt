# 利维坦解锁法令，点法令置 flag 并经 lev.330 发放一艘到首都，被摧毁后年度补发
# 延迟 1 天给 refresh_auto_generated_ship_designs 留结算窗口，否则 create_ship 取不到设计
# potential 的 NOT 已解锁判断使解锁后法令自动隐藏

# 噬星者
lev_unlock_stellarite = {
    icon = "GFX_edict_type_policy"
    length = 1
    resources = {
        category = edicts
        cost = {
        }
    }
    potential = {
        NOT = { has_country_flag = lev_stellarite_unlocked }
    }
    allow = {
        always = yes
    }
    effect = {
        set_country_flag = lev_stellarite_unlocked
        refresh_auto_generated_ship_designs = yes
        lev_growth_init = { TYPE = stellarite }
        country_event = {
            id = lev.330
            days = 1
        }
    }
    ai_weight = {
        weight = 0
    }
}
