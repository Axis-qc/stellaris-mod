# 利维坦解锁法令
# 解锁途径：点法令 → set_country_flag lev_<TYPE>_unlocked → 对应舰种在星港解锁建造
# 不直接发船；折跃发放走 lev_stellarite_debug_edicts.txt 的调试法令（br_debug_mode 门控）
# potential 带 NOT 已解锁判断，解锁后法令自动隐藏，防重复点击

# 噬星者
lev_unlock_stellarite = {
    icon = "GFX_edict_type_policy"
    length = 1
    resources = {
        category = edicts
        cost = {
        }
    }
    potential = {
        NOT = { has_country_flag = lev_stellarite_unlocked }
    }
    allow = {
        always = yes
    }
    effect = {
        set_country_flag = lev_stellarite_unlocked
        refresh_auto_generated_ship_designs = yes
        lev_growth_init = { TYPE = stellarite }
    }
    ai_weight = {
        weight = 0
    }
}
