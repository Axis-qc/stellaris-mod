# 利维坦法令：每只两条（只解锁 / 解锁并发放），均为测试法令，须先开 br_test_toggle_debug

# 噬星者
lev_unlock_stellarite = {
	icon = "GFX_edict_type_policy"
	length = 1
	resources = {
		category = edicts
		cost = {
		}
	}
	potential = {
		has_country_flag = br_debug_mode
		NOT = { has_country_flag = lev_stellarite_unlocked }
	}
	allow = {
		always = yes
	}
	effect = {
		set_country_flag = lev_stellarite_unlocked
		refresh_auto_generated_ship_designs = yes
		lev_growth_init = { TYPE = stellarite }
	}
	ai_weight = {
		weight = 0
	}
}

# 解锁并发放
lev_test_spawn_stellarite = {
	icon = "GFX_edict_type_policy"
	length = 1
	resources = {
		category = edicts
		cost = {
		}
	}
	potential = {
		has_country_flag = br_debug_mode
	}
	allow = {
		always = yes
	}
	effect = {
		set_country_flag = lev_stellarite_unlocked
		refresh_auto_generated_ship_designs = yes
		if = {
			limit = {
				NOT = {
					is_variable_set = lev_stellarite_energy
				}
			}
			set_variable = {
				which = lev_stellarite_energy
				value = 0
			}
		}
		# 成长值系统初始化（含成长值 / 等级 / 营养档位 / 阈值 / 各档消耗收益）
		lev_growth_init = { TYPE = stellarite }
		country_event = {
			id = lev.3201
			days = 1
		}
	}
	ai_weight = {
		weight = 0
	}
}
