# 激活即生效，length = 1 瞬间结束，可反复点击切换。
# 以下法令仅在 br_debug_mode 激活时可见。
# 全局调试开关 — 始终可见，用于打开/关闭调试模式
br_test_toggle_debug = {
	icon = "GFX_edict_type_policy"
	length = 1
	resources = {
		category = edicts
		cost = {
		}
	}
	potential = {
		always = yes
	}
	allow = {
		always = yes
	}
	effect = {
		if = {
			limit = {
				has_country_flag = br_debug_mode
			}
			remove_country_flag = br_debug_mode
		}
		else = {
			set_country_flag = br_debug_mode
		}
	}
	ai_weight = {
		weight = 0
	}
}

# 测试法令 — 直接触发新周目（模拟BOSS击杀胜利）
br_test_trigger_cycle = {
	icon = "GFX_edict_type_policy"
	length = 1
	resources = {
		category = edicts
		cost = {
		}
	}
	potential = {
		has_country_flag = br_debug_mode
	}
	allow = {
		always = yes
	}
	effect = {
		set_global_flag = br_pve_victory
		every_country = {
			limit = {
				has_country_flag = br_player_country
			}
			set_country_flag = br_pve_victory
		}
		if = {
			limit = {
				exists = event_target:br_pve_alliance_leader
			}
			event_target:br_pve_alliance_leader = {
				country_event = {
					id = br.950
				}
			}
		}
		every_country = {
			limit = {
				has_country_flag = br_player_country
				NOT = {
					is_same_value = event_target:br_pve_alliance_leader
				}
			}
			country_event = {
				id = br.902
			}
		}
	}
	ai_weight = {
		weight = 0
	}
}
