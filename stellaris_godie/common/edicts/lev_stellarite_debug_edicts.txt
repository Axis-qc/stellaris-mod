# 利维坦·噬星者 调试法令
# 门控复用 br_debug_mode；length = 1 即点即生效，可反复点

# 发放一艘到首都
lev_test_spawn_stellarite = {
    icon = "GFX_edict_type_policy"
    length = 1
    resources = {
        category = edicts
        cost = {
        }
    }
    potential = {
        has_country_flag = br_debug_mode
    }
    allow = {
        always = yes
    }
    effect = {
        set_country_flag = lev_stellarite_unlocked
        refresh_auto_generated_ship_designs = yes
        if = {
            limit = {
                NOT = {
                    is_variable_set = lev_stellarite_energy
                }
            }
            set_variable = {
                which = lev_stellarite_energy
                value = 0
            }
        }
        # 成长值系统初始化（含成长值 / 等级 / 营养档位 / 阈值 / 各档消耗收益）
        lev_growth_init = { TYPE = stellarite }
        country_event = {
            id = lev.3201
            days = 1
        }
    }
    ai_weight = {
        weight = 0
    }
}

# 能量拉满 + 成长等级拉满
lev_test_energy_max = {
    icon = "GFX_edict_type_policy"
    length = 1
    resources = {
        category = edicts
        cost = {
        }
    }
    potential = {
        has_country_flag = br_debug_mode
    }
    allow = {
        always = yes
    }
    effect = {
        set_variable = {
            which = lev_stellarite_energy
            value = 10
        }
        # 成长等级直接给到封顶前的最高档（12 级）
        set_variable = {
            which = lev_stellarite_growth_level
            value = 12
        }
        set_variable = {
            which = lev_stellarite_growth
            value = 0
        }
        lev_growth_init = { TYPE = stellarite }
    }
    ai_weight = {
        weight = 0
    }
}
