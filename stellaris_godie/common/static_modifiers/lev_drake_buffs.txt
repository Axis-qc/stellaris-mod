# 按维度拆分（一个 add_modifier 只有一个 mult），数值由事件层
# mult = root.owner.value:lev_sk_* 动态决定（等级驱动）。
# 百分比字段写 0.01；小百分比（<10%）写 0.001，数值 = 百分比×10。
# 敌方 debuff 字段带负号、数值传正幅度。

lev_drake_sk1_fr = { ship_fire_rate_mult = 0.01 }

# sk2 龙翼护体（自身）
